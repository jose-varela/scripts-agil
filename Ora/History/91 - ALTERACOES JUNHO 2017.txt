CREATE OR REPLACE FUNCTION GET_AMOSTRA(V_POSTO       IN NUMBER,
                                       V_ATENDIMENTO IN NUMBER,
                                       V_CORREL      IN NUMBER) RETURN NUMBER IS RESULT NUMBER;
BEGIN
     SELECT NVL(MAX(AMOSTRA),0) INTO RESULT FROM AMOSTRAS
      WHERE POSTO = V_POSTO 
        AND ATENDIMENTO = V_ATENDIMENTO 
        AND CORREL = V_CORREL;

    RETURN(RESULT);
END GET_AMOSTRA;

CREATE TABLE EXAMES_SITUACOES
(POSTO                 NUMBER(3)    NOT NULL,
 ATENDIMENTO           NUMBER(6)    NOT NULL,
 CORREL                NUMBER(2)    NOT NULL,
 AMOSTRA               NUMBER(2)    NOT NULL,
 CORREL_SITUACAO       NUMBER(2)    NOT NULL,
 SITUACAO              VARCHAR2(1)  NOT NULL,
 DATA_SITUACAO         DATE         NOT NULL,
 USUARIO_SITUACAO      VARCHAR2(10) NOT NULL,
 ESTACAO_SITUACAO      VARCHAR2(20) NOT NULL
)
TABLESPACE LIS_DATA;

ALTER TABLE EXAMES_SITUACOES ADD CONSTRAINT EXAMES_SITUACOES_PK 
      PRIMARY KEY (POSTO, ATENDIMENTO, CORREL, CORREL_SITUACAO);
	  

ALTER TABLE EXAMES_SITUACOES  ADD CONSTRAINT EXAMES_SITUACOES_EXAMES_FK 
      FOREIGN KEY (POSTO, ATENDIMENTO, CORREL) REFERENCES EXAMES(POSTO, ATENDIMENTO, CORREL);


CREATE OR REPLACE FUNCTION GET_ULTIMO_CORREL_SITUACAO(V_POSTO       IN NUMBER,
                                                      V_ATENDIMENTO IN NUMBER,
                                                      V_CORREL      IN NUMBER) RETURN NUMBER IS RESULT NUMBER;
BEGIN
     SELECT NVL(MAX(ES.CORREL_SITUACAO)+1,0) INTO RESULT
	   FROM EXAMES_SITUACOES ES
       WHERE ES.POSTO = V_POSTO
         AND ES.ATENDIMENTO = V_ATENDIMENTO
         AND ES.CORREL = V_CORREL;
    RETURN(RESULT);
END GET_ULTIMO_CORREL_SITUACAO;
/

CREATE OR REPLACE PROCEDURE PRC_EXAMES_SITUACOES(V_POSTO           IN     EXAMES_SITUACOES.POSTO%TYPE,
                                                V_ATENDIMENTO      IN     EXAMES_SITUACOES.ATENDIMENTO%TYPE,
                                                V_CORREL           IN     EXAMES_SITUACOES.CORREL%TYPE,
                                                V_AMOSTRA          IN OUT EXAMES_SITUACOES.AMOSTRA%TYPE,
                                                V_CORREL_SITUACAO  IN OUT EXAMES_SITUACOES.CORREL_SITUACAO%TYPE,
                                                V_SITUACAO         IN     EXAMES_SITUACOES.SITUACAO%TYPE,
                                                V_DATA_SITUACAO    IN     EXAMES_SITUACOES.DATA_SITUACAO%TYPE,
                                                V_USUARIO_SITUACAO IN     EXAMES_SITUACOES.USUARIO_SITUACAO%TYPE,
                                                V_ESTACAO_SITUACAO IN     EXAMES_SITUACOES.ESTACAO_SITUACAO%TYPE,
                                                V_FUNCAO           IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT GET_AMOSTRA(V_POSTO, V_ATENDIMENTO, V_CORREL) INTO V_AMOSTRA FROM DUAL;
     SELECT GET_ULTIMO_CORREL_SITUACAO(V_POSTO, V_ATENDIMENTO, V_CORREL) INTO V_CORREL_SITUACAO FROM DUAL;

     INSERT INTO EXAMES_SITUACOES VALUES (V_POSTO, V_ATENDIMENTO, V_CORREL, V_AMOSTRA, V_CORREL_SITUACAO, V_SITUACAO, V_DATA_SITUACAO, V_USUARIO_SITUACAO, V_ESTACAO_SITUACAO);
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE EXAMES_SITUACOES 
	  WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND SITUACAO = V_SITUACAO;

  END IF;

  COMMIT WORK;

END PRC_EXAMES_SITUACOES;

CREATE OR REPLACE TRIGGER EXAMES_SITUACOES_TRG
 AFTER UPDATE ON EXAMES FOR EACH ROW
  WHEN (NEW.SITUACAO <> OLD.SITUACAO)
  
DECLARE
 V_AMOSTRA         EXAMES_SITUACOES.AMOSTRA%TYPE;
 V_CORREL_SITUACAO EXAMES_SITUACOES.CORREL_SITUACAO%TYPE;

BEGIN
     SELECT GET_AMOSTRA(:NEW.POSTO, :NEW.ATENDIMENTO, :NEW.CORREL) INTO V_AMOSTRA FROM DUAL;
     SELECT GET_ULTIMO_CORREL_SITUACAO(:NEW.POSTO, :NEW.ATENDIMENTO, :NEW.CORREL) INTO V_CORREL_SITUACAO FROM DUAL;

     INSERT INTO EXAMES_SITUACOES VALUES (:NEW.POSTO, :NEW.ATENDIMENTO, :NEW.CORREL, V_AMOSTRA, V_CORREL_SITUACAO, :NEW.SITUACAO, SYSDATE, 'AUTO', 'AUTO');
END;

CREATE OR REPLACE FUNCTION IS_URGENTE(V_POSTO       IN NUMBER,
                                      V_ATENDIMENTO IN NUMBER,
                                      V_CORREL      IN NUMBER) RETURN CHAR IS QT_URGENTE NUMBER; 
BEGIN
    SELECT COUNT(1) INTO QT_URGENTE FROM EXAMES_SITUACOES
     WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND SITUACAO = 'U';

    IF QT_URGENTE = 0 THEN
       RETURN 'N';
    ELSE
       RETURN 'S';
    END IF;
END IS_URGENTE;
