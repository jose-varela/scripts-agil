MINVALUE 0 
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;

CREATE TABLE PROCEDIMENTOS_URGENTES(
    ID_PROCEDIMENTO_URGENTE NUMBER(9) NOT NULL,
    MNEMONICO VARCHAR2(7) NOT NULL,
    POSTO_REA NUMBER(3) NOT NULL
)
TABLESPACE LIS_DATA;

-- fim bloco

ALTER TABLE PROCEDIMENTOS_URGENTES ADD CONSTRAINT PROCEDIMENTOS_URGENTES_PK 
      PRIMARY KEY (ID_PROCEDIMENTO_URGENTE);
-- fim bloco

ALTER TABLE PROCEDIMENTOS_URGENTES ADD CONSTRAINT PROCED_URGENTES_MNEMONICO_FK
    FOREIGN KEY (MNEMONICO) REFERENCES PROCEDIMENTOS(MNEMONICO);

-- fim bloco

ALTER TABLE PROCEDIMENTOS_URGENTES ADD CONSTRAINT PROCED_URGENTES_POSTO_REA_FK
    FOREIGN KEY (POSTO_REA) REFERENCES POSTOS(POSTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_PROCEDIMENTOS_URGENTES(
                                                V_ID_PROCEDIMENTO_URGENTE IN OUT PROCEDIMENTOS_URGENTES.ID_PROCEDIMENTO_URGENTE%TYPE,
                                                V_MNEMONICO               IN PROCEDIMENTOS_URGENTES.MNEMONICO%TYPE,
                                                V_POSTO_REA               IN PROCEDIMENTOS_URGENTES.POSTO_REA%TYPE,
                                                V_FUNCAO                  IN NUMBER
                                            ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_PROCEDIMENTOS_URGENTES.NEXTVAL INTO V_ID_PROCEDIMENTO_URGENTE FROM DUAL;
        INSERT INTO PROCEDIMENTOS_URGENTES VALUES(V_ID_PROCEDIMENTO_URGENTE, V_MNEMONICO, V_POSTO_REA);

    ELSIF V_FUNCAO = 2 THEN
        UPDATE PROCEDIMENTOS_URGENTES SET MNEMONICO = V_MNEMONICO, POSTO_REA = V_POSTO_REA
                                    WHERE ID_PROCEDIMENTO_URGENTE = V_ID_PROCEDIMENTO_URGENTE;

    ELSIF V_FUNCAO = 3 THEN
        DELETE PROCEDIMENTOS_URGENTES WHERE ID_PROCEDIMENTO_URGENTE = V_ID_PROCEDIMENTO_URGENTE;

    END IF;

    COMMIT WORK;
END PRC_PROCEDIMENTOS_URGENTES;

INCREMENT BY 1
START WITH 1

-- fim bloco

CREATE TABLE SEQUENCIAS_SETORES(
    ID_SEQUENCIA_SETOR NUMBER(4) NOT NULL,
    SETOR NUMBER(2) NOT NULL,
    POSTO_REA NUMBER(3) NOT NULL,
    DATA_INICIO_SEQUENCIA DATE NOT NULL,
    INICIO_SEQUENCIA NUMBER(7) DEFAULT 1 NOT NULL,
    QTD_DIAS_VALIDADE NUMBER(3) NOT NULL,
	SEQUENCIA_SETOR NUMBER(7) NOT NULL,
    LAYOUT_ETIQUETA VARCHAR2(250)
)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_PK 
      PRIMARY KEY (ID_SEQUENCIA_SETOR);
-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_SETOR_FK
    FOREIGN KEY (SETOR) REFERENCES SETORES(SETOR);
-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_POSTO_FK
    FOREIGN KEY (POSTO_REA) REFERENCES POSTOS(POSTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_SEQUENCIAS_SETORES(
                                                V_ID_SEQUENCIA_SETOR    IN OUT SEQUENCIAS_SETORES.ID_SEQUENCIA_SETOR%TYPE,
                                                V_SETOR                 IN SEQUENCIAS_SETORES.SETOR%TYPE,
                                                V_POSTO_REA             IN SEQUENCIAS_SETORES.POSTO_REA%TYPE,
                                                V_DATA_INICIO_SEQUENCIA IN SEQUENCIAS_SETORES.DATA_INICIO_SEQUENCIA%TYPE,
                                                V_INICIO_SEQUENCIA      IN SEQUENCIAS_SETORES.INICIO_SEQUENCIA%TYPE,
                                                V_QTD_DIAS_VALIDADE     IN SEQUENCIAS_SETORES.QTD_DIAS_VALIDADE%TYPE,
                                                V_SEQUENCIA_SETOR       IN SEQUENCIAS_SETORES.SEQUENCIA_SETOR%TYPE,
                                                V_LAYOUT_ETIQUETA       IN SEQUENCIAS_SETORES.LAYOUT_ETIQUETA%TYPE,
                                                V_FUNCAO                IN NUMBER
                                            ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_SEQUENCIAS_SETORES.NEXTVAL INTO V_ID_SEQUENCIA_SETOR FROM DUAL;
        INSERT INTO SEQUENCIAS_SETORES VALUES(
                                            V_ID_SEQUENCIA_SETOR, V_SETOR, V_POSTO_REA, V_DATA_INICIO_SEQUENCIA, V_INICIO_SEQUENCIA,
                                            V_QTD_DIAS_VALIDADE, V_INICIO_SEQUENCIA, V_LAYOUT_ETIQUETA);

    ELSIF V_FUNCAO = 2 THEN
        UPDATE SEQUENCIAS_SETORES SET
                                    SETOR = V_SETOR, POSTO_REA = V_POSTO_REA, DATA_INICIO_SEQUENCIA = V_DATA_INICIO_SEQUENCIA, INICIO_SEQUENCIA = V_INICIO_SEQUENCIA,
                                    QTD_DIAS_VALIDADE = V_QTD_DIAS_VALIDADE, SEQUENCIA_SETOR = V_SEQUENCIA_SETOR, LAYOUT_ETIQUETA = V_LAYOUT_ETIQUETA
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;

    ELSIF V_FUNCAO = 3 THEN
        DELETE SEQUENCIAS_SETORES WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;

    ELSIF V_FUNCAO = 4 THEN
        UPDATE SEQUENCIAS_SETORES SET 
                                     SEQUENCIA_SETOR = SEQUENCIA_SETOR + 1
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;
    ELSIF V_FUNCAO = 5 THEN
        UPDATE SEQUENCIAS_SETORES SET
                                    DATA_INICIO_SEQUENCIA = SYSDATE,
                                    SEQUENCIA_SETOR = INICIO_SEQUENCIA
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;
    END IF;

    COMMIT WORK;
END PRC_SEQUENCIAS_SETORES;

                                                V_USUARIO            IN EXAMES_SEQUENCIAS_SETORES.USUARIO%TYPE,

                                            ) IS

        SELECT SQ_EXAMES_SEQUENCIAS_SETORES.NEXTVAL INTO V_ID_EXAME_SEQUENCIA FROM DUAL;

                                            V_ID_EXAME_SEQUENCIA, V_POSTO, V_ATENDIMENTO, V_CORREL, V_AMOSTRA,
                                            V_SEQUENCIA_SETOR, V_DATA_CAD, V_USUARIO);

    ELSIF V_FUNCAO = 2 THEN

CREATE OR REPLACE PROCEDURE PRC_ATENDIMENTOS(V_POSTO                      IN     ATENDIMENTOS.POSTO%TYPE,
                                             V_REGISTRO                   IN     ATENDIMENTOS.REGISTRO%TYPE,
                                             V_ATENDIMENTO                IN OUT ATENDIMENTOS.ATENDIMENTO%TYPE,
                                             V_CONVENIO                   IN     ATENDIMENTOS.CONVENIO%TYPE,
                                             V_MATRICULA                  IN     ATENDIMENTOS.MATRICULA%TYPE,
                                             V_DATA_ATD                   IN     ATENDIMENTOS.DATA_ATD%TYPE,
                                             V_DATA_FAT                   IN     ATENDIMENTOS.DATA_FAT%TYPE,
                                             V_DATA_ENT                   IN     ATENDIMENTOS.DATA_ENT%TYPE,
                                             V_HORA_ENT                   IN     ATENDIMENTOS.HORA_ENT%TYPE,
                                             V_MED_SOL                    IN     ATENDIMENTOS.MED_SOL%TYPE,
                                             V_LIB_FAT                    IN     ATENDIMENTOS.LIB_FAT%TYPE,
                                             V_FATURAMENTO                IN     ATENDIMENTOS.FATURAMENTO%TYPE,
                                             V_USUARIO                    IN     ATENDIMENTOS.USUARIO%TYPE,
                                             V_DATA_CAD                   IN     ATENDIMENTOS.DATA_CAD%TYPE,
                                             V_POSTO_ENTREGA              IN     ATENDIMENTOS.POSTO_ENTREGA%TYPE,
                                             V_SITUACAO                   IN     ATENDIMENTOS.SITUACAO%TYPE,
                                             V_DATA_DESDE                 IN     ATENDIMENTOS.DATA_FAT%TYPE,
                                             V_DATA_ATE                   IN     ATENDIMENTOS.DATA_FAT%TYPE,
                                             V_CONDICAO_FATURAMENTO       IN     ATENDIMENTOS.FATURAMENTO%TYPE,
                                             V_CODIGO_PLANO               IN     ATENDIMENTOS.CODIGO_PLANO%TYPE,
                                             V_DATA_VALIDADE              IN     ATENDIMENTOS.DATA_VALIDADE%TYPE,
                                             V_ID_FATURA                  IN     ATENDIMENTOS.ID_FATURA%TYPE,
                                             V_GUIA_PRINCIPAL             IN     ATENDIMENTOS.GUIA_PRINCIPAL%TYPE,
                                             V_IDENTIFICACAO_BENEFICIARIO IN     ATENDIMENTOS.IDENTIFICACAO_BENEFICIARIO%TYPE,
                                             V_CARATER_ATENDIMENTO        IN     ATENDIMENTOS.CARATER_ATENDIMENTO%TYPE,
                                             V_FUNCAO                     IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 OR V_FUNCAO = 9 THEN
     IF V_FUNCAO = 1 THEN
       SELECT SQ_ATENDIMENTO.NEXTVAL INTO V_ATENDIMENTO FROM DUAL;
     END IF;

     INSERT INTO ATENDIMENTOS VALUES (V_POSTO, V_REGISTRO, V_ATENDIMENTO, V_CONVENIO,
        V_MATRICULA, V_DATA_ATD, V_DATA_FAT, V_DATA_ENT, V_HORA_ENT, V_MED_SOL, V_LIB_FAT,
        V_FATURAMENTO, V_USUARIO, V_DATA_CAD, V_POSTO_ENTREGA, V_SITUACAO, V_CODIGO_PLANO, V_DATA_VALIDADE, V_ID_FATURA, V_GUIA_PRINCIPAL, V_IDENTIFICACAO_BENEFICIARIO,
		V_CARATER_ATENDIMENTO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE ATENDIMENTOS SET CONVENIO     = V_CONVENIO,      MATRICULA = V_MATRICULA,
           DATA_FAT      = V_DATA_FAT,      DATA_ENT  = V_DATA_ENT,
           HORA_ENT = V_HORA_ENT, MED_SOL       = V_MED_SOL,       LIB_FAT   = V_LIB_FAT,
           USUARIO  = V_USUARIO,  FATURAMENTO   = V_FATURAMENTO,   DATA_CAD  = V_DATA_CAD,      
           POSTO    = V_POSTO,    POSTO_ENTREGA = V_POSTO_ENTREGA, SITUACAO  = V_SITUACAO,
           CODIGO_PLANO = V_CODIGO_PLANO, DATA_VALIDADE = V_DATA_VALIDADE, ID_FATURA = V_ID_FATURA, 
		   GUIA_PRINCIPAL = V_GUIA_PRINCIPAL, IDENTIFICACAO_BENEFICIARIO = V_IDENTIFICACAO_BENEFICIARIO,
		   CARATER_ATENDIMENTO = V_CARATER_ATENDIMENTO
        WHERE REGISTRO = V_REGISTRO AND  ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE ATENDIMENTOS WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE ATENDIMENTOS SET DATA_ENT  = V_DATA_ENT,
                                   HORA_ENT = V_HORA_ENT,
                                   POSTO_ENTREGA = V_POSTO_ENTREGA,
                                   DATA_CAD = V_DATA_CAD
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE ATENDIMENTOS SET LIB_FAT = V_LIB_FAT
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 6 THEN
     IF V_CONDICAO_FATURAMENTO = '' THEN
        UPDATE ATENDIMENTOS SET FATURAMENTO = V_FATURAMENTO, ID_FATURA = V_ID_FATURA
               WHERE CONVENIO = V_CONVENIO 
                 AND (V_CODIGO_PLANO IS NULL OR CODIGO_PLANO = V_CODIGO_PLANO)
                 AND DATA_FAT >= V_DATA_DESDE AND DATA_FAT <= V_DATA_ATE 
                 AND LIB_FAT = 'S';
     ELSE 
        UPDATE ATENDIMENTOS SET FATURAMENTO = V_FATURAMENTO, ID_FATURA = V_ID_FATURA
               WHERE CONVENIO = V_CONVENIO 
                 AND (V_CODIGO_PLANO IS NULL OR CODIGO_PLANO = V_CODIGO_PLANO)
                 AND DATA_FAT >= V_DATA_DESDE AND DATA_FAT <= V_DATA_ATE 
                 AND LIB_FAT = 'S' AND FATURAMENTO = V_CONDICAO_FATURAMENTO;

     END IF;

  ELSIF V_FUNCAO = 7 THEN
     UPDATE ATENDIMENTOS SET LIB_FAT = V_LIB_FAT, DATA_FAT = V_DATA_FAT
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 8 THEN
     UPDATE ATENDIMENTOS SET FATURAMENTO = V_FATURAMENTO
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 10 THEN
     UPDATE ATENDIMENTOS SET DATA_ENT  = V_DATA_ENT,
                             HORA_ENT = V_HORA_ENT
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 11 THEN
     UPDATE ATENDIMENTOS SET MED_SOL  = V_MED_SOL
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 12 THEN
     UPDATE ATENDIMENTOS SET CONVENIO            = V_CONVENIO,
                             DATA_FAT            = V_DATA_FAT,
                             LIB_FAT             = V_LIB_FAT,
                             MED_SOL             = V_MED_SOL,
                             MATRICULA           = V_MATRICULA,
							 GUIA_PRINCIPAL      = V_GUIA_PRINCIPAL,
							 CODIGO_PLANO        = V_CODIGO_PLANO,
							 DATA_VALIDADE       = V_DATA_VALIDADE,
							 CARATER_ATENDIMENTO = V_CARATER_ATENDIMENTO
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 13 THEN
     UPDATE ATENDIMENTOS SET MATRICULA = V_MATRICULA, 
	                         GUIA_PRINCIPAL = V_GUIA_PRINCIPAL, 
							 IDENTIFICACAO_BENEFICIARIO = V_IDENTIFICACAO_BENEFICIARIO
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 14 THEN
     UPDATE ATENDIMENTOS SET SITUACAO = V_SITUACAO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 15 THEN
        UPDATE ATENDIMENTOS SET ID_FATURA = NULL
               WHERE CONVENIO = V_CONVENIO 
                 AND (V_CODIGO_PLANO IS NULL OR CODIGO_PLANO = V_CODIGO_PLANO)
                 AND DATA_FAT >= V_DATA_DESDE AND DATA_FAT <= V_DATA_ATE;

  ELSIF V_FUNCAO = 16 THEN
     UPDATE ATENDIMENTOS SET DATA_ATD = V_DATA_ATD
               WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  END IF;

  COMMIT WORK;
END PRC_ATENDIMENTOS;
