      PRIMARY KEY (CRM);
	  

CREATE OR REPLACE PROCEDURE PRC_MEDICOS(V_CRM             IN OUT MEDICOS.CRM%TYPE,
                                        V_NOME            IN     MEDICOS.NOME%TYPE,
                                        V_CPF             IN     MEDICOS.CPF%TYPE,
                                        V_TIPO            IN     MEDICOS.TIPO%TYPE,
                                        V_STATUS          IN     MEDICOS.STATUS%TYPE,
                                        V_UF_CONSELHO     IN     MEDICOS.UF_CONSELHO%TYPE,
                                        V_TIPO_CR         IN     MEDICOS.TIPO_CR%TYPE,
                                        V_TELEFONE        IN     MEDICOS.TELEFONE%TYPE,
                                        V_TELEFONE2       IN     MEDICOS.TELEFONE2%TYPE,
                                        V_EMAIL           IN     MEDICOS.EMAIL%TYPE,
                                        V_PASS            IN     MEDICOS.PASS%TYPE,
                                        V_ID_MEDICO       IN OUT MEDICOS.ID_MEDICO%TYPE,
                                        V_DATA_NAS        IN     MEDICOS.DATA_NAS%TYPE,
                                        V_PREFIXO_NOME    IN     MEDICOS.PREFIXO_NOME%TYPE,
                                        V_CODIGO_UNIMED   IN     MEDICOS.CODIGO_UNIMED%TYPE,
                                        V_ID_CBO          IN     MEDICOS.ID_CBO%TYPE,
                                        V_ESPECIALIZACAO  IN     MEDICOS.ESPECIALIZACAO%TYPE,
                                        V_CNS             IN     MEDICOS.CNS%TYPE,
                                        V_FUNCAO          IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_MEDICOS.NEXTVAL INTO V_ID_MEDICO FROM DUAL;

     INSERT INTO MEDICOS VALUES (V_CRM, V_NOME, V_CPF, V_TIPO, V_STATUS, V_UF_CONSELHO, V_TIPO_CR, V_TELEFONE, V_TELEFONE2, V_EMAIL, V_PASS, V_ID_MEDICO, V_DATA_NAS, V_PREFIXO_NOME, V_CODIGO_UNIMED, V_ID_CBO, V_ESPECIALIZACAO, V_CNS);
  
  ELSIF V_FUNCAO = 2 THEN
     UPDATE MEDICOS SET NOME = V_NOME, CPF = V_CPF, TIPO = V_TIPO, STATUS = V_STATUS,
                              UF_CONSELHO = V_UF_CONSELHO, TIPO_CR = V_TIPO_CR,
                              TELEFONE = V_TELEFONE, TELEFONE2 = V_TELEFONE2, EMAIL = V_EMAIL, 
                              PASS = V_PASS, DATA_NAS = V_DATA_NAS, PREFIXO_NOME = V_PREFIXO_NOME, 
                              CODIGO_UNIMED = V_CODIGO_UNIMED, ID_CBO = V_ID_CBO, ESPECIALIZACAO = V_ESPECIALIZACAO,
                              CNS= V_CNS 
      WHERE CRM = V_CRM;

  ELSIF V_FUNCAO = 3 THEN
     DELETE MEDICOS WHERE CRM = V_CRM;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE MEDICOS SET NOME = V_NOME, CPF = V_CPF 
      WHERE ID_MEDICO = V_ID_MEDICO;

  END IF;

  COMMIT WORK;

END PRC_MEDICOS;

CREATE TABLE MOTIVOS
(ID_MOTIVO        NUMBER(6) NOT NULL,
 MOTIVO           VARCHAR2(500) NOT NULL,
 TIPO_MOTIVO      VARCHAR2(2) NOT NULL,
 MNEMONICO_MOTIVO VARCHAR2(6),
 MOTIVO_FORMAT    LONG,
 STATUS           VARCHAR2(1) NOT NULL,
 PRIMEIRA_AMOSTRA VARCHAR2(1) NOT NULL)
TABLESPACE LIS_DATA;

ALTER TABLE MOTIVOS ADD CONSTRAINT MOTIVOS_PK 
      PRIMARY KEY (ID_MOTIVO);

CREATE SEQUENCE SQ_MOTIVO
MINVALUE 0
MAXVALUE 999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;

CREATE OR REPLACE PROCEDURE PRC_MOTIVOS(V_ID_MOTIVO        IN OUT MOTIVOS.ID_MOTIVO%TYPE,
                                        V_MOTIVO           IN     MOTIVOS.MOTIVO%TYPE,
                                        V_TIPO_MOTIVO      IN     MOTIVOS.TIPO_MOTIVO%TYPE,
                                        V_MNEMONICO_MOTIVO IN     MOTIVOS.MNEMONICO_MOTIVO%TYPE,
                                        V_MOTIVO_FORMAT    IN     MOTIVOS.MOTIVO_FORMAT%TYPE,
                                        V_STATUS           IN     MOTIVOS.STATUS%TYPE,
                                        V_PRIMEIRA_AMOSTRA IN     MOTIVOS.PRIMEIRA_AMOSTRA%TYPE,
                                        V_FUNCAO           IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT TO_CHAR(SQ_MOTIVO.NEXTVAL) INTO V_ID_MOTIVO FROM DUAL;

     INSERT INTO MOTIVOS VALUES (V_ID_MOTIVO, V_MOTIVO, V_TIPO_MOTIVO, V_MNEMONICO_MOTIVO, V_MOTIVO_FORMAT, V_STATUS, V_PRIMEIRA_AMOSTRA);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE MOTIVOS SET MOTIVO = V_MOTIVO, TIPO_MOTIVO = V_TIPO_MOTIVO, MNEMONICO_MOTIVO = V_MNEMONICO_MOTIVO, MOTIVO_FORMAT = V_MOTIVO_FORMAT,
	                    STATUS = V_STATUS, PRIMEIRA_AMOSTRA = V_PRIMEIRA_AMOSTRA
                       WHERE ID_MOTIVO = V_ID_MOTIVO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE MOTIVOS WHERE ID_MOTIVO = V_ID_MOTIVO;
  END IF;

  COMMIT WORK;

END PRC_MOTIVOS;

CREATE OR REPLACE PROCEDURE PRC_AMOSTRAS(V_POSTO        IN AMOSTRAS.POSTO%TYPE,
                                         V_ATENDIMENTO  IN AMOSTRAS.ATENDIMENTO%TYPE,
                                         V_CORREL       IN AMOSTRAS.CORREL%TYPE,
                                         V_MNEMONICO    IN AMOSTRAS.MNEMONICO%TYPE,
                                         V_AMOSTRA      IN OUT AMOSTRAS.AMOSTRA%TYPE,
                                         V_USUARIO      IN AMOSTRAS.USUARIO%TYPE,
                                         V_USUARIO_COLE IN AMOSTRAS.USUARIO_COLE%TYPE,
                                         V_DATA_CAD     IN AMOSTRAS.DATA_CAD%TYPE,
                                         V_DATA_COLE    IN AMOSTRAS.DATA_COLE%TYPE,
                                         V_OBSERVACOES  IN AMOSTRAS.OBSERVACOES%TYPE,
                                         V_IMPRIME_OBS  IN AMOSTRAS.IMPRIME_OBS%TYPE,
										 V_ID_MOTIVO    IN AMOSTRAS.ID_MOTIVO%TYPE,
                                         V_FUNCAO       IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
		IF V_AMOSTRA < 1 THEN
		     SELECT NVL(MAX(AMOSTRA)+1,1) INTO V_AMOSTRA FROM AMOSTRAS
			  WHERE POSTO = V_POSTO 
                AND ATENDIMENTO = V_ATENDIMENTO
				AND CORREL = V_CORREL;
		END IF;

		INSERT INTO AMOSTRAS VALUES (V_POSTO, V_ATENDIMENTO, V_CORREL, V_MNEMONICO,
                                        V_AMOSTRA, V_USUARIO, V_USUARIO_COLE, V_DATA_CAD,
                                        V_DATA_COLE, V_OBSERVACOES, V_IMPRIME_OBS, V_ID_MOTIVO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE AMOSTRAS SET USUARIO_COLE = V_USUARIO_COLE, DATA_COLE = V_DATA_COLE
            WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND AMOSTRA = V_AMOSTRA;

  ELSIF V_FUNCAO = 3 THEN
     DELETE AMOSTRAS WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO 
                             AND CORREL = V_CORREL AND AMOSTRA = V_AMOSTRA;
							 
  ELSIF V_FUNCAO = 4 THEN
     UPDATE AMOSTRAS SET OBSERVACOES = V_OBSERVACOES, IMPRIME_OBS = V_IMPRIME_OBS
            WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND AMOSTRA = V_AMOSTRA;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE AMOSTRAS SET ID_MOTIVO = V_ID_MOTIVO
            WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND AMOSTRA = V_AMOSTRA;
  END IF;

  COMMIT WORK;

END PRC_AMOSTRAS;
