                                              V_SEM_VALOR                   IN     PROCEDIMENTOS.SEM_VALOR%TYPE,
                                              V_TEM_EVOLUCAO                IN     PROCEDIMENTOS.TEM_EVOLUCAO%TYPE,
                                              V_QUEBRA_PAGINA               IN     PROCEDIMENTOS.QUEBRA_PAGINA%TYPE,
                                              V_AVISO_ETIQUETA              IN     PROCEDIMENTOS.AVISO_ETIQUETA%TYPE,

CREATE OR REPLACE FUNCTION GET_REGISTRO(V_POSTO IN NUMBER, V_ATENDIMENTO IN NUMBER) RETURN NUMBER IS RESULT NUMBER;
BEGIN
     SELECT NVL(REGISTRO,-1) INTO RESULT FROM ATENDIMENTOS
      WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

    RETURN(RESULT);
END GET_REGISTRO;
