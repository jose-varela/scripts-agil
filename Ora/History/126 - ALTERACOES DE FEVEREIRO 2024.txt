create or replace
FUNCTION GET_TEMPO_ENTREGA(V_MNEMONICO     IN VARCHAR2,
                                             V_ID_ACOMODACAO IN NUMBER) RETURN VARCHAR2 IS vTEMPO VARCHAR2(8);

vSQL           VARCHAR2(256);    --string que recebe o comando SQL dinâmico
vNOME_TABELA   VARCHAR2(30); --PLS_INTEGER;      --Nome tabela do Catalogo

BEGIN

     --PASSO 1: monta e executa SQL dinâmico para pegar o nro do catalogo de acordo com a acomodacao
     vSQL := 'SELECT  NVL(ATE.ID_TEMPO_ENTREGA,1) ' ||
             '  FROM ACOMODACOES AC, ACOMODACOES_TEMPOS_ENTREGA ATE ' ||
             ' WHERE AC.ID_ACOMODACAO = ' || V_ID_ACOMODACAO ||
             '   AND ATE.ID_ACOMODACAO(+) = AC.ID_ACOMODACAO';

     EXECUTE IMMEDIATE vSQL INTO vNOME_TABELA;

     --PASSO 2: monta o nome da tabela TEMPOS_ENTREGAx
     vNOME_TABELA := 'TEMPOS_ENTREGA' || vNOME_TABELA;


     --PASSO 3: monta e executa SQL dinâmico para buscar o TEMPO referente a tabela TEMPOS_ENTREGAx
     vSQL := 'SELECT TEMPO FROM ' || vNOME_TABELA ||
             ' WHERE MNEMONICO = ''' || V_MNEMONICO || '''';

     EXECUTE IMMEDIATE vSQL INTO vTEMPO;


    RETURN(vTEMPO);

END GET_TEMPO_ENTREGA;

create or replace FUNCTION GET_COUNT_EXAMES_SITUACAO(V_POSTO       IN NUMBER,
                                                     V_ATENDIMENTO IN NUMBER,
                                                     V_SITUACAO    IN VARCHAR2) RETURN NUMBER IS RESULT NUMBER;
BEGIN
  IF V_SITUACAO = 'FINALIZADO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.PENDENCIA <> 'S'
        AND E.SITUACAO IN ('R','I','L')
        AND E.SITUACAO_INTERFACE IN ('N','L','U','D','T','G');

  ELSIF V_SITUACAO = 'ANDAMENTO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.PENDENCIA <> 'S'
        AND E.SITUACAO NOT IN ('A','C','F')
        AND E.SITUACAO_INTERFACE IN ('S','E','I','P');

  ELSIF V_SITUACAO = 'AGUARDANDO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.PENDENCIA <> 'S'
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO IN ('R','N','U')
        AND E.SITUACAO_INTERFACE = 'R';

  ELSIF V_SITUACAO = 'PENDENCIA' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO NOT IN ('A','C','F')
        AND E.PENDENCIA = 'S';

  ELSIF V_SITUACAO = 'NAO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.PENDENCIA <> 'S'
        AND E.SITUACAO IN ('N','U')
        AND E.SITUACAO_INTERFACE = 'N';

  ELSIF V_SITUACAO = 'TODOS' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO NOT IN ('A','C','F');

   END IF;
   RETURN(RESULT);
END GET_COUNT_EXAMES_SITUACAO;
-- fim bloco

create or replace FUNCTION GET_SITUACAO_EXAME(V_PENDENCIA           IN VARCHAR2,
                                              V_SITUACAO            IN VARCHAR2,
                                              V_SITUACAO_INTERFACE  IN VARCHAR2) RETURN VARCHAR2 IS RESULT VARCHAR2(15);
BEGIN
   IF V_PENDENCIA = 'S' THEN
 	  RESULT := 'PENDENCIA';
   ELSIF V_SITUACAO IN ('I','L') THEN
 	  RESULT := 'FINALIZADO';
   ELSIF V_SITUACAO = 'R' AND V_SITUACAO_INTERFACE IN ('L','U','D','T','G') THEN 
 	  RESULT := 'FINALIZADO';
   ELSIF V_SITUACAO = 'R' AND V_SITUACAO_INTERFACE = 'N' THEN 
 	  RESULT := 'AGUARDANDO';
   ELSIF V_SITUACAO_INTERFACE = 'R' AND V_SITUACAO IN ('R','N','U') THEN 
 	  RESULT := 'AGUARDANDO';
   ELSIF V_SITUACAO_INTERFACE IN ('S','E','I','P') THEN 
 	  RESULT := 'ANDAMENTO';
   ELSE 
	  RESULT := 'NAO';
   END IF;
   
   RETURN(RESULT);
END GET_SITUACAO_EXAME;
