(POSTO_REALIZANTE     NUMBER(3)    NOT NULL,
 MNEMONICO            VARCHAR2(4)  NOT NULL,
 GRUPO_ETIQUETA       NUMBER(6)    NOT NULL)
TABLESPACE LIS_DATA;

ALTER TABLE EXCECOES_PADRAO_ETIQUETAS ADD CONSTRAINT EXCECOES_PADRAO_ETIQUETAS_PK 
      PRIMARY KEY (POSTO_REALIZANTE, MNEMONICO)
	  USING INDEX
	  TABLESPACE LIS_INDX;

ALTER TABLE EXCECOES_PADRAO_ETIQUETAS ADD CONSTRAINT EXCECOES_PADRAO_ETIQ_PROCED_FK 
      FOREIGN KEY (MNEMONICO) REFERENCES PROCEDIMENTOS(MNEMONICO);

ALTER TABLE EXCECOES_PADRAO_ETIQUETAS ADD CONSTRAINT EXCECOES_PADRAO_ETIQ_POSTOS_FK 
      FOREIGN KEY (POSTO_REALIZANTE) REFERENCES POSTOS(POSTO);

ALTER TABLE EXCECOES_PADRAO_ETIQUETAS ADD CONSTRAINT EXCECOES_PADRAO_ETIQUETAS_FK 
      FOREIGN KEY (GRUPO_ETIQUETA) REFERENCES GRUPOS_ETIQUETAS(GRUPO_ETIQUETA);


CREATE OR REPLACE PROCEDURE PRC_EXCECOES_PADRAO_ETIQUETAS(V_POSTO_REALIZANTE IN  EXCECOES_PADRAO_ETIQUETAS.POSTO_REALIZANTE%TYPE,
                                                          V_MNEMONICO        IN  EXCECOES_PADRAO_ETIQUETAS.MNEMONICO%TYPE,                                                     
                                                          V_GRUPO_ETIQUETA   IN  EXCECOES_PADRAO_ETIQUETAS.GRUPO_ETIQUETA%TYPE,
                                                          V_FUNCAO           IN  NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO EXCECOES_PADRAO_ETIQUETAS VALUES (V_POSTO_REALIZANTE, V_MNEMONICO, V_GRUPO_ETIQUETA);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE EXCECOES_PADRAO_ETIQUETAS SET GRUPO_ETIQUETA = V_GRUPO_ETIQUETA
                                       WHERE POSTO_REALIZANTE = V_POSTO_REALIZANTE AND MNEMONICO = V_MNEMONICO;
                    
  ELSIF V_FUNCAO = 3 THEN
     DELETE EXCECOES_PADRAO_ETIQUETAS WHERE POSTO_REALIZANTE = V_POSTO_REALIZANTE AND MNEMONICO = V_MNEMONICO;

  END IF;

  COMMIT WORK;

END PRC_EXCECOES_PADRAO_ETIQUETAS;
/
