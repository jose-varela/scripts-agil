CREATE SEQUENCE SQ_SEQUENCIAS_SETORES
MINVALUE 0 
MAXVALUE 9999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;

CREATE TABLE SEQUENCIAS_SETORES(
    ID_SEQUENCIA_SETOR NUMBER NOT NULL,
    SETOR NUMBER NOT NULL,
    POSTO_REA NUMBER NOT NULL,
    DATA_INICIO_SEQUENCIA DATE NOT NULL,
    QTD_DIAS_VALIDADE NUMBER NOT NULL,
    LAYOUT_ETIQUETA VARCHAR2(250)
)
TABLESPACE LIS_DATA;

-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_PK 
      PRIMARY KEY (ID_SEQUENCIA_SETOR);

-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_SETOR_FK
    FOREIGN KEY (SETOR) REFERENCES SETORES(SETOR);

-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_POSTO_FK
    FOREIGN KEY (POSTO_REA) REFERENCES POSTOS(POSTO);

-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_SEQUENCIAS_SETORES(
                                                V_ID_SEQUENCIA_SETOR    IN SEQUENCIAS_SETORES.ID_SEQUENCIA_SETOR%TYPE,
                                                V_SETOR                 IN SEQUENCIAS_SETORES.SETOR%TYPE,
                                                V_POSTO_REA             IN SEQUENCIAS_SETORES.POSTO_REA%TYPE,
                                                V_DATA_INICIO_SEQUENCIA IN SEQUENCIAS_SETORES.DATA_INICIO_SEQUENCIA%TYPE,
                                                V_QTD_DIAS_VALIDADE     IN SEQUENCIAS_SETORES.QTD_DIAS_VALIDADE%TYPE,
                                                V_LAYOUT_ETIQUETA       IN SEQUENCIAS_SETORES.LAYOUT_ETIQUETA%TYPE,
                                                V_FUNCAO                IN NUMBER
                                            ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_SEQUENCIAS_SETORES.NEXTVAL INTO V_ID_SEQUENCIA_SETOR FROM DUAL;
        INSERT INTO SEQUENCIAS_SETORES VALUES(
                                            V_SETOR, V_POSTO_REA, V_DATA_INICIO_SEQUENCIA,
                                            V_QTD_DIAS_VALIDADE, V_LAYOUT_ETIQUETA);

    ELSIF V_FUNCAO = 2 THEN
        UPDATE SEQUENCIAS_SETORES SET (
                                    SETOR = V_SETOR, POSTO_REA = V_POSTO_REA, DATA_INICIO_SEQUENCIA = V_DATA_INICIO_SEQUENCIA,
                                    QTD_DIAS_VALIDADE = V_QTD_DIAS_VALIDADE, LAYOUT_ETIQUETA = V_LAYOUT_ETIQUETA)
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;

    ELSIF V_FUNCAO = 3 THEN
        DELETE SEQUENCIAS_SETORES WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;

    END IF;

    COMMIT WORK;
END PRC_SEQUENCIAS_SETORES;
