CREATE TABLE GRUPOS_POSTOS
(GRUPO_POSTO        NUMBER(2) NOT NULL,
 NOME_GRUPO         VARCHAR2(50) NOT NULL,
 POSTO              NUMBER(3) NOT NULL)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE GRUPOS_POSTOS ADD CONSTRAINT GRUPOS_POSTOS_PK 
      PRIMARY KEY (GRUPO_POSTO);
-- fim bloco

CREATE SEQUENCE SQ_GRUPO_POSTO MINVALUE 0 
MAXVALUE 999999
INCREMENT BY 1
START WITH 0
NOCACHE
NOORDER;
-- fim bloco

ALTER TABLE GRUPOS_POSTOS ADD CONSTRAINT GRUPOS_POSTOS_POSTOS_FK 
      FOREIGN KEY (POSTO) REFERENCES POSTOS(POSTO);
-- fim bloco


CREATE OR REPLACE PROCEDURE PRC_GRUPOS_POSTOS(V_GRUPO_POSTO         IN OUT GRUPOS_POSTOS.GRUPO_POSTO%TYPE,
                                              V_NOME_GRUPO          IN     GRUPOS_POSTOS.NOME_GRUPO%TYPE,
                                              V_POSTO               IN     GRUPOS_POSTOS.POSTO%TYPE,
                                              V_FUNCAO              IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_GRUPO_POSTO.NEXTVAL INTO V_GRUPO_POSTO FROM DUAL;
     INSERT INTO GRUPOS_POSTOS VALUES (V_GRUPO_POSTO, V_NOME_GRUPO, V_POSTO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE GRUPOS_POSTOS SET NOME_GRUPO = V_NOME_GRUPO, POSTO = V_POSTO
      WHERE GRUPO_POSTO = V_GRUPO_POSTO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE GRUPOS_POSTOS WHERE GRUPO_POSTO = V_GRUPO_POSTO;
  END IF;

  COMMIT WORK;

END PRC_GRUPOS_POSTOS;
