--
-- SEQ_ORCAMENTO
-- Contexto: integracao AGIL -> Marduk-CRM (docs/api-contrato-agil.md).
-- Gera o codigo curto do orcamento (campo codigoAgil, mandado no POST
-- /orcamentos e impresso no recibo da Consulta de Precos) -- o AGIL gera
-- esse codigo, nao o Marduk-CRM, porque o envio e fire-and-forget e nao
-- pode esperar resposta de rede antes de imprimir. START WITH 100000 pra
-- comecar com 6 digitos (mais facil de ler/digitar de volta na recuperacao
-- do orcamento); MAXVALUE alto o bastante pra nunca precisar de CYCLE.
--
CREATE SEQUENCE SEQ_ORCAMENTO
    START WITH 100000
    INCREMENT BY 1
    MAXVALUE 999999999
    NOCACHE
    NOCYCLE;
-- fim bloco

--
-- ATENDIMENTOS - solicitante avulso
-- Contexto: cliente de rede publica precisa cadastrar o medico solicitante
-- so pelo nome, sem vinculo com o cadastro de MEDICOS (sem CRM formal, nao
-- passa por TISS). Em vez de afrouxar ATENDIMENTOS_MEDICOS_FK ou criar um
-- registro coringa dentro de MEDICOS, ID_MEDICO_SOLICITANTE passa a aceitar
-- NULL -- a FK continua ativa e obrigatoria sempre que houver valor, so
-- passa a permitir a ausencia dele quando o atendimento usa solicitante
-- avulso. O nome avulso fica numa tabela satelite separada (bloco seguinte),
-- nao dentro de ATENDIMENTOS, pra nao alterar a tabela mais central do
-- sistema pra um dado que a maioria dos clientes nunca vai usar.
--
ALTER TABLE ATENDIMENTOS MODIFY ("ID_MEDICO_SOLICITANTE" NULL);
-- fim bloco

--
-- ATEND_SOLICITANTE_AVULSO
-- Tabela satelite 1:0..1 de ATENDIMENTOS. So existe linha aqui quando
-- ID_MEDICO_SOLICITANTE (bloco acima) esta NULL. FK com ON DELETE CASCADE
-- pra acompanhar o ciclo de vida do atendimento sem precisar de passo extra
-- na exclusao.
--
CREATE TABLE ATEND_SOLICITANTE_AVULSO (
  POSTO            NUMBER        NOT NULL,
  ATENDIMENTO      NUMBER        NOT NULL,
  NOME_SOLICITANTE VARCHAR2(120) NOT NULL
);
-- fim bloco

ALTER TABLE ATEND_SOLICITANTE_AVULSO ADD CONSTRAINT ATEND_SOLICITANTE_AVULSO_PK
  PRIMARY KEY (POSTO, ATENDIMENTO) ENABLE;
-- fim bloco

ALTER TABLE ATEND_SOLICITANTE_AVULSO ADD CONSTRAINT ATEND_SOLICITANTE_AVULSO_FK
  FOREIGN KEY (POSTO, ATENDIMENTO) REFERENCES ATENDIMENTOS (POSTO, ATENDIMENTO) ON DELETE CASCADE ENABLE;
-- fim bloco

create or replace
PROCEDURE PRC_ATEND_SOLICITANTE_AVULSO(V_POSTO            IN ATEND_SOLICITANTE_AVULSO.POSTO%TYPE,
                                        V_ATENDIMENTO      IN ATEND_SOLICITANTE_AVULSO.ATENDIMENTO%TYPE,
                                        V_NOME_SOLICITANTE IN ATEND_SOLICITANTE_AVULSO.NOME_SOLICITANTE%TYPE,
                                        V_FUNCAO           IN NUMBER) IS
  V_ACHEI NUMBER(1);
  CURSOR CUR_PROCURA IS
    SELECT 1 FROM ATEND_SOLICITANTE_AVULSO WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;
BEGIN
  IF V_FUNCAO = 1 THEN
     OPEN CUR_PROCURA;
     FETCH CUR_PROCURA INTO V_ACHEI;
     IF CUR_PROCURA%NOTFOUND THEN
        INSERT INTO ATEND_SOLICITANTE_AVULSO VALUES (V_POSTO, V_ATENDIMENTO, V_NOME_SOLICITANTE);
     ELSE
        UPDATE ATEND_SOLICITANTE_AVULSO SET NOME_SOLICITANTE = V_NOME_SOLICITANTE
         WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;
     END IF;
     CLOSE CUR_PROCURA;
  ELSIF V_FUNCAO = 3 THEN
     DELETE ATEND_SOLICITANTE_AVULSO WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;
  END IF;
  COMMIT WORK;
END PRC_ATEND_SOLICITANTE_AVULSO;
-- fim bloco

--
-- VIEWS - solicitante avulso
-- VEX_ATENDIMENTOS e VW_ATENDIMENTOS faziam M.ID_MEDICO = A.ID_MEDICO_SOLICITANTE
-- sem outer join -- com a coluna agora aceitando NULL, isso faria essas views
-- sumirem com a linha inteira do atendimento avulso (equijoin nao casa NULL).
-- RESULTADOS_WEB/VW_RESULTADOS ja usavam M.ID_MEDICO(+), entao a linha nao
-- sumia, mas NOME_MED voltava NULL. Fix comum aos quatro: LEFT JOIN em MEDICOS
-- e em ATEND_SOLICITANTE_AVULSO, com COALESCE(nome do medico, nome avulso).
--
CREATE OR REPLACE
FORCE VIEW "VEX_ATENDIMENTOS" ("POSTO",
                               "ATENDIMENTO",
                               "REGISTRO",
                               "DATA_ATD",
                               "SOLICITANTE",
                               "DATA_ENTREGA",
                               "CONVENIO",
                               "ID_SOLICITANTE",
                               "NOME_SOLICITANTE",
                               "NOME_CONVENIO",
                               "NOME_POSTO",
                               "ACOMODACAO",
                               "LEITO",
                               "SALDO_DEVEDOR",
                               "SITUACAO_EXAMES_EXPERIENCE") AS
SELECT A.POSTO,
       A.ATENDIMENTO,
       A.REGISTRO,
       A.DATA_ATD,
       A.ID_MEDICO_SOLICITANTE,
       A.DATA_ENT,
       A.CONVENIO,
       M.ID_MEDICO,
       COALESCE(M.NOME, SA.NOME_SOLICITANTE) NOME_SOLICITANTE,
       C.NOME NOME_CONVENIO,
       P.NOME NOME_POSTO,
       AC.NOME ACOMODACAO,
       CA.LEITO,
  (SELECT SUM(D.VALOR)
   FROM DEVEDORES D,
        CONTAS_ATENDIMENTOS CA
   WHERE CA.POSTO = A.POSTO
     AND CA.ATENDIMENTO = A.ATENDIMENTO
     AND D.CONTA_HOSP = CA.CONTA_HOSP), GET_SITUACAO_EXAMES(A.POSTO, A.ATENDIMENTO)
FROM ATENDIMENTOS A,
     MEDICOS M,
     ATEND_SOLICITANTE_AVULSO SA,
     CONVENIOS C,
     POSTOS P,
     ACOMODACOES AC,
     COMPLEMENTO_ATENDIMENTOS CA
WHERE A.SITUACAO = 'A' -- cedro   AND TO_CHAR(A.DATA_ATD,'YYYY') >= '2010'

  AND M.ID_MEDICO(+) = A.ID_MEDICO_SOLICITANTE
  AND SA.POSTO(+) = A.POSTO
  AND SA.ATENDIMENTO(+) = A.ATENDIMENTO
  AND C.CONVENIO = A.CONVENIO
  AND P.POSTO = A.POSTO
  AND CA.POSTO(+) = A.POSTO
  AND CA.ATENDIMENTO(+) = A.ATENDIMENTO
  AND AC.ID_ACOMODACAO(+) = CA.ID_ACOMODACAO
ORDER BY POSTO,
         ATENDIMENTO;
--fim bloco

                              "SOLICITANTE",

                              "SITUACAO_EXAMES_EXPERIENCE") AS

       A.ATENDIMENTO,
       A.REGISTRO,
       A.DATA_ATD,
       A.ID_MEDICO_SOLICITANTE,
       A.DATA_ENT,
       A.CONVENIO,
       M.ID_MEDICO,
       COALESCE(M.NOME, SA.NOME_SOLICITANTE) NOME_SOLICITANTE,
       C.NOME NOME_CONVENIO,
       P.NOME NOME_POSTO,
       AC.NOME ACOMODACAO,
       CA.LEITO,
  (SELECT SUM(D.VALOR)
   FROM DEVEDORES D,
        CONTAS_ATENDIMENTOS CA
   WHERE CA.POSTO = A.POSTO
     AND CA.ATENDIMENTO = A.ATENDIMENTO
     AND D.CONTA_HOSP = CA.CONTA_HOSP), GET_SITUACAO_EXAMES(A.POSTO, A.ATENDIMENTO)

     MEDICOS M,
     ATEND_SOLICITANTE_AVULSO SA,

     POSTOS P,
     ACOMODACOES AC,
     COMPLEMENTO_ATENDIMENTOS CA

  AND CA.ATENDIMENTO(+) = A.ATENDIMENTO

ORDER BY POSTO,
         ATENDIMENTO;

create or replace
PROCEDURE PRC_PERFIS_EXAMES(V_ID_PERFIL   IN OUT PERFIS_EXAMES.ID_PERFIL%TYPE,
                             V_NOME            IN PERFIS_EXAMES.NOME%TYPE,
                             V_DESCRICAO       IN PERFIS_EXAMES.DESCRICAO%TYPE,
                             V_STATUS          IN PERFIS_EXAMES.STATUS%TYPE,
                             V_USUARIO_CAD     IN PERFIS_EXAMES.USUARIO_CAD%TYPE,
                             V_FUNCAO          IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SEQ_PERFIL_EXAME.NEXTVAL INTO V_ID_PERFIL FROM DUAL;

     INSERT INTO PERFIS_EXAMES (ID_PERFIL, NOME, DESCRICAO, STATUS, DATA_CAD, USUARIO_CAD)
       VALUES (V_ID_PERFIL, V_NOME, V_DESCRICAO, V_STATUS, SYSDATE, V_USUARIO_CAD);
  ELSIF V_FUNCAO = 2 THEN
     UPDATE PERFIS_EXAMES
        SET NOME = V_NOME, DESCRICAO = V_DESCRICAO, STATUS = V_STATUS
      WHERE ID_PERFIL = V_ID_PERFIL;
  ELSIF V_FUNCAO = 3 THEN
     DELETE PERFIS_EXAMES WHERE ID_PERFIL = V_ID_PERFIL;
  END IF;
  COMMIT WORK;
END PRC_PERFIS_EXAMES;
-- fim bloco

create or replace
PROCEDURE PRC_PERFIS_EXAMES_ITENS(V_ID_PERFIL  IN PERFIS_EXAMES_ITENS.ID_PERFIL%TYPE,
                                   V_MNEMONICO  IN PERFIS_EXAMES_ITENS.MNEMONICO%TYPE,
                                   V_ORDEM      IN PERFIS_EXAMES_ITENS.ORDEM%TYPE,
                                   V_FUNCAO     IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO PERFIS_EXAMES_ITENS (ID_PERFIL, MNEMONICO, ORDEM)
       VALUES (V_ID_PERFIL, V_MNEMONICO, V_ORDEM);
  ELSIF V_FUNCAO = 2 THEN
     UPDATE PERFIS_EXAMES_ITENS SET ORDEM = V_ORDEM
      WHERE ID_PERFIL = V_ID_PERFIL AND MNEMONICO = V_MNEMONICO;
  ELSIF V_FUNCAO = 3 THEN
     DELETE PERFIS_EXAMES_ITENS WHERE ID_PERFIL = V_ID_PERFIL AND MNEMONICO = V_MNEMONICO;
  END IF;
  COMMIT WORK;
END PRC_PERFIS_EXAMES_ITENS;

--                    controle do reenvio (espera crescente ate 1 h, para de

--
-- PRC_USUARIOS: a FUNCAO 4 (troca de senha no login) passa a gravar tambem o novo
-- vencimento. Assinatura inalterada (V_DATA_VALIDADE ja existia). NVL mantem o
-- vencimento atual quando um executavel antigo chama a FUNCAO 4 sem a data.
-- Base: texto vivo da procedure (user_source) em 24/09/2026.
--
CREATE OR REPLACE PROCEDURE PRC_USUARIOS(V_USER_CODIGO    IN OUT USUARIOS.USER_CODIGO%TYPE,
                                         V_USER_TIPO      IN     USUARIOS.USER_TIPO%TYPE,
                                         V_USER_NOME      IN     USUARIOS.USER_NOME%TYPE,
                                         V_USER_ID        IN     USUARIOS.USER_ID%TYPE,
                                         V_SENHA          IN     USUARIOS.SENHA%TYPE,
                                         V_GRUPO          IN     USUARIOS.GRUPO%TYPE,
                                         V_DATA_CAD       IN     USUARIOS.DATA_CAD%TYPE,
                                         V_DATA_VALIDADE  IN     USUARIOS.DATA_VALIDADE%TYPE,
                                         V_VALIDADE       IN     USUARIOS.VALIDADE%TYPE,
                                         V_ATIVO          IN     USUARIOS.ATIVO%TYPE,
                                         V_FECHA_APP      IN     USUARIOS.FECHA_APP%TYPE,
                                         V_ID_MEDICO      IN     USUARIOS.ID_MEDICO%TYPE,
										 V_ALIAS          IN     USUARIOS.ALIAS%TYPE,
                                         V_FUNCAO         IN     NUMBER) IS

BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_USUARIOS.NEXTVAL INTO V_USER_CODIGO FROM DUAL;
     INSERT INTO USUARIOS VALUES (V_USER_CODIGO, V_USER_TIPO, V_USER_NOME, V_USER_ID,
                                        V_SENHA,       V_GRUPO,     V_DATA_CAD,  V_DATA_VALIDADE,
                                        V_VALIDADE,    V_ATIVO,     V_FECHA_APP, EMPTY_BLOB(),
                                        V_ID_MEDICO,   V_ALIAS);

   ELSIF V_FUNCAO = 2 THEN
     UPDATE USUARIOS SET USER_NOME  = V_USER_NOME, SENHA = V_SENHA, GRUPO = V_GRUPO,
                               DATA_CAD   = V_DATA_CAD,  DATA_VALIDADE = V_DATA_VALIDADE,
                               VALIDADE   = V_VALIDADE,  ATIVO = V_ATIVO, FECHA_APP = V_FECHA_APP,
                               ID_MEDICO  = V_ID_MEDICO, ALIAS = V_ALIAS
                            WHERE USER_CODIGO = V_USER_CODIGO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE USUARIOS WHERE USER_CODIGO = V_USER_CODIGO;

   ELSIF V_FUNCAO = 4 THEN
     UPDATE USUARIOS SET SENHA         = V_SENHA,
                         DATA_VALIDADE = NVL(V_DATA_VALIDADE, DATA_VALIDADE)
                            WHERE USER_CODIGO = V_USER_CODIGO;

  END IF;

  COMMIT WORK;

END PRC_USUARIOS;
