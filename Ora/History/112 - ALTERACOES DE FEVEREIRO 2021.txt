create or replace FUNCTION GET_VALOR_EXAMES(V_POSTO           IN NUMBER,
                                            V_ATENDIMENTO     IN NUMBER) RETURN NUMBER IS
L_RESULT NUMBER;

BEGIN
     SELECT SUM(E.VALOR_HON) INTO L_RESULT 
       FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO NOT IN ('A','C','F');
		
    RETURN(L_RESULT);
	
END GET_VALOR_EXAMES;

create or replace FUNCTION GET_COUNT_EXAMES_SITUACAO(V_POSTO       IN NUMBER,
                                                     V_ATENDIMENTO IN NUMBER,
                                                     V_SITUACAO    IN VARCHAR2) RETURN NUMBER IS RESULT NUMBER;
BEGIN
  IF V_SITUACAO = 'FINALIZADO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.PENDENCIA <> 'S'
        AND E.SITUACAO IN ('R','I','L')
        AND E.SITUACAO_INTERFACE IN ('N','L','U','D','T');

  ELSIF V_SITUACAO = 'ANDAMENTO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.PENDENCIA <> 'S'
        AND E.SITUACAO NOT IN ('A','C','F')
        AND E.SITUACAO_INTERFACE IN ('S','E','I','P');

  ELSIF V_SITUACAO = 'AGUARDANDO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.PENDENCIA <> 'S'
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO IN ('R','N','U')
        AND E.SITUACAO_INTERFACE = 'R';

  ELSIF V_SITUACAO = 'PENDENCIA' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO NOT IN ('A','C','F')
        AND E.PENDENCIA = 'S';

  ELSIF V_SITUACAO = 'NAO' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.PENDENCIA <> 'S'
        AND E.SITUACAO IN ('N','U')
        AND E.SITUACAO_INTERFACE = 'N';

  ELSIF V_SITUACAO = 'TODOS' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO NOT IN ('A','C','F');

   END IF;
   RETURN(RESULT);
END GET_COUNT_EXAMES_SITUACAO;

		   A.ID_FATURA,

		   A.CARATER_ATENDIMENTO,
		   A.IDENTIFICACAO_BENEFICIARIO,
		   SUM(C.VALOR) VALOR_PAGO,

		   SUM(D.VALOR) VALOR_DEVEDOR,
		   GET_VALOR_EXAMES(A.POSTO, A.ATENDIMENTO) VALOR_EXAMES,
		   NVL(SUM(C.VALOR), GET_VALOR_EXAMES(A.POSTO, A.ATENDIMENTO)) VALOR_ATENDIMENTO,
   		   GET_COUNT_EXAMES_SITUACAO(A.POSTO, A.ATENDIMENTO, 'TODOS') QT_EXAMES

		   DEVEDORES D
      WHERE CA.POSTO(+) = A.POSTO
        AND CA.ATENDIMENTO(+) = A.ATENDIMENTO
        AND D.CONTA_HOSP(+) = CA.CONTA_HOSP
        AND C.CONTA_HOSP(+) = CA.CONTA_HOSP
		AND C.STATUS(+) = 'A'
   GROUP BY A.POSTO, 
            A.ATENDIMENTO,
	        CA.CONTA_HOSP,
            A.DATA_ATD,

		    A.CONVENIO,

		    A.ID_FATURA,
		    A.GUIA_PRINCIPAL,
		    A.CARATER_ATENDIMENTO,

create or replace FUNCTION GET_ETIQUETA_MNEMONICOS(V_POSTO       IN NUMBER,
                                                   V_ATENDIMENTO IN NUMBER,
										           V_ETIQUETA    IN NUMBER) RETURN VARCHAR2 IS MNEMONICOS VARCHAR2(2000);

    CURSOR CUR_MNEMONICOS IS
      SELECT MNEMONICO 
	    FROM EXAMES 
	   WHERE POSTO = V_POSTO 
	     AND ATENDIMENTO = V_ATENDIMENTO 
		 AND ETIQUETA = V_ETIQUETA;

BEGIN

MNEMONICOS := '';

FOR MNEMONICOS_REC in CUR_MNEMONICOS
LOOP
    MNEMONICOS := MNEMONICOS || MNEMONICOS_REC.MNEMONICO || ' ';
END LOOP;

RETURN MNEMONICOS;

END;

create or replace FUNCTION GET_ETIQUETA_FLAG(V_POSTO       IN NUMBER,
                                             V_ATENDIMENTO IN NUMBER,
										     V_ETIQUETA    IN NUMBER) RETURN VARCHAR2 IS RESULT_FLAG VARCHAR2(20);

    CURSOR CUR_FLAGS IS
      SELECT P.AVISO_ETIQUETA FLAG 
	    FROM EXAMES E, PROCEDIMENTOS P 
	   WHERE E.POSTO = V_POSTO 
	     AND E.ATENDIMENTO = V_ATENDIMENTO 
		 AND E.ETIQUETA = V_ETIQUETA
		 AND P.MNEMONICO = E.MNEMONICO;

BEGIN

RESULT_FLAG := '';

FOR FLAGS_REC in CUR_FLAGS
LOOP
    RESULT_FLAG := RESULT_FLAG || FLAGS_REC.FLAG || ' ';
END LOOP;

RETURN RESULT_FLAG;

END;

create or replace FUNCTION GET_ETIQUETA_URGENTE(V_POSTO       IN NUMBER,
                                                V_ATENDIMENTO IN NUMBER,
										        V_ETIQUETA    IN NUMBER) RETURN VARCHAR2 IS RESULT_URGENTE VARCHAR2(1);

    CURSOR CUR_URGENTE IS
      SELECT IS_URGENTE(POSTO,ATENDIMENTO,CORREL) URGENTE 
	    FROM EXAMES 
	   WHERE POSTO = V_POSTO 
	     AND ATENDIMENTO = V_ATENDIMENTO 
		 AND ETIQUETA = V_ETIQUETA;

BEGIN

RESULT_URGENTE := '';

FOR URGENTE_REC in CUR_URGENTE
LOOP
    IF URGENTE_REC.URGENTE = 'S' THEN
       RESULT_URGENTE := 'U';
	END IF;   
END LOOP;

RETURN RESULT_URGENTE;

END;

create or replace FUNCTION GET_ETIQUETA_TERCEIRIZADO(V_POSTO       IN NUMBER,
                                                     V_ATENDIMENTO IN NUMBER,
										             V_ETIQUETA    IN NUMBER) RETURN VARCHAR2 IS RESULT_TERCEIRIZADO VARCHAR2(1);

    CURSOR CUR_TERCEIRIZADO IS
      SELECT P.TIPO_POSTO
	    FROM EXAMES E, POSTOS P 
	   WHERE E.POSTO = V_POSTO 
	     AND E.ATENDIMENTO = V_ATENDIMENTO 
		 AND E.ETIQUETA = V_ETIQUETA
		 AND P.POSTO = E.POSTO_REA;

BEGIN

RESULT_TERCEIRIZADO := '';

FOR TERCEIRIZADO_REC in CUR_TERCEIRIZADO
LOOP
    IF TERCEIRIZADO_REC.TIPO_POSTO = 'A' THEN
       RESULT_TERCEIRIZADO := 'T';
	END IF;   
END LOOP;

RETURN RESULT_TERCEIRIZADO;

END;
