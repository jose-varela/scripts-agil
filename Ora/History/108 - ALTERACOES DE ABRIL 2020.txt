create or replace PROCEDURE PRC_RESULTADOS_B2B(V_ID_RESULTADO_B2B  IN OUT RESULTADOS_B2B.ID_RESULTADO_B2B%TYPE,
                                       V_POSTO      IN RESULTADOS_B2B.POSTO%TYPE,
                                       V_ATENDIMENTO     IN RESULTADOS_B2B.ATENDIMENTO%TYPE,
                                       V_CORREL    IN RESULTADOS_B2B.CORREL%TYPE,
                                       V_EQUIPAMENTO    IN RESULTADOS_B2B.EQUIPAMENTO%TYPE,
                                       V_DATA_B2B    IN RESULTADOS_B2B.DATA_B2B%TYPE,
                                       V_USUARIO_B2B    IN RESULTADOS_B2B.USUARIO_B2B%TYPE,
                                       V_RESULTADO_EXTERNO    IN RESULTADOS_B2B.RESULTADO_EXTERNO%TYPE,
                                       V_FUNCAO    IN NUMBER) IS      

BEGIN
  IF V_FUNCAO = 1 THEN
	  SELECT SQ_RESULTADOS_B2B.NEXTVAL INTO V_ID_RESULTADO_B2B FROM DUAL;
	 
    INSERT INTO RESULTADOS_B2B VALUES (V_ID_RESULTADO_B2B, V_POSTO, V_ATENDIMENTO, V_CORREL, 
                                       V_EQUIPAMENTO, V_DATA_B2B, V_USUARIO_B2B, V_RESULTADO_EXTERNO);           
  
  ELSIF V_FUNCAO = 2 THEN
     UPDATE RESULTADOS_B2B SET POSTO = V_POSTO, ATENDIMENTO = V_ATENDIMENTO, CORREL = V_CORREL,
                               EQUIPAMENTO = V_EQUIPAMENTO, DATA_B2B = V_DATA_B2B, USUARIO_B2B = V_USUARIO_B2B,
                               RESULTADO_EXTERNO = V_RESULTADO_EXTERNO
                      WHERE ID_RESULTADO_B2B = V_ID_RESULTADO_B2B;	
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE RESULTADOS_B2B WHERE ID_RESULTADO_B2B = V_ID_RESULTADO_B2B;

  ELSIF V_FUNCAO = 4 THEN
     DELETE RESULTADOS_B2B WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  END IF;

  COMMIT WORK;

END PRC_RESULTADOS_B2B;
