--
-- TERMOS_MEDICOS
--
CREATE TABLE TERMOS_MEDICOS 
   (ID_MEDICO       NUMBER(6,0) NOT NULL ENABLE, 
	ID_TERMO        NUMBER(4,0), 
	DATA_CAD        DATE NOT NULL ENABLE, 
	USUARIO         VARCHAR2(10) NOT NULL ENABLE, 
	ID_TERMO_MEDICO NUMBER(4,0))
  TABLESPACE "LIS_DATA" ;
 -- fim bloco

ALTER TABLE TERMOS_MEDICOS ADD CONSTRAINT TERMOS_MEDICOS_PK 
                               PRIMARY KEY (ID_TERMO_MEDICO) ENABLE;
-- fim bloco

CREATE INDEX TERMOS_MED_MEDICO_I ON TERMOS_MEDICOS (ID_MEDICO);
-- fim bloco

ALTER TABLE TERMOS_MEDICOS ADD CONSTRAINT TERMOS_MED_TERMOS_FK FOREIGN KEY (ID_TERMO) REFERENCES TERMOS(ID_TERMO);
-- fim bloco

ALTER TABLE TERMOS_MEDICOS ADD CONSTRAINT TERMOS_MED_MEDICOS_FK FOREIGN KEY (ID_MEDICO) REFERENCES MEDICOS(ID_MEDICO);
-- fim bloco


CREATE SEQUENCE SQ_TERMOS_MEDICOS
MINVALUE 0
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

create or replace
PROCEDURE PRC_TERMOS_MEDICOS(V_ID_MEDICO             IN     TERMOS_MEDICOS.ID_MEDICO%TYPE,
                             V_ID_TERMO              IN     TERMOS_MEDICOS.ID_TERMO%TYPE,
                             V_DATA_CAD              IN     TERMOS_MEDICOS.DATA_CAD%TYPE,
                             V_USUARIO               IN     TERMOS_MEDICOS.USUARIO%TYPE,
                             V_ID_TERMO_MEDICO       IN OUT TERMOS_MEDICOS.ID_TERMO_MEDICO%TYPE,
                             V_FUNCAO                IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_TERMOS_MEDICOS.NEXTVAL INTO V_ID_TERMO_MEDICO FROM DUAL;
     
     INSERT INTO TERMOS_MEDICOS VALUES (V_ID_MEDICO, V_ID_TERMO, V_DATA_CAD, V_USUARIO, V_ID_TERMO_MEDICO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE TERMOS_MEDICOS SET ID_TERMO = V_ID_TERMO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
      WHERE ID_TERMO_MEDICO = V_ID_TERMO_MEDICO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE TERMOS_MEDICOS WHERE ID_TERMO_MEDICO = V_ID_TERMO_MEDICO;

  END IF;

  COMMIT WORK;

END PRC_TERMOS_MEDICOS;

CREATE TABLE IMAGENS_ATENDIMENTOS (
    ID_IMAGEM_ATENDIMENTO NUMBER(6)  NOT NULL,     
    POSTO                 NUMBER(2)  NOT NULL,    
    ATENDIMENTO           NUMBER(9)  NOT NULL,
    IMAGEM_ATENDIMENTO    BLOB, 
    DATA_CAD              DATE,                
    USUARIO               VARCHAR2(10) NOT NULL,
    TIPO_IMAGEM           VARCHAR2(1)  NOT NULL
);
-- fim bloco

ALTER TABLE IMAGENS_ATENDIMENTOS ADD CONSTRAINT IMAGENS_ATENDIMENTOS_PK PRIMARY KEY (ID_IMAGEM_ATENDIMENTO);
-- fim bloco

ALTER TABLE IMAGENS_ATENDIMENTOS ADD CONSTRAINT IMAGENS_ATENDIMENTOS_FK 
    FOREIGN KEY (POSTO, ATENDIMENTO) REFERENCES ATENDIMENTOS(POSTO, ATENDIMENTO);
-- fim bloco


CREATE SEQUENCE SQ_IMAGENS_ATENDIMENTOS
MINVALUE 0
MAXVALUE 999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_IMAGENS_ATENDIMENTOS(
                                        V_ID_IMAGEM_ATENDIMENTO IN OUT IMAGENS_ATENDIMENTOS.ID_IMAGEM_ATENDIMENTO%TYPE,
                                        V_POSTO                 IN     IMAGENS_ATENDIMENTOS.POSTO%TYPE,
                                        V_ATENDIMENTO           IN     IMAGENS_ATENDIMENTOS.ATENDIMENTO%TYPE,
                                        V_IMAGEM_ATENDIMENTO    IN     IMAGENS_ATENDIMENTOS.IMAGEM_ATENDIMENTO%TYPE,
                                        V_DATA_CAD              IN     IMAGENS_ATENDIMENTOS.DATA_CAD%TYPE,
                                        V_USUARIO               IN     IMAGENS_ATENDIMENTOS.USUARIO%TYPE,
                                        V_TIPO_IMAGEM           IN     IMAGENS_ATENDIMENTOS.TIPO_IMAGEM%TYPE,
                                        V_FUNCAO           IN     NUMBER
                                        ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_IMAGENS_ATENDIMENTOS.NEXTVAL INTO V_ID_IMAGEM_ATENDIMENTO FROM DUAL;
        INSERT INTO IMAGENS_ATENDIMENTOS (ID_IMAGEM_ATENDIMENTO, POSTO, ATENDIMENTO, IMAGEM_ATENDIMENTO, DATA_CAD, USUARIO, TIPO_IMAGEM)
        VALUES (V_ID_IMAGEM_ATENDIMENTO, V_POSTO, V_ATENDIMENTO, EMPTY_BLOB, V_DATA_CAD, V_USUARIO, V_TIPO_IMAGEM);

    ELSIF V_FUNCAO = 2 THEN
        UPDATE IMAGENS_ATENDIMENTOS 
        SET POSTO           = V_POSTO,
            ATENDIMENTO     = V_ATENDIMENTO,
            USUARIO         = V_USUARIO,
            DATA_CAD        = V_DATA_CAD,
            TIPO_IMAGEM     = V_TIPO_IMAGEM
        WHERE ID_IMAGEM_ATENDIMENTO = V_ID_IMAGEM_ATENDIMENTO;

    ELSIF V_FUNCAO = 3 THEN
        DELETE FROM IMAGENS_ATENDIMENTOS WHERE ID_IMAGEM_ATENDIMENTO = V_ID_IMAGEM_ATENDIMENTO;

    END IF;

    COMMIT WORK;
END PRC_IMAGENS_ATENDIMENTOS;
