--
-- CAIXA_ALTERACOES - novo tipo 'D' (desconto acima do maximo)
-- Contexto: feature motivo-desconto-acima-maximo (PD-2026-0002, laboratorio
-- Analyses). O desconto acima do maximo passa a exigir e gravar motivo,
-- reaproveitando o historico de alteracoes do caixa ja usado para troca de
-- convenio ('C') e cancelamento de pagamento ('P'). Precisa de dois ajustes
-- na tabela: o CHECK de TIPO so aceitava C/P, e nao havia coluna para
-- guardar o percentual do desconto (so VALOR_CAIXA em R$) -- sem ela o
-- relatorio de caixa nao conseguiria reconstruir o "(Y%)" do efeito ao
-- reler do banco depois de gravado.
--
ALTER TABLE CAIXA_ALTERACOES DROP CONSTRAINT CAIXA_ALTERACOES_TIPO_CK;
-- fim bloco

ALTER TABLE CAIXA_ALTERACOES ADD CONSTRAINT CAIXA_ALTERACOES_TIPO_CK
  CHECK (TIPO IN ('C', 'P', 'D')) ENABLE;
-- fim bloco

ALTER TABLE CAIXA_ALTERACOES ADD (PERCENTUAL NUMBER(5,2) DEFAULT 0 NOT NULL);
-- fim bloco

--
-- EXAMES_ALTERACOES - tabela nova (historico de alteracoes de exame)
-- Contexto: feature motivo-desliberacao-exame-impresso (PD-2026-0003, pedido
-- companheiro do mesmo laboratorio Analyses). Deliberar (eliminar a
-- liberacao de) um exame ja impresso passa a exigir e gravar motivo.
-- Diferente do caixa, EXAMES nao tem equivalente a CAIXA_ALTERACOES -- esta
-- tabela nova cobre esse historico, seguindo a mesma convencao (sequence +
-- tabela + PK + CHECK + indice). Guarda MNEMONICO alem da chave
-- (POSTO/ATENDIMENTO/CORREL) para leitura direto no banco sem join.

CREATE SEQUENCE SEQ_EXAME_ALTERACAO START WITH 1 INCREMENT BY 1 NOCACHE;

-- PRC_MARDUK_FIN_ENFILEIRA_REC - procedure nova (enfileira um pagamento)
-- Contexto: feature marduk-finance-recebimentos (T-02.01). Chamada pela
-- PRC_CAIXA (funcoes 1, 2, 3, 4 e 5; nunca a 6 - importacao, D-05) antes do
-- COMMIT dela, na mesma transacao do pagamento. So enfileira com o Finance
-- ligado (SERVICOS_ADM.MARDUK_FINANCE_ATIVO). Nunca propaga erro: o caixa
-- tem de gravar mesmo se a fila falhar. Nao faz COMMIT.
-- OPERACAO e informativa (a fila do AGIL decide de novo no envio, pelo
-- estado do CAIXA e de ENVIADO).
--
CREATE OR REPLACE PROCEDURE PRC_MARDUK_FIN_ENFILEIRA_REC(V_ID_CAIXA IN CAIXA.ID_CAIXA%TYPE) IS
  V_ATIVO NUMBER;
BEGIN
  SELECT COUNT(*) INTO V_ATIVO FROM SERVICOS_ADM WHERE MARDUK_FINANCE_ATIVO = 'S';
  IF V_ATIVO = 0 THEN
    RETURN;
  END IF;

  MERGE INTO MARDUK_FINANCE_RECEBIMENTOS F
  USING (SELECT ID_CAIXA, POSTO, CONTA_HOSP, STATUS FROM CAIXA WHERE ID_CAIXA = V_ID_CAIXA) C
     ON (F.ID_CAIXA = C.ID_CAIXA)
  WHEN MATCHED THEN UPDATE SET
       F.OPERACAO          = CASE WHEN C.STATUS = 'C' THEN 'CANCELAR'
                                  WHEN F.ENVIADO = 'S' THEN 'ALTERAR'
                                  ELSE 'CRIAR' END,
       F.ENVIAR            = 'S',
       F.TENTATIVAS        = 0,
       F.PROXIMA_TENTATIVA = NULL,
       F.ULTIMO_ERRO       = NULL,
       F.ATUALIZADO_EM     = SYSDATE
  WHEN NOT MATCHED THEN INSERT (ID_CAIXA, POSTO, CONTA_HOSP, OPERACAO)
       VALUES (C.ID_CAIXA, C.POSTO, C.CONTA_HOSP, CASE WHEN C.STATUS = 'C' THEN 'CANCELAR' ELSE 'CRIAR' END);
EXCEPTION
  WHEN OTHERS THEN
    NULL;
END PRC_MARDUK_FIN_ENFILEIRA_REC;

--
-- PRC_CAIXA - recriada (enfileira o pagamento no Marduk Finance)
-- Contexto: feature marduk-finance-recebimentos (T-02.02). Corpo identico ao
-- da USER_SOURCE de 2026-10-15 (o prc_caixa.sql do repositorio esta
-- desatualizado), mais a chamada a PRC_MARDUK_FIN_ENFILEIRA_REC antes do
-- COMMIT (D-04, D-17): 1 inclui, 2 e 5 alteram, 4 cancela um, 3 cancela os
-- que estavam ativos na conta, 6 (importacao) nao enfileira (D-05).
-- A procedure de enfileiramento nunca propaga erro; o caixa grava sempre.
--
CREATE OR REPLACE PROCEDURE PRC_CAIXA(V_CONTA_HOSP      IN CAIXA.CONTA_HOSP%TYPE,
                                       V_REGISTRO       IN CAIXA.REGISTRO%TYPE,
                                       V_CONTA          IN CAIXA.CONTA%TYPE,
                                       V_DATA_PAG       IN CAIXA.DATA_PAG%TYPE,
                                       V_VALOR          IN CAIXA.VALOR%TYPE,
                                       V_DESCONTO       IN CAIXA.DESCONTO%TYPE,
                                       V_TIPO_DTO       IN CAIXA.TIPO_DTO%TYPE,
                                       V_USUARIO        IN CAIXA.USUARIO%TYPE,
                                       V_DATA_CAD       IN CAIXA.DATA_CAD%TYPE,
                                       V_CORREL         IN OUT CAIXA.CORREL%TYPE,
                                       V_ID_FORMA_PAGAMENTO       IN CAIXA.ID_FORMA_PAGAMENTO%TYPE,
                                       V_CATALOGO       IN CAIXA.CATALOGO%TYPE,
                                       V_USUARIO_DTO    IN CAIXA.USUARIO_DTO%TYPE,
                                       V_USUARIO_ALT    IN CAIXA.USUARIO_ALT%TYPE,
                                       V_DATA_ALT       IN CAIXA.DATA_ALT%TYPE,
                                       V_ID_BANDEIRA    IN CAIXA.ID_BANDEIRA%TYPE,
                                       V_ID_OPERADORA   IN CAIXA.ID_OPERADORA%TYPE,
                                       V_COD_PAGAMENTO  IN CAIXA.COD_PAGAMENTO%TYPE,
                                       V_PARCELAS       IN CAIXA.PARCELAS%TYPE,
                                       V_ID_CAIXA       IN OUT CAIXA.ID_CAIXA%TYPE,
                                       V_ID_RESPONSAVEL_FINANCEIRO IN CAIXA.ID_RESPONSAVEL_FINANCEIRO%TYPE,
                                       V_STATUS         IN CAIXA.STATUS%TYPE,
                                       V_POSTO          IN CAIXA.POSTO%TYPE,
                                       V_FUNCAO         IN NUMBER) IS
   TYPE T_IDS IS TABLE OF CAIXA.ID_CAIXA%TYPE;
   V_IDS T_IDS := T_IDS();

   -- Marduk Finance: pagamento de POSTO+CONTA_HOSP+REGISTRO+CORREL (2, 4 e 5)
   PROCEDURE ENFILEIRA_PAGAMENTO IS
   BEGIN
     FOR R IN (SELECT ID_CAIXA FROM CAIXA
                WHERE POSTO = V_POSTO AND CONTA_HOSP = V_CONTA_HOSP
                  AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL) LOOP
       PRC_MARDUK_FIN_ENFILEIRA_REC(R.ID_CAIXA);
     END LOOP;
   EXCEPTION
     WHEN OTHERS THEN NULL;
   END;
 BEGIN
   IF (V_FUNCAO = 1) OR (V_FUNCAO = 6) THEN
      IF (V_FUNCAO = 1) THEN
         SELECT NVL(MAX(C.CORREL)+1,1) INTO V_CORREL FROM CAIXA C
         WHERE C.POSTO = V_POSTO
           AND C.CONTA_HOSP = V_CONTA_HOSP;
      END IF;
      SELECT SQ_CAIXA.NEXTVAL INTO V_ID_CAIXA FROM DUAL;
      INSERT INTO CAIXA VALUES (V_CONTA_HOSP,  V_REGISTRO,    V_CONTA,     V_DATA_PAG,
                                V_VALOR,       V_DESCONTO,    V_TIPO_DTO,  V_USUARIO,
                                V_DATA_CAD,    V_CORREL,      V_ID_FORMA_PAGAMENTO,  V_CATALOGO,
                                V_USUARIO_DTO, V_USUARIO_ALT, V_DATA_ALT, V_ID_BANDEIRA,
                                V_ID_OPERADORA, V_COD_PAGAMENTO, V_PARCELAS, V_ID_CAIXA,
                                V_ID_RESPONSAVEL_FINANCEIRO, V_STATUS, V_POSTO);
      -- Marduk Finance: so o pagamento feito aqui; o importado (6) nao (D-05)
      IF (V_FUNCAO = 1) THEN
         PRC_MARDUK_FIN_ENFILEIRA_REC(V_ID_CAIXA);
      END IF;
   ELSIF V_FUNCAO = 2 THEN
      UPDATE CAIXA SET VALOR = V_VALOR
                       WHERE POSTO = V_POSTO
                         AND CONTA_HOSP = V_CONTA_HOSP
                         AND REGISTRO = V_REGISTRO
                         AND CORREL = V_CORREL;
      ENFILEIRA_PAGAMENTO;
   ELSIF V_FUNCAO = 3 THEN
     -- Marduk Finance: so os que estavam ativos viram cancelamento
     BEGIN
       SELECT ID_CAIXA BULK COLLECT INTO V_IDS FROM CAIXA
        WHERE POSTO = V_POSTO AND CONTA_HOSP = V_CONTA_HOSP AND STATUS = 'A';
     EXCEPTION
       WHEN OTHERS THEN V_IDS := T_IDS();
     END;
     UPDATE CAIXA SET STATUS = 'C' WHERE POSTO = V_POSTO
                                     AND CONTA_HOSP = V_CONTA_HOSP;
     FOR I IN 1 .. V_IDS.COUNT LOOP
       PRC_MARDUK_FIN_ENFILEIRA_REC(V_IDS(I));
     END LOOP;
   ELSIF V_FUNCAO = 4 THEN
     UPDATE CAIXA SET STATUS = 'C', USUARIO_ALT = V_USUARIO_ALT, DATA_ALT = V_DATA_ALT
      WHERE POSTO = V_POSTO
        AND CONTA_HOSP = V_CONTA_HOSP
        AND REGISTRO = V_REGISTRO
        AND CORREL = V_CORREL;
     ENFILEIRA_PAGAMENTO;
   ELSIF V_FUNCAO = 5 THEN
      UPDATE CAIXA SET VALOR = V_VALOR,
                       DATA_PAG = V_DATA_PAG,
                       DESCONTO = V_DESCONTO,
                       TIPO_DTO = V_TIPO_DTO,
                       ID_FORMA_PAGAMENTO = V_ID_FORMA_PAGAMENTO,
                       USUARIO_ALT = V_USUARIO_ALT,
                       DATA_ALT = V_DATA_ALT,
                       ID_BANDEIRA = V_ID_BANDEIRA,
                       ID_OPERADORA = V_ID_OPERADORA,
                       COD_PAGAMENTO = V_COD_PAGAMENTO,
                       PARCELAS = V_PARCELAS,
                       ID_RESPONSAVEL_FINANCEIRO = V_ID_RESPONSAVEL_FINANCEIRO
                  WHERE POSTO = V_POSTO
                    AND CONTA_HOSP = V_CONTA_HOSP
                    AND REGISTRO = V_REGISTRO
                    AND CORREL = V_CORREL;
      ENFILEIRA_PAGAMENTO;
   END IF;
   COMMIT WORK;
 END PRC_CAIXA;
/

--
-- finance-apoio T-02.02 - PRC_POSTOS_APOIO: grava CNPJ e configuracao de um posto de apoio (tipo 'A'), sem tocar a assinatura de PRC_POSTOS (D-17)
--
-- >>> finance-apoio T-02.02 inicio
CREATE OR REPLACE PROCEDURE PRC_POSTOS_APOIO(V_POSTO                   IN POSTOS.POSTO%TYPE,
                                             V_CNPJ                    IN VARCHAR2,
                                             V_SITUACOES_COBRAVEIS     IN POSTOS.APOIO_SITUACOES_COBRAVEIS%TYPE,
                                             V_COBRA_CORTESIA          IN POSTOS.APOIO_COBRA_CORTESIA%TYPE,
                                             V_ENVIA_NAO_FINALIZADO    IN POSTOS.APOIO_ENVIA_NAO_FINALIZADO%TYPE) IS
  V_TIPO POSTOS.TIPO_POSTO%TYPE;
  V_DIGITOS VARCHAR2(40);
BEGIN
  BEGIN
    SELECT TIPO_POSTO INTO V_TIPO FROM POSTOS WHERE POSTO = V_POSTO;
  EXCEPTION WHEN NO_DATA_FOUND THEN
    RAISE_APPLICATION_ERROR(-20101, 'Posto inexistente: ' || V_POSTO);
  END;
  IF V_TIPO <> 'A' THEN
    RAISE_APPLICATION_ERROR(-20101, 'O posto ' || V_POSTO || ' nao e um laboratorio de apoio (tipo A)');
  END IF;

  V_DIGITOS := REGEXP_REPLACE(V_CNPJ, '[^0-9]', '');
  IF V_DIGITOS IS NOT NULL AND LENGTH(V_DIGITOS) <> 14 THEN
    RAISE_APPLICATION_ERROR(-20102, 'CNPJ deve ter 14 digitos');
  END IF;

  IF V_SITUACOES_COBRAVEIS IS NULL OR NOT REGEXP_LIKE(V_SITUACOES_COBRAVEIS, '^[ACDEFIJLNOPRT](,[ACDEFIJLNOPRT])*$') THEN
    RAISE_APPLICATION_ERROR(-20103, 'Situacoes cobraveis invalidas: use letras de situacao B2B separadas por virgula');
  END IF;

  UPDATE POSTOS SET CNPJ = V_DIGITOS,
                    APOIO_SITUACOES_COBRAVEIS = V_SITUACOES_COBRAVEIS,
                    APOIO_COBRA_CORTESIA = V_COBRA_CORTESIA,
                    APOIO_ENVIA_NAO_FINALIZADO = V_ENVIA_NAO_FINALIZADO
              WHERE POSTO = V_POSTO;
  COMMIT WORK;
END PRC_POSTOS_APOIO;

-- finance-apoio T-02.03 - preco do apoio por exame com vigencia (D-06): manutencao e preco vigente
--
-- >>> finance-apoio T-02.03 inicio
CREATE OR REPLACE PROCEDURE PRC_POSTOS_APOIO_PRECOS(V_ID_PRECO     IN OUT POSTOS_APOIO_PRECOS.ID_PRECO%TYPE,
                                                    V_POSTO        IN     POSTOS_APOIO_PRECOS.POSTO%TYPE,
                                                    V_MNEMONICO    IN     POSTOS_APOIO_PRECOS.MNEMONICO%TYPE,
                                                    V_VALOR        IN     POSTOS_APOIO_PRECOS.VALOR%TYPE,
                                                    V_VIGENCIA_INI IN     POSTOS_APOIO_PRECOS.VIGENCIA_INI%TYPE,
                                                    V_VIGENCIA_FIM IN     POSTOS_APOIO_PRECOS.VIGENCIA_FIM%TYPE,
                                                    V_FUNCAO       IN     NUMBER) IS
  V_TIPO POSTOS.TIPO_POSTO%TYPE;
  V_MNEMO POSTOS_APOIO_PRECOS.MNEMONICO%TYPE;
  V_INI DATE;
  V_FIM DATE;
  V_N NUMBER;
BEGIN
  IF V_FUNCAO IN (1, 2) THEN
    BEGIN
      SELECT TIPO_POSTO INTO V_TIPO FROM POSTOS WHERE POSTO = V_POSTO;
    EXCEPTION WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20101, 'Posto inexistente: ' || V_POSTO);
    END;
    IF V_TIPO <> 'A' THEN
      RAISE_APPLICATION_ERROR(-20101, 'O posto ' || V_POSTO || ' nao e um laboratorio de apoio (tipo A)');
    END IF;

    V_MNEMO := UPPER(TRIM(V_MNEMONICO));
    V_INI := TRUNC(V_VIGENCIA_INI);
    V_FIM := TRUNC(V_VIGENCIA_FIM);

    SELECT COUNT(*) INTO V_N FROM POSTOS_APOIO_PRECOS
     WHERE POSTO = V_POSTO AND UPPER(TRIM(MNEMONICO)) = V_MNEMO
       AND ID_PRECO <> NVL(V_ID_PRECO, -1)
       AND VIGENCIA_INI <= NVL(V_FIM, DATE '9999-12-31')
       AND NVL(VIGENCIA_FIM, DATE '9999-12-31') >= V_INI;
    IF V_N > 0 THEN
      RAISE_APPLICATION_ERROR(-20111, 'Ja existe preco deste exame neste apoio em parte desse periodo de vigencia');
    END IF;
  END IF;

  IF V_FUNCAO = 1 THEN
    SELECT SQ_POSTOS_APOIO_PRECOS.NEXTVAL INTO V_ID_PRECO FROM DUAL;
    INSERT INTO POSTOS_APOIO_PRECOS (ID_PRECO, POSTO, MNEMONICO, VALOR, VIGENCIA_INI, VIGENCIA_FIM)
         VALUES (V_ID_PRECO, V_POSTO, V_MNEMO, V_VALOR, V_INI, V_FIM);
  ELSIF V_FUNCAO = 2 THEN
    UPDATE POSTOS_APOIO_PRECOS SET POSTO = V_POSTO, MNEMONICO = V_MNEMO, VALOR = V_VALOR,
                                   VIGENCIA_INI = V_INI, VIGENCIA_FIM = V_FIM
     WHERE ID_PRECO = V_ID_PRECO;
  ELSIF V_FUNCAO = 3 THEN
    DELETE POSTOS_APOIO_PRECOS WHERE ID_PRECO = V_ID_PRECO;
  END IF;
  COMMIT WORK;
END PRC_POSTOS_APOIO_PRECOS;
/

CREATE OR REPLACE FUNCTION FNC_PRECO_APOIO(V_POSTO     IN NUMBER,
                                           V_MNEMONICO IN VARCHAR2,
                                           V_DATA      IN DATE) RETURN NUMBER IS
  V_VALOR NUMBER;
BEGIN
  SELECT VALOR INTO V_VALOR FROM (
    SELECT VALOR FROM POSTOS_APOIO_PRECOS
     WHERE POSTO = V_POSTO
       AND UPPER(TRIM(MNEMONICO)) = UPPER(TRIM(V_MNEMONICO))
       AND VIGENCIA_INI <= TRUNC(V_DATA)
       AND (VIGENCIA_FIM IS NULL OR VIGENCIA_FIM >= TRUNC(V_DATA))
     ORDER BY VIGENCIA_INI DESC)
   WHERE ROWNUM = 1;
  RETURN V_VALOR;
EXCEPTION WHEN NO_DATA_FOUND THEN
  RETURN NULL;
END FNC_PRECO_APOIO;

--
-- finance-apoio T-02.04 - reenfileira o lancamento quando o B2B de um exame de apoio muda (D-12)
-- Varios caminhos gravam EXAMES_B2B (PRC_EXAMES_B2B, UPDATE direto, outros executaveis): o trigger cobre todos.
-- So age com MARDUK_FINANCE_ATIVO e MARDUK_FINANCE_APOIO_CFG.ATIVO ligados, em exame cujo posto realizante e de apoio,
-- e so em lancamento ja criado no Finance (ID_LANCAMENTO) e nao cancelado. Nunca derruba a gravacao do exame.
--
-- >>> finance-apoio T-02.04 inicio
CREATE OR REPLACE PROCEDURE PRC_MARDUK_FIN_ENFILEIRA_LANC(V_POSTO IN NUMBER, V_ATENDIMENTO IN NUMBER, V_CORREL IN NUMBER) IS
  V_ATIVO NUMBER;
  V_APOIO NUMBER;
  V_E_APOIO NUMBER;
BEGIN
  SELECT COUNT(*) INTO V_ATIVO FROM SERVICOS_ADM WHERE MARDUK_FINANCE_ATIVO = 'S';
  IF V_ATIVO = 0 THEN RETURN; END IF;
  SELECT COUNT(*) INTO V_APOIO FROM MARDUK_FINANCE_APOIO_CFG WHERE ATIVO = 'S';
  IF V_APOIO = 0 THEN RETURN; END IF;

  SELECT COUNT(*) INTO V_E_APOIO FROM EXAMES E, POSTOS P
   WHERE E.POSTO = V_POSTO AND E.ATENDIMENTO = V_ATENDIMENTO AND E.CORREL = V_CORREL
     AND P.POSTO = E.POSTO_REA AND P.TIPO_POSTO = 'A';
  IF V_E_APOIO = 0 THEN RETURN; END IF;

  UPDATE MARDUK_FINANCE_LANCAMENTOS
     SET OPERACAO = 'ALTERAR', ENVIAR = 'S', TENTATIVAS = 0, PROXIMA_TENTATIVA = NULL,
         ULTIMO_ERRO = NULL, ATUALIZADO_EM = SYSDATE
   WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO
     AND ID_LANCAMENTO IS NOT NULL AND OPERACAO <> 'CANCELAR';
EXCEPTION
  WHEN OTHERS THEN
    NULL;
END PRC_MARDUK_FIN_ENFILEIRA_LANC;
