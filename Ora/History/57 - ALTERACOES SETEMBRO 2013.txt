CREATE OR REPLACE PROCEDURE PRC_ATENDIMENTOS_B2B(V_POSTO                   IN ATENDIMENTOS_B2B.POSTO%TYPE,
                                                 V_ATENDIMENTO             IN ATENDIMENTOS_B2B.ATENDIMENTO%TYPE,
                                                 V_ID_SERVICO_B2B          IN ATENDIMENTOS_B2B.ID_SERVICO_B2B%TYPE,
                                                 V_CODIGO_EXTERNO          IN ATENDIMENTOS_B2B.CODIGO_EXTERNO%TYPE,
                                                 V_DATA_B2B                IN ATENDIMENTOS_B2B.DATA_B2B%TYPE,
                                                 V_SITUACAO_B2B            IN ATENDIMENTOS_B2B.SITUACAO_B2B%TYPE,
                                                 V_FUNCAO                  IN NUMBER) IS
BEGIN
     DECLARE
            V_ACHEI  NUMBER(8);
            
            CURSOR CUR_PROCURA IS
                   SELECT ATENDIMENTO FROM ATENDIMENTOS_B2B
                    WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND ID_SERVICO_B2B = V_ID_SERVICO_B2B;
      BEGIN
           IF V_FUNCAO = 1 THEN
              INSERT INTO ATENDIMENTOS_B2B VALUES (V_POSTO, V_ATENDIMENTO, V_ID_SERVICO_B2B, V_CODIGO_EXTERNO, V_DATA_B2B, V_SITUACAO_B2B);

           ELSIF V_FUNCAO = 2 THEN
              UPDATE ATENDIMENTOS_B2B SET CODIGO_EXTERNO = V_CODIGO_EXTERNO, SITUACAO_B2B = V_SITUACAO_B2B, DATA_B2B = V_DATA_B2B
	                                WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND ID_SERVICO_B2B = V_ID_SERVICO_B2B;

           ELSIF V_FUNCAO = 3 THEN
              DELETE ATENDIMENTOS_B2B WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND ID_SERVICO_B2B = V_ID_SERVICO_B2B;
	 
           ELSIF V_FUNCAO = 4 THEN
                OPEN CUR_PROCURA;
                FETCH CUR_PROCURA INTO V_ACHEI;
  
                IF CUR_PROCURA%NOTFOUND THEN
                   INSERT INTO ATENDIMENTOS_B2B VALUES (V_POSTO, V_ATENDIMENTO, V_ID_SERVICO_B2B, V_CODIGO_EXTERNO, V_DATA_B2B, V_SITUACAO_B2B);
				ELSE   
                    UPDATE ATENDIMENTOS_B2B SET CODIGO_EXTERNO = V_CODIGO_EXTERNO, SITUACAO_B2B = V_SITUACAO_B2B, DATA_B2B = V_DATA_B2B
	                                      WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND ID_SERVICO_B2B = V_ID_SERVICO_B2B;
                END IF;
           END IF;
      END;
	  
      COMMIT WORK;
END PRC_ATENDIMENTOS_B2B;

CREATE OR REPLACE PROCEDURE PRC_CLIENTES_B2B(V_REGISTRO           IN CLIENTES_B2B.REGISTRO%TYPE,
                                             V_ID_SERVICO_B2B     IN CLIENTES_B2B.ID_SERVICO_B2B%TYPE,
                                             V_CODIGO_EXTERNO     IN CLIENTES_B2B.CODIGO_EXTERNO%TYPE,

                                             V_DATA_NAS_EXTERNO   IN CLIENTES_B2B.DATA_NAS_EXTERNO%TYPE,
                                             V_SEXO_EXTERNO       IN CLIENTES_B2B.SEXO_EXTERNO%TYPE,
                                             V_IDENTIDADE_EXTERNO IN CLIENTES_B2B.IDENTIDADE_EXTERNO%TYPE,
                                             V_CPF_EXTERNO        IN CLIENTES_B2B.CPF_EXTERNO%TYPE,
                                             V_FUNCAO             IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO CLIENTES_B2B VALUES (V_REGISTRO, V_ID_SERVICO_B2B, V_CODIGO_EXTERNO, V_NOME_EXTERNO, V_DATA_NAS_EXTERNO, V_SEXO_EXTERNO, V_IDENTIDADE_EXTERNO, V_CPF_EXTERNO);

  ELSIF V_FUNCAO = 2 THEN
        UPDATE CLIENTES_B2B SET NOME_EXTERNO = V_NOME_EXTERNO, DATA_NAS_EXTERNO = V_DATA_NAS_EXTERNO, SEXO_EXTERNO = V_SEXO_EXTERNO, IDENTIDADE_EXTERNO = V_IDENTIDADE_EXTERNO, CPF_EXTERNO = V_CPF_EXTERNO
                          WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CLIENTES_B2B WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
                            
  ELSIF V_FUNCAO = 4 THEN

                         WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
