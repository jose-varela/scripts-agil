-- =========================

CREATE TABLE CANAIS_COMUNICACAO (
  ID_CANAL_COMUNICACAO  NUMBER        NOT NULL,
  CODIGO_CANAL          VARCHAR2(50)  NOT NULL,
  NOME_CANAL            VARCHAR2(100) NOT NULL,
  DESCRICAO_CANAL       VARCHAR2(255),
  TIPO_CANAL            CHAR(5)       NOT NULL,   -- Ex.: 'WHATS','MAIL','SMS','PUSH'...
  ATIVO                 CHAR(1)       DEFAULT 'S' NOT NULL,
  CONFIGURACAO_CANAL    CLOB,
  CONSTRAINT CANAIS_COMUNICACAO_PK        PRIMARY KEY (ID_CANAL_COMUNICACAO),
  CONSTRAINT CANAIS_COMUNICACAO_ATIVO_CK  CHECK (ATIVO IN ('S','N')),
  CONSTRAINT CANAIS_COMUNICACAO_CODIGO_UK UNIQUE (CODIGO_CANAL)
);
-- fim bloco

CREATE TABLE TEMPLATES_COMUNICACAO (
  ID_TEMPLATE_COMUNICACAO NUMBER NOT NULL,
  NOME_TEMPLATE           VARCHAR2(50) NOT NULL,
  CONTEUDO_TEMPLATE       CLOB   NOT NULL,
  CONSTRAINT TEMPLATES_COMUNICACAO_PK PRIMARY KEY (ID_TEMPLATE_COMUNICACAO)
);
-- fim bloco

CREATE TABLE EVENTOS_COMUNICACAO (
  ID_EVENTO_COMUNICACAO    NUMBER        NOT NULL,
  ID_CANAL_COMUNICACAO     NUMBER        NOT NULL,
  ID_TEMPLATE_COMUNICACAO  NUMBER        NOT NULL,
  NOME_EVENTO              VARCHAR2(100),
  TAG_EVENTO               VARCHAR2(50)  NOT NULL,
  EXPRESSAO_LOGICA         VARCHAR2(255) NOT NULL,
  ORDEM_PRIORIDADE         NUMBER        DEFAULT 1,
  CONSTRAINT EVENTOS_COMUNICACAO_PK          PRIMARY KEY (ID_EVENTO_COMUNICACAO),
  CONSTRAINT EVENTOS_COMUNICACAO_CANAL_FK    FOREIGN KEY (ID_CANAL_COMUNICACAO)
      REFERENCES CANAIS_COMUNICACAO (ID_CANAL_COMUNICACAO),
  CONSTRAINT EVENTOS_COM_TPL_FK              FOREIGN KEY (ID_TEMPLATE_COMUNICACAO)
      REFERENCES TEMPLATES_COMUNICACAO (ID_TEMPLATE_COMUNICACAO),
  CONSTRAINT EVENTOS_COM_TAGEVT_PRIOR_UK     UNIQUE (TAG_EVENTO, ORDEM_PRIORIDADE)
);
-- fim bloco

CREATE TABLE ENVIOS_COMUNICACAO_LOG (
  ID_ENVIO_COMUNICACAO_LOG NUMBER        NOT NULL,
  ID_EVENTO_COMUNICACAO    NUMBER        NOT NULL,
  ID_CANAL_COMUNICACAO     NUMBER        NOT NULL,
  ID_TEMPLATE_COMUNICACAO  NUMBER        NOT NULL,
  ORIGEM                   VARCHAR2(150) NOT NULL,
  DESTINO                  VARCHAR2(150) NOT NULL,
  NOME_PACIENTE            VARCHAR2(150),
  DATA_ENVIO               DATE          DEFAULT SYSDATE,
  STATUS                   VARCHAR2(20),
  MENSAGEM_ERRO            CLOB,
  CONSTRAINT ENVIOS_COMUNICACAO_LOG_PK     PRIMARY KEY (ID_ENVIO_COMUNICACAO_LOG),
  CONSTRAINT ENVIOS_COM_LOG_EVT_FK         FOREIGN KEY (ID_EVENTO_COMUNICACAO)
      REFERENCES EVENTOS_COMUNICACAO (ID_EVENTO_COMUNICACAO),
  CONSTRAINT ENVIOS_COM_LOG_CANAL_FK       FOREIGN KEY (ID_CANAL_COMUNICACAO)
      REFERENCES CANAIS_COMUNICACAO (ID_CANAL_COMUNICACAO),
  CONSTRAINT ENVIOS_COM_LOG_TPL_FK         FOREIGN KEY (ID_TEMPLATE_COMUNICACAO)
      REFERENCES TEMPLATES_COMUNICACAO (ID_TEMPLATE_COMUNICACAO),
  CONSTRAINT ENVIOS_COM_LOG_STS_CK         CHECK (STATUS IN ('SUCESSO','FALHA','IGNORADO'))
);
-- fim bloco

CREATE TABLE FILA_COMUNICACAO (
  ID_FILA_COMUNICACAO   NUMBER        NOT NULL,
  TAG_EVENTO            VARCHAR2(50)  NOT NULL,
  ID_REFERENCIA         VARCHAR2(100),
  ORIGEM                VARCHAR2(150) NOT NULL,
  DESTINO               VARCHAR2(150) NOT NULL,
  STATUS                VARCHAR2(20)  DEFAULT 'PENDENTE' NOT NULL, -- PENDENTE|PROCESSANDO|ENVIADO|ERRO|ABORTADO
  PRIORIDADE            NUMBER        DEFAULT 0           NOT NULL,
  TENTATIVAS            NUMBER        DEFAULT 0           NOT NULL,
  MAX_TENTATIVAS        NUMBER        DEFAULT 5           NOT NULL,
  PROXIMA_TENTATIVA     DATE,
  AGENDAR_PARA          DATE,
  RESERVADO_POR         VARCHAR2(100),
  DATA_RESERVA          DATE,
  VISIVEL_ATE           DATE,
  PAYLOAD               CLOB,
  MENSAGEM_ERRO         VARCHAR2(4000),
  DATA_INSERCAO         DATE          DEFAULT SYSDATE     NOT NULL,
  DATA_PROCESSAMENTO    DATE,
  CHAVE_IDEMPOTENTE     VARCHAR2(200),

  CONSTRAINT FILA_COMUNICACAO_PK        PRIMARY KEY (ID_FILA_COMUNICACAO),
  CONSTRAINT FILA_COM_STS_CK            CHECK (STATUS IN ('PENDENTE','PROCESSANDO','ENVIADO','ERRO','ABORTADO'))
);

-- CANAIS_COMUNICACAO
CREATE OR REPLACE PROCEDURE PRC_CANAIS_COMUNICACAO (
  V_ID_CANAL_COMUNICACAO  IN OUT CANAIS_COMUNICACAO.ID_CANAL_COMUNICACAO%TYPE,
  V_CODIGO_CANAL          IN     CANAIS_COMUNICACAO.CODIGO_CANAL%TYPE,
  V_NOME_CANAL            IN     CANAIS_COMUNICACAO.NOME_CANAL%TYPE,
  V_DESCRICAO_CANAL       IN     CANAIS_COMUNICACAO.DESCRICAO_CANAL%TYPE,
  V_TIPO_CANAL            IN     CANAIS_COMUNICACAO.TIPO_CANAL%TYPE,
  V_ATIVO                 IN     CANAIS_COMUNICACAO.ATIVO%TYPE,
  V_CONFIGURACAO_CANAL    IN     CANAIS_COMUNICACAO.CONFIGURACAO_CANAL%TYPE,
  V_FUNCAO                IN     NUMBER
) IS
BEGIN
  IF V_FUNCAO = 1 THEN
    SELECT SQ_CANAIS_COMUNICACAO.NEXTVAL INTO V_ID_CANAL_COMUNICACAO FROM DUAL;
    INSERT INTO CANAIS_COMUNICACAO (ID_CANAL_COMUNICACAO, CODIGO_CANAL, NOME_CANAL, DESCRICAO_CANAL, TIPO_CANAL, ATIVO, CONFIGURACAO_CANAL) 
	                        VALUES (V_ID_CANAL_COMUNICACAO, V_CODIGO_CANAL, V_NOME_CANAL, V_DESCRICAO_CANAL, V_TIPO_CANAL, V_ATIVO, V_CONFIGURACAO_CANAL);
							
  ELSIF V_FUNCAO = 2 THEN
    UPDATE CANAIS_COMUNICACAO SET
      CODIGO_CANAL = V_CODIGO_CANAL,
      NOME_CANAL = V_NOME_CANAL,
      DESCRICAO_CANAL = V_DESCRICAO_CANAL,
      TIPO_CANAL = V_TIPO_CANAL,
      ATIVO = V_ATIVO,
      CONFIGURACAO_CANAL = V_CONFIGURACAO_CANAL
    WHERE ID_CANAL_COMUNICACAO = V_ID_CANAL_COMUNICACAO;

  ELSIF V_FUNCAO = 3 THEN
    DELETE FROM CANAIS_COMUNICACAO WHERE ID_CANAL_COMUNICACAO = V_ID_CANAL_COMUNICACAO;
  END IF;

  COMMIT WORK;
END PRC_CANAIS_COMUNICACAO;

-- TEMPLATES_COMUNICACAO
CREATE OR REPLACE PROCEDURE PRC_TEMPLATES_COMUNICACAO (
  V_ID_TEMPLATE_COMUNICACAO IN OUT TEMPLATES_COMUNICACAO.ID_TEMPLATE_COMUNICACAO%TYPE,
  V_NOME_TEMPLATE           IN     TEMPLATES_COMUNICACAO.NOME_TEMPLATE%TYPE,
  V_CONTEUDO_TEMPLATE       IN     TEMPLATES_COMUNICACAO.CONTEUDO_TEMPLATE%TYPE,
  V_FUNCAO                  IN     NUMBER
) IS
BEGIN
  IF V_FUNCAO = 1 THEN
    SELECT SQ_TEMPLATES_COMUNICACAO.NEXTVAL INTO V_ID_TEMPLATE_COMUNICACAO FROM DUAL;
    INSERT INTO TEMPLATES_COMUNICACAO (ID_TEMPLATE_COMUNICACAO, NOME_TEMPLATE, CONTEUDO_TEMPLATE)
    VALUES (V_ID_TEMPLATE_COMUNICACAO, V_NOME_TEMPLATE, V_CONTEUDO_TEMPLATE);
  ELSIF V_FUNCAO = 2 THEN
    UPDATE TEMPLATES_COMUNICACAO
       SET CONTEUDO_TEMPLATE = V_CONTEUDO_TEMPLATE
     WHERE ID_TEMPLATE_COMUNICACAO = V_ID_TEMPLATE_COMUNICACAO;
  ELSIF V_FUNCAO = 3 THEN
    DELETE FROM TEMPLATES_COMUNICACAO WHERE ID_TEMPLATE_COMUNICACAO = V_ID_TEMPLATE_COMUNICACAO;
  END IF;
  COMMIT WORK;
END PRC_TEMPLATES_COMUNICACAO;

-- EVENTOS_COMUNICACAO
CREATE OR REPLACE PROCEDURE PRC_EVENTOS_COMUNICACAO (
  V_ID_EVENTO_COMUNICACAO    IN OUT EVENTOS_COMUNICACAO.ID_EVENTO_COMUNICACAO%TYPE,
  V_ID_CANAL_COMUNICACAO     IN     EVENTOS_COMUNICACAO.ID_CANAL_COMUNICACAO%TYPE,
  V_ID_TEMPLATE_COMUNICACAO  IN     EVENTOS_COMUNICACAO.ID_TEMPLATE_COMUNICACAO%TYPE,
  V_NOME_EVENTO              IN     EVENTOS_COMUNICACAO.NOME_EVENTO%TYPE,
  V_TAG_EVENTO               IN     EVENTOS_COMUNICACAO.TAG_EVENTO%TYPE,
  V_EXPRESSAO_LOGICA         IN     EVENTOS_COMUNICACAO.EXPRESSAO_LOGICA%TYPE,
  V_ORDEM_PRIORIDADE         IN     EVENTOS_COMUNICACAO.ORDEM_PRIORIDADE%TYPE,
  V_FUNCAO                   IN     NUMBER
) IS
BEGIN
  IF V_FUNCAO = 1 THEN
    SELECT SQ_EVENTOS_COMUNICACAO.NEXTVAL INTO V_ID_EVENTO_COMUNICACAO FROM DUAL;
    INSERT INTO EVENTOS_COMUNICACAO (
      ID_EVENTO_COMUNICACAO, ID_CANAL_COMUNICACAO, ID_TEMPLATE_COMUNICACAO,
      NOME_EVENTO, TAG_EVENTO, EXPRESSAO_LOGICA, ORDEM_PRIORIDADE
    ) VALUES (
      V_ID_EVENTO_COMUNICACAO, V_ID_CANAL_COMUNICACAO, V_ID_TEMPLATE_COMUNICACAO,
      V_NOME_EVENTO, V_TAG_EVENTO, V_EXPRESSAO_LOGICA, V_ORDEM_PRIORIDADE
    );
  ELSIF V_FUNCAO = 2 THEN
    UPDATE EVENTOS_COMUNICACAO SET
      ID_CANAL_COMUNICACAO    = V_ID_CANAL_COMUNICACAO,
      ID_TEMPLATE_COMUNICACAO = V_ID_TEMPLATE_COMUNICACAO,
      NOME_EVENTO             = V_NOME_EVENTO,
      TAG_EVENTO              = V_TAG_EVENTO,
      EXPRESSAO_LOGICA        = V_EXPRESSAO_LOGICA,
      ORDEM_PRIORIDADE        = V_ORDEM_PRIORIDADE
    WHERE ID_EVENTO_COMUNICACAO = V_ID_EVENTO_COMUNICACAO;
  ELSIF V_FUNCAO = 3 THEN
    DELETE FROM EVENTOS_COMUNICACAO WHERE ID_EVENTO_COMUNICACAO = V_ID_EVENTO_COMUNICACAO;
  END IF;
  COMMIT WORK;
END PRC_EVENTOS_COMUNICACAO;

  V_ID_CANAL_COMUNICACAO     IN ENVIOS_COMUNICACAO_LOG.ID_CANAL_COMUNICACAO%TYPE,

  V_NOME_PACIENTE            IN ENVIOS_COMUNICACAO_LOG.NOME_PACIENTE%TYPE,
  V_STATUS                   IN ENVIOS_COMUNICACAO_LOG.STATUS%TYPE,
  V_MENSAGEM_ERRO            IN ENVIOS_COMUNICACAO_LOG.MENSAGEM_ERRO%TYPE
) IS
  V_ID_ENVIO NUMBER;
BEGIN
  SELECT SQ_ENVIOS_COMUNICACAO_LOG.NEXTVAL INTO V_ID_ENVIO FROM DUAL;

  INSERT INTO ENVIOS_COMUNICACAO_LOG (

    V_ORIGEM, V_DESTINO, V_NOME_PACIENTE, SYSDATE, V_STATUS, V_MENSAGEM_ERRO
  );

  COMMIT WORK;

-- FILA_COMUNICACAO
CREATE OR REPLACE PROCEDURE PRC_FILA_COMUNICACAO (
  V_ID_FILA_COMUNICACAO  IN OUT FILA_COMUNICACAO.ID_FILA_COMUNICACAO%TYPE,
  V_TAG_EVENTO           IN     FILA_COMUNICACAO.TAG_EVENTO%TYPE,
  V_ID_REFERENCIA        IN     FILA_COMUNICACAO.ID_REFERENCIA%TYPE,
  V_ORIGEM               IN     FILA_COMUNICACAO.ORIGEM%TYPE,
  V_DESTINO              IN     FILA_COMUNICACAO.DESTINO%TYPE,
  V_STATUS               IN     FILA_COMUNICACAO.STATUS%TYPE            DEFAULT 'PENDENTE',
  V_PRIORIDADE           IN     FILA_COMUNICACAO.PRIORIDADE%TYPE        DEFAULT 0,
  V_TENTATIVAS           IN     FILA_COMUNICACAO.TENTATIVAS%TYPE        DEFAULT 0,
  V_MAX_TENTATIVAS       IN     FILA_COMUNICACAO.MAX_TENTATIVAS%TYPE    DEFAULT 5,
  V_PROXIMA_TENTATIVA    IN     FILA_COMUNICACAO.PROXIMA_TENTATIVA%TYPE DEFAULT NULL,
  V_AGENDAR_PARA         IN     FILA_COMUNICACAO.AGENDAR_PARA%TYPE      DEFAULT NULL,
  V_RESERVADO_POR        IN     FILA_COMUNICACAO.RESERVADO_POR%TYPE     DEFAULT NULL,
  V_DATA_RESERVA         IN     FILA_COMUNICACAO.DATA_RESERVA%TYPE      DEFAULT NULL,
  V_VISIVEL_ATE          IN     FILA_COMUNICACAO.VISIVEL_ATE%TYPE       DEFAULT NULL,
  V_PAYLOAD              IN     FILA_COMUNICACAO.PAYLOAD%TYPE           DEFAULT NULL,
  V_MENSAGEM_ERRO        IN     FILA_COMUNICACAO.MENSAGEM_ERRO%TYPE     DEFAULT NULL,
  V_CHAVE_IDEMPOTENTE    IN     FILA_COMUNICACAO.CHAVE_IDEMPOTENTE%TYPE DEFAULT NULL,
  V_FUNCAO               IN     NUMBER
) IS
BEGIN
  IF V_FUNCAO = 1 THEN
    -- INSERT (enqueue)
    SELECT SQ_FILA_COMUNICACAO.NEXTVAL INTO V_ID_FILA_COMUNICACAO FROM DUAL;
    INSERT INTO FILA_COMUNICACAO (
      ID_FILA_COMUNICACAO, TAG_EVENTO, ID_REFERENCIA, ORIGEM, DESTINO,
      STATUS, PRIORIDADE, TENTATIVAS, MAX_TENTATIVAS, PROXIMA_TENTATIVA, AGENDAR_PARA,
      RESERVADO_POR, DATA_RESERVA, VISIVEL_ATE,
      PAYLOAD, MENSAGEM_ERRO,
      DATA_INSERCAO, CHAVE_IDEMPOTENTE
    ) VALUES (
      V_ID_FILA_COMUNICACAO, V_TAG_EVENTO, V_ID_REFERENCIA, V_ORIGEM, V_DESTINO,
      V_STATUS, V_PRIORIDADE, V_TENTATIVAS, V_MAX_TENTATIVAS, V_PROXIMA_TENTATIVA, V_AGENDAR_PARA,
      V_RESERVADO_POR, V_DATA_RESERVA, V_VISIVEL_ATE,
      V_PAYLOAD, V_MENSAGEM_ERRO,
      SYSDATE, V_CHAVE_IDEMPOTENTE
    );

  ELSIF V_FUNCAO = 2 THEN
    -- UPDATE (status/tentativas/erro/datas/reserva)
    UPDATE FILA_COMUNICACAO SET
      STATUS             = NVL(V_STATUS, STATUS),
      PRIORIDADE         = NVL(V_PRIORIDADE, PRIORIDADE),
      TENTATIVAS         = NVL(V_TENTATIVAS, TENTATIVAS),
      MAX_TENTATIVAS     = NVL(V_MAX_TENTATIVAS, MAX_TENTATIVAS),
      PROXIMA_TENTATIVA  = V_PROXIMA_TENTATIVA,
      AGENDAR_PARA       = V_AGENDAR_PARA,
      RESERVADO_POR      = V_RESERVADO_POR,
      DATA_RESERVA       = V_DATA_RESERVA,
      VISIVEL_ATE        = V_VISIVEL_ATE,
      MENSAGEM_ERRO      = V_MENSAGEM_ERRO,
      DATA_PROCESSAMENTO = CASE WHEN V_STATUS IN ('ENVIADO','ERRO','ABORTADO') THEN SYSDATE ELSE DATA_PROCESSAMENTO END
    WHERE ID_FILA_COMUNICACAO = V_ID_FILA_COMUNICACAO;

  ELSIF V_FUNCAO = 3 THEN
    -- DELETE (remoção explícita)
    DELETE FROM FILA_COMUNICACAO WHERE ID_FILA_COMUNICACAO = V_ID_FILA_COMUNICACAO;

  END IF;

  COMMIT WORK;
END PRC_FILA_COMUNICACAO;
