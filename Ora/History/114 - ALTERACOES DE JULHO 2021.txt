create or replace
PROCEDURE PRC_CLIENTES_B2B(V_REGISTRO  IN CLIENTES_B2B.REGISTRO%TYPE,
                                              V_EQUIPAMENTO     IN CLIENTES_B2B.EQUIPAMENTO%TYPE,
                                              V_CODIGO_EXTERNO     IN CLIENTES_B2B.CODIGO_EXTERNO%TYPE,
                                              V_NOME_EXTERNO       IN CLIENTES_B2B.NOME_EXTERNO%TYPE,
                                              V_DATA_NAS_EXTERNO   IN CLIENTES_B2B.DATA_NAS_EXTERNO%TYPE,
                                              V_SEXO_EXTERNO       IN CLIENTES_B2B.SEXO_EXTERNO%TYPE,
                                              V_IDENTIDADE_EXTERNO IN CLIENTES_B2B.IDENTIDADE_EXTERNO%TYPE,
                                              V_CPF_EXTERNO        IN CLIENTES_B2B.CPF_EXTERNO%TYPE,
                                              V_FUNCAO             IN NUMBER) IS
 BEGIN
   DECLARE
          V_ACHEI  NUMBER(8);
          CURSOR CUR_PROCURA IS
                    SELECT REGISTRO FROM CLIENTES_B2B
                     WHERE REGISTRO = V_REGISTRO AND EQUIPAMENTO = V_EQUIPAMENTO;
   BEGIN                  
        IF V_FUNCAO = 1 THEN
           INSERT INTO CLIENTES_B2B VALUES (V_REGISTRO, V_EQUIPAMENTO, V_CODIGO_EXTERNO, V_NOME_EXTERNO, V_DATA_NAS_EXTERNO, V_SEXO_EXTERNO, V_IDENTIDADE_EXTERNO, V_CPF_EXTERNO);
        ELSIF V_FUNCAO = 2 THEN
            UPDATE CLIENTES_B2B SET NOME_EXTERNO = V_NOME_EXTERNO, DATA_NAS_EXTERNO = V_DATA_NAS_EXTERNO, SEXO_EXTERNO = V_SEXO_EXTERNO, IDENTIDADE_EXTERNO = V_IDENTIDADE_EXTERNO, CPF_EXTERNO = V_CPF_EXTERNO
                              WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
        ELSIF V_FUNCAO = 3 THEN
            DELETE CLIENTES_B2B WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
        ELSIF V_FUNCAO = 4 THEN
            OPEN CUR_PROCURA;
            FETCH CUR_PROCURA INTO V_ACHEI;
            IF CUR_PROCURA%NOTFOUND THEN
               INSERT INTO CLIENTES_B2B VALUES (V_REGISTRO, V_EQUIPAMENTO, V_CODIGO_EXTERNO, V_NOME_EXTERNO, V_DATA_NAS_EXTERNO, V_SEXO_EXTERNO, V_IDENTIDADE_EXTERNO, V_CPF_EXTERNO);
 				   ELSE   
               UPDATE CLIENTES_B2B SET REGISTRO = V_REGISTRO
                                 WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
           END IF;                      
      END IF;
   END;
   COMMIT WORK;
 END PRC_CLIENTES_B2B;
