MINVALUE 0

CREATE OR REPLACE PROCEDURE PRC_INTERFACES_ENVIO(
                                                V_ID_INTERFACE_ENVIO   IN OUT INTERFACES_ENVIO.ID_INTERFACE_ENVIO%TYPE,
                                                V_NOME                 IN     INTERFACES_ENVIO.NOME%TYPE,
                                                V_TIPO_INTERFACE_ENVIO IN     INTERFACES_ENVIO.TIPO_INTERFACE_ENVIO%TYPE,
                                                V_URL                  IN     INTERFACES_ENVIO.URL%TYPE,
                                                V_PORTA                IN     INTERFACES_ENVIO.PORTA%TYPE,
                                                V_USUARIO              IN     INTERFACES_ENVIO.USUARIO%TYPE,
                                                V_SENHA                IN     INTERFACES_ENVIO.SENHA%TYPE,
                                                V_DEVICES              IN     INTERFACES_ENVIO.DEVICES%TYPE,
                                                V_FUNCAO               IN NUMBER
                                            ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_INTERFACES_ENVIO.NEXTVAL INTO V_ID_INTERFACE_ENVIO FROM DUAL;
        INSERT INTO INTERFACES_ENVIO VALUES(V_ID_INTERFACE_ENVIO,  
		                                    V_NOME,                
		                                    V_TIPO_INTERFACE_ENVIO,
		                                    V_URL,                 
		                                    V_PORTA,               
		                                    V_USUARIO,             
		                                    V_SENHA,               
		                                    V_DEVICES);             
		

    ELSIF V_FUNCAO = 2 THEN
        UPDATE INTERFACES_ENVIO SET NOME = V_NOME, 
									TIPO_INTERFACE_ENVIO = V_TIPO_INTERFACE_ENVIO,
									URL = V_URL,
									PORTA = V_PORTA,
									USUARIO = V_USUARIO,
									SENHA = V_SENHA,
									DEVICES = V_DEVICES
                              WHERE ID_INTERFACE_ENVIO = V_ID_INTERFACE_ENVIO;

    ELSIF V_FUNCAO = 3 THEN
        DELETE INTERFACES_ENVIO WHERE ID_INTERFACE_ENVIO = V_ID_INTERFACE_ENVIO;

    END IF;

    COMMIT WORK;
END PRC_INTERFACES_ENVIO;

									   V_CONFIGURACAO              IN     POSTOS.CONFIGURACAO%TYPE,

									   V_ID_INTERFACE_ENVIO_EMAIL  IN     POSTOS.ID_INTERFACE_ENVIO_EMAIL%TYPE,
                                       V_FUNCAO                    IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_POSTO.NEXTVAL INTO V_POSTO FROM DUAL;
     INSERT INTO POSTOS VALUES (V_POSTO, V_NOME, V_INFORMATIZADO, V_HORA_ENT, V_INTERNO, V_DATA_CAD,

                                V_TIPO_POSTO, V_PASS, V_ENDERECO, V_MNEMONICO_POSTO, V_NOME_FICHA, EMPTY_BLOB, V_LAYOUT_PADRAO,
								V_IMPRIME_PARCIAL_DIGITADOS, V_CONFIGURACAO, V_ID_INTERFACE_ENVIO_SMS, V_ID_INTERFACE_ENVIO_EMAIL);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE POSTOS SET NOME = V_NOME,     INFORMATIZADO = V_INFORMATIZADO, 
                             HORA_ENT = V_HORA_ENT, INTERNO = V_INTERNO, DATA_CAD = V_DATA_CAD,
