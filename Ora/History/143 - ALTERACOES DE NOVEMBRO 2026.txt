-- >>> iris-resultado-liberado inicio
-- resultado_disponivel so sai com exame liberado de fato. O script 142 tratava
-- SITUACAO R como liberado, mas quem nao tem direito de liberar tambem grava R.
-- Liberado segue GET_SITUACAO_EXAME: R com interface L, U, D, T, G ou O; I, L, X e Z
-- ja passaram pela liberacao. C e F nao contam. Uma vez por atendimento. Nunca
-- derruba a gravacao do exame. Nao faz COMMIT.
CREATE OR REPLACE PROCEDURE PRC_IRIS_ENFILEIRA_RESULTADO(V_POSTO IN NUMBER, V_ATENDIMENTO IN NUMBER) IS
  V_LIGADO   NUMBER;
  V_PENDENTE NUMBER;
  V_LIBERADO NUMBER;
BEGIN
  SELECT COUNT(*) INTO V_LIGADO FROM INTEGRACOES_CFG
   WHERE INTEGRACAO = 'IRIS' AND VALOR = 'S'
     AND PARAMETRO IN ('ATIVO', 'EVENTO_RESULTADO_DISPONIVEL');
  IF V_LIGADO < 2 THEN RETURN; END IF;

  SELECT COUNT(CASE WHEN SITUACAO NOT IN ('I', 'L', 'X', 'Z', 'C', 'F')
                     AND NOT (SITUACAO = 'R' AND NVL(SITUACAO_INTERFACE, '#') IN ('L', 'U', 'D', 'T', 'G', 'O')) THEN 1 END),
         COUNT(CASE WHEN SITUACAO IN ('I', 'L', 'X', 'Z')
                      OR (SITUACAO = 'R' AND SITUACAO_INTERFACE IN ('L', 'U', 'D', 'T', 'G', 'O')) THEN 1 END)
    INTO V_PENDENTE, V_LIBERADO
    FROM EXAMES WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;
  IF V_PENDENTE > 0 OR V_LIBERADO = 0 THEN RETURN; END IF;

  INSERT INTO IRIS_NOTIFICACOES (ID_NOTIFICACAO, EVENTO, REFERENCIA, POSTO, ATENDIMENTO, SITUACAO, USUARIO)
  SELECT SEQ_IRIS_NOTIFICACAO.NEXTVAL, 'resultado_disponivel', V_POSTO || '-' || V_ATENDIMENTO,
         V_POSTO, V_ATENDIMENTO, 'P', 'TRIGGER'
    FROM DUAL
   WHERE NOT EXISTS (SELECT 1 FROM IRIS_NOTIFICACOES
                      WHERE EVENTO = 'resultado_disponivel' AND POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO);
EXCEPTION
  WHEN OTHERS THEN
    NULL;
END PRC_IRIS_ENFILEIRA_RESULTADO;
/

-- Passa a olhar tambem a interface: exame gravado sem liberar fica R, e quando o
-- medico libera so SITUACAO_INTERFACE muda. A verificacao continua no fim do comando.
CREATE OR REPLACE TRIGGER TRG_EXAMES_IRIS_RESULTADO
FOR UPDATE OF SITUACAO, SITUACAO_INTERFACE ON EXAMES
COMPOUND TRIGGER
  TYPE T_ATENDIMENTOS IS TABLE OF VARCHAR2(30) INDEX BY VARCHAR2(30);
  G_ATENDIMENTOS T_ATENDIMENTOS;

  AFTER EACH ROW IS
  BEGIN
    IF :NEW.SITUACAO = 'R'
       AND (NVL(:OLD.SITUACAO, '#') <> 'R'
            OR NVL(:OLD.SITUACAO_INTERFACE, '#') <> NVL(:NEW.SITUACAO_INTERFACE, '#')) THEN
      G_ATENDIMENTOS(:NEW.POSTO || '-' || :NEW.ATENDIMENTO) := :NEW.POSTO || '-' || :NEW.ATENDIMENTO;
    END IF;
  EXCEPTION
    WHEN OTHERS THEN
      NULL;
  END AFTER EACH ROW;

  AFTER STATEMENT IS
    V_CHAVE VARCHAR2(30);
  BEGIN
    V_CHAVE := G_ATENDIMENTOS.FIRST;
    WHILE V_CHAVE IS NOT NULL LOOP
      PRC_IRIS_ENFILEIRA_RESULTADO(TO_NUMBER(SUBSTR(V_CHAVE, 1, INSTR(V_CHAVE, '-') - 1)),
                                   TO_NUMBER(SUBSTR(V_CHAVE, INSTR(V_CHAVE, '-') + 1)));
      V_CHAVE := G_ATENDIMENTOS.NEXT(V_CHAVE);
    END LOOP;
    G_ATENDIMENTOS.DELETE;
  EXCEPTION
    WHEN OTHERS THEN
      G_ATENDIMENTOS.DELETE;
  END AFTER STATEMENT;
END TRG_EXAMES_IRIS_RESULTADO;
/

--
-- LIMITES_MENSAIS_EXAMES
-- Regras de limite mensal de exames do laboratorio. Cada regra conta os exames
-- ativos (SITUACAO fora de C e F, atendimento nao cancelado) do mes calendario
-- de ATENDIMENTOS.DATA_ATD. POSTO, CONVENIO e MNEMONICO nulos = qualquer um:
--   geral                     POSTO, CONVENIO e MNEMONICO nulos
--   por convenio              CONVENIO
--   por posto                 POSTO
--   procedimento por posto    MNEMONICO e POSTO
--   procedimento por convenio MNEMONICO e CONVENIO
-- Ao atingir LIMITE, a tela de Atendimento so inclui exame (ou grava atendimento
-- novo) com a senha de quem tem a permissao acLiberaLimiteMensalExames. A funcao
-- liga pela chave INTEGRACOES_CFG ATENDIMENTO/LIMITE_MENSAL_EXAMES (Opcoes).
--
CREATE SEQUENCE SEQ_LIMITE_MENSAL_EXAME START WITH 1 INCREMENT BY 1 MAXVALUE 999999999 NOCACHE;
-- fim bloco

CREATE TABLE LIMITES_MENSAIS_EXAMES (
  ID_LIMITE  NUMBER(9)     NOT NULL,
  DESCRICAO  VARCHAR2(60)  NOT NULL,
  POSTO      NUMBER(3),
  CONVENIO   NUMBER(3),
  MNEMONICO  VARCHAR2(7),
  LIMITE     NUMBER(9)     NOT NULL,
  ATIVO      CHAR(1)       DEFAULT 'S' NOT NULL,
  USUARIO    VARCHAR2(30),
  DATA_CAD   DATE          DEFAULT SYSDATE NOT NULL,
  CONSTRAINT LIMITES_MENSAIS_EXAMES_PK PRIMARY KEY (ID_LIMITE),
  CONSTRAINT LIMITES_MENSAIS_EXAMES_LIM CHECK (LIMITE > 0),
  CONSTRAINT LIMITES_MENSAIS_EXAMES_ATV CHECK (ATIVO IN ('S', 'N'))
);
-- fim bloco

-- Uma regra so por combinacao de posto, convenio e procedimento.
CREATE UNIQUE INDEX LIMITES_MENSAIS_EXAMES_UK ON LIMITES_MENSAIS_EXAMES
  (NVL(POSTO, -1), NVL(CONVENIO, -1), NVL(MNEMONICO, '#'));
-- fim bloco

--
-- LIMITES_MENSAIS_LIBERACOES
-- Historico de cada liberacao por senha: qual regra estava no limite, em qual
-- atendimento, a contagem e o limite no momento, quem liberou (dono da senha) e
-- quem operava a tela. ATENDIMENTO fica nulo quando a liberacao foi dada antes de
-- gravar um atendimento novo e o numero ainda nao existia.
--
CREATE SEQUENCE SEQ_LIMITE_MENSAL_LIBERACAO START WITH 1 INCREMENT BY 1 MAXVALUE 999999999 NOCACHE;
-- fim bloco

CREATE TABLE LIMITES_MENSAIS_LIBERACOES (
  ID_LIBERACAO      NUMBER(9)     NOT NULL,
  DATA_HORA         DATE          DEFAULT SYSDATE NOT NULL,
  ID_LIMITE         NUMBER(9)     NOT NULL,
  POSTO             NUMBER(3),
  ATENDIMENTO       NUMBER(9),
  MNEMONICO         VARCHAR2(7),
  CONTAGEM          NUMBER(9)     NOT NULL,
  LIMITE            NUMBER(9)     NOT NULL,
  USUARIO_LIBEROU   VARCHAR2(30)  NOT NULL,
  USUARIO_OPERADOR  VARCHAR2(30),
  CONSTRAINT LIMITES_MENSAIS_LIBERACOES_PK PRIMARY KEY (ID_LIBERACAO)
);
-- fim bloco

CREATE INDEX LIMITES_MENSAIS_LIBERACOES_I1 ON LIMITES_MENSAIS_LIBERACOES (POSTO, ATENDIMENTO);
-- fim bloco

-- CRUD da tela Limites Mensais de Exames. A tela usa -1 para "Todos" (o cadastro
-- padrao troca numero nulo por 0, e o posto 0 existe), e a procedure grava NULL.
create or replace
PROCEDURE PRC_LIMITES_MENSAIS_EXAMES(V_ID_LIMITE  IN OUT LIMITES_MENSAIS_EXAMES.ID_LIMITE%TYPE,
                                     V_DESCRICAO  IN LIMITES_MENSAIS_EXAMES.DESCRICAO%TYPE,
                                     V_POSTO      IN NUMBER,
                                     V_CONVENIO   IN NUMBER,
                                     V_MNEMONICO  IN LIMITES_MENSAIS_EXAMES.MNEMONICO%TYPE,
                                     V_LIMITE     IN LIMITES_MENSAIS_EXAMES.LIMITE%TYPE,
                                     V_ATIVO      IN LIMITES_MENSAIS_EXAMES.ATIVO%TYPE,
                                     V_USUARIO    IN LIMITES_MENSAIS_EXAMES.USUARIO%TYPE,
                                     V_FUNCAO     IN NUMBER) IS
  W_POSTO     NUMBER;
  W_CONVENIO  NUMBER;
  W_MNEMONICO VARCHAR2(7);
BEGIN
  W_POSTO := CASE WHEN V_POSTO IS NULL OR V_POSTO = -1 THEN NULL ELSE V_POSTO END;
  W_CONVENIO := CASE WHEN V_CONVENIO IS NULL OR V_CONVENIO = -1 THEN NULL ELSE V_CONVENIO END;
  W_MNEMONICO := UPPER(TRIM(V_MNEMONICO));
  IF V_FUNCAO = 1 THEN
     SELECT SEQ_LIMITE_MENSAL_EXAME.NEXTVAL INTO V_ID_LIMITE FROM DUAL;

     INSERT INTO LIMITES_MENSAIS_EXAMES (ID_LIMITE, DESCRICAO, POSTO, CONVENIO, MNEMONICO, LIMITE, ATIVO, USUARIO, DATA_CAD)
       VALUES (V_ID_LIMITE, V_DESCRICAO, W_POSTO, W_CONVENIO, W_MNEMONICO, V_LIMITE, NVL(V_ATIVO, 'S'), V_USUARIO, SYSDATE);
  ELSIF V_FUNCAO = 2 THEN
     UPDATE LIMITES_MENSAIS_EXAMES
        SET DESCRICAO = V_DESCRICAO, POSTO = W_POSTO, CONVENIO = W_CONVENIO, MNEMONICO = W_MNEMONICO,
            LIMITE = V_LIMITE, ATIVO = NVL(V_ATIVO, 'S'), USUARIO = V_USUARIO
      WHERE ID_LIMITE = V_ID_LIMITE;
  ELSIF V_FUNCAO = 3 THEN
     DELETE LIMITES_MENSAIS_EXAMES WHERE ID_LIMITE = V_ID_LIMITE;
  END IF;
  COMMIT WORK;
END PRC_LIMITES_MENSAIS_EXAMES;
/
-- fim bloco
