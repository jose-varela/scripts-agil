CREATE TABLE MAP_EQUIPAMENTOS_SETORES(
    EQUIPAMENTO NUMBER(3) NOT NULL,
    SETOR NUMBER(6) NOT NULL,
    MAP_SETOR VARCHAR2(20) NOT NULL
)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE MAP_EQUIPAMENTOS_SETORES ADD CONSTRAINT MAP_EQUIPAMENTOS_SETORES_PK
    PRIMARY KEY (EQUIPAMENTO, SETOR)
    USING INDEX
    TABLESPACE LIS_INDX PCTFREE 20
    STORAGE (INITIAL 10K NEXT 10K PCTINCREASE 0);
-- fim bloco

ALTER TABLE MAP_EQUIPAMENTOS_SETORES ADD CONSTRAINT MAP_EQUIP_SET_SETORES_FK
    FOREIGN KEY (SETOR) REFERENCES SETORES(SETOR);
-- fim bloco

ALTER TABLE MAP_EQUIPAMENTOS_SETORES ADD CONSTRAINT MAP_EQUIP_SET_EQUIPAMENTOS_FK
    FOREIGN KEY (EQUIPAMENTO) REFERENCES EQUIPAMENTOS(EQUIPAMENTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_MAP_EQUIPAMENTOS_SETORES(V_EQUIPAMENTO IN MAP_EQUIPAMENTOS_SETORES.EQUIPAMENTO%TYPE,
                                                         V_SETOR   IN MAP_EQUIPAMENTOS_SETORES.SETOR%TYPE,
                                                         V_MAP_SETOR  IN MAP_EQUIPAMENTOS_SETORES.MAP_SETOR%TYPE,
                                                         V_FUNCAO      IN NUMBER) IS      
BEGIN
  IF V_FUNCAO = 1 THEN
    INSERT INTO MAP_EQUIPAMENTOS_SETORES VALUES (V_EQUIPAMENTO, V_SETOR, V_MAP_SETOR);           
  
  ELSIF V_FUNCAO = 2 THEN
    UPDATE MAP_EQUIPAMENTOS_SETORES SET MAP_SETOR = V_MAP_SETOR WHERE EQUIPAMENTO = V_EQUIPAMENTO AND SETOR = V_SETOR;

  ELSIF V_FUNCAO = 3 THEN
    DELETE MAP_EQUIPAMENTOS_SETORES WHERE EQUIPAMENTO = V_EQUIPAMENTO AND SETOR = V_SETOR;

  END IF;

  COMMIT WORK;

END PRC_MAP_EQUIPAMENTOS_SETORES;
