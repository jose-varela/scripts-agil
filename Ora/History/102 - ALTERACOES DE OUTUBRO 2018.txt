CREATE SEQUENCE SQ_PROCEDIMENTOS_URGENTES
MINVALUE 0 
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

CREATE TABLE PROCEDIMENTOS_URGENTES(
    ID_PROCEDIMENTO_URGENTE NUMBER(9) NOT NULL,
    MNEMONICO VARCHAR2(7) NOT NULL,
    POSTO_REA NUMBER(3) NOT NULL
)
TABLESPACE LIS_DATA;

-- fim bloco

ALTER TABLE PROCEDIMENTOS_URGENTES ADD CONSTRAINT PROCEDIMENTOS_URGENTES_PK 
      PRIMARY KEY (ID_PROCEDIMENTO_URGENTE);
-- fim bloco

ALTER TABLE PROCEDIMENTOS_URGENTES ADD CONSTRAINT PROCED_URGENTES_MNEMONICO_FK
    FOREIGN KEY (MNEMONICO) REFERENCES PROCEDIMENTOS(MNEMONICO);
-- fim bloco

ALTER TABLE PROCEDIMENTOS_URGENTES ADD CONSTRAINT PROCED_URGENTES_POSTO_REA_FK
    FOREIGN KEY (POSTO_REA) REFERENCES POSTOS(POSTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_PROCEDIMENTOS_URGENTES(
                                                V_ID_PROCEDIMENTO_URGENTE IN OUT PROCEDIMENTOS_URGENTES.ID_PROCEDIMENTO_URGENTE%TYPE,
                                                V_MNEMONICO               IN PROCEDIMENTOS_URGENTES.MNEMONICO%TYPE,
                                                V_POSTO_REA               IN PROCEDIMENTOS_URGENTES.POSTO_REA%TYPE,
                                                V_FUNCAO                  IN NUMBER
                                            ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_PROCEDIMENTOS_URGENTES.NEXTVAL INTO V_ID_PROCEDIMENTO_URGENTE FROM DUAL;
        INSERT INTO PROCEDIMENTOS_URGENTES VALUES(V_ID_PROCEDIMENTO_URGENTE, V_MNEMONICO, V_POSTO_REA);

    ELSIF V_FUNCAO = 2 THEN
        UPDATE PROCEDIMENTOS_URGENTES SET MNEMONICO = V_MNEMONICO, POSTO_REA = V_POSTO_REA
                                    WHERE ID_PROCEDIMENTO_URGENTE = V_ID_PROCEDIMENTO_URGENTE;

    ELSIF V_FUNCAO = 3 THEN
        DELETE PROCEDIMENTOS_URGENTES WHERE ID_PROCEDIMENTO_URGENTE = V_ID_PROCEDIMENTO_URGENTE;

    END IF;

    COMMIT WORK;
END PRC_PROCEDIMENTOS_URGENTES;

CREATE SEQUENCE SQ_SEQUENCIAS_SETORES
MINVALUE 0 
MAXVALUE 9999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

CREATE TABLE SEQUENCIAS_SETORES(
    ID_SEQUENCIA_SETOR NUMBER(4) NOT NULL,
    SETOR NUMBER(2) NOT NULL,
    POSTO_REA NUMBER(3) NOT NULL,
    DATA_INICIO_SEQUENCIA DATE NOT NULL,
    INICIO_SEQUENCIA NUMBER(7) DEFAULT 1 NOT NULL,
    QTD_DIAS_VALIDADE NUMBER(3) NOT NULL,
	SEQUENCIA_SETOR NUMBER(7) NOT NULL,
    LAYOUT_ETIQUETA VARCHAR2(250)
)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_PK 
      PRIMARY KEY (ID_SEQUENCIA_SETOR);
-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_SETOR_FK
    FOREIGN KEY (SETOR) REFERENCES SETORES(SETOR);
-- fim bloco

ALTER TABLE SEQUENCIAS_SETORES ADD CONSTRAINT SEQUENCIAS_SETORES_POSTO_FK
    FOREIGN KEY (POSTO_REA) REFERENCES POSTOS(POSTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_SEQUENCIAS_SETORES(
                                                V_ID_SEQUENCIA_SETOR    IN OUT SEQUENCIAS_SETORES.ID_SEQUENCIA_SETOR%TYPE,
                                                V_SETOR                 IN SEQUENCIAS_SETORES.SETOR%TYPE,
                                                V_POSTO_REA             IN SEQUENCIAS_SETORES.POSTO_REA%TYPE,
                                                V_DATA_INICIO_SEQUENCIA IN SEQUENCIAS_SETORES.DATA_INICIO_SEQUENCIA%TYPE,
                                                V_INICIO_SEQUENCIA      IN SEQUENCIAS_SETORES.INICIO_SEQUENCIA%TYPE,
                                                V_QTD_DIAS_VALIDADE     IN SEQUENCIAS_SETORES.QTD_DIAS_VALIDADE%TYPE,
                                                V_SEQUENCIA_SETOR       IN SEQUENCIAS_SETORES.SEQUENCIA_SETOR%TYPE,
                                                V_LAYOUT_ETIQUETA       IN SEQUENCIAS_SETORES.LAYOUT_ETIQUETA%TYPE,
                                                V_FUNCAO                IN NUMBER
                                            ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_SEQUENCIAS_SETORES.NEXTVAL INTO V_ID_SEQUENCIA_SETOR FROM DUAL;
        INSERT INTO SEQUENCIAS_SETORES VALUES(
                                            V_ID_SEQUENCIA_SETOR, V_SETOR, V_POSTO_REA, V_DATA_INICIO_SEQUENCIA, V_INICIO_SEQUENCIA,
                                            V_QTD_DIAS_VALIDADE, V_INICIO_SEQUENCIA, V_LAYOUT_ETIQUETA);

    ELSIF V_FUNCAO = 2 THEN
        UPDATE SEQUENCIAS_SETORES SET
                                    SETOR = V_SETOR, POSTO_REA = V_POSTO_REA, DATA_INICIO_SEQUENCIA = V_DATA_INICIO_SEQUENCIA, INICIO_SEQUENCIA = V_INICIO_SEQUENCIA,
                                    QTD_DIAS_VALIDADE = V_QTD_DIAS_VALIDADE, SEQUENCIA_SETOR = V_SEQUENCIA_SETOR, LAYOUT_ETIQUETA = V_LAYOUT_ETIQUETA
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;

    ELSIF V_FUNCAO = 3 THEN
        DELETE SEQUENCIAS_SETORES WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;

    ELSIF V_FUNCAO = 4 THEN
        UPDATE SEQUENCIAS_SETORES SET 
                                     SEQUENCIA_SETOR = SEQUENCIA_SETOR + 1
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;
    ELSIF V_FUNCAO = 5 THEN
        UPDATE SEQUENCIAS_SETORES SET
                                    DATA_INICIO_SEQUENCIA = SYSDATE,
                                    SEQUENCIA_SETOR = INICIO_SEQUENCIA
                                WHERE ID_SEQUENCIA_SETOR = V_ID_SEQUENCIA_SETOR;
    END IF;

    COMMIT WORK;
END PRC_SEQUENCIAS_SETORES;
