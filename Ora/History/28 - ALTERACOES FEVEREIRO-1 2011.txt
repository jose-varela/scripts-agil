-- conservantes

CREATE TABLE CONSERVANTES
(ID_CONSERVANTE    NUMBER(2)    NOT NULL,
 NOME              VARCHAR2(50) NOT NULL,
 STATUS            VARCHAR2(1)  NOT NULL)
TABLESPACE LIS_DATA;
-- fim bloco


ALTER TABLE CONSERVANTES ADD CONSTRAINT CONSERVANTES_PK 
      PRIMARY KEY (ID_CONSERVANTE);
-- fim bloco

CREATE SEQUENCE SQ_CONSERVANTE
MINVALUE 0 
MAXVALUE 99
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_CONSERVANTES(V_ID_CONSERVANTE    IN OUT CONSERVANTES.ID_CONSERVANTE%TYPE,
                                              V_NOME               IN CONSERVANTES.NOME%TYPE,
                                              V_STATUS             IN CONSERVANTES.STATUS%TYPE,
                                              V_FUNCAO             IN NUMBER) IS
BEGIN                                                           
  IF V_FUNCAO = 1 THEN                                          
     SELECT SQ_CONSERVANTE.NEXTVAL INTO V_ID_CONSERVANTE FROM DUAL;
     INSERT INTO CONSERVANTES VALUES(V_ID_CONSERVANTE, V_NOME, V_STATUS);

  ELSIF V_FUNCAO = 2 THEN
            UPDATE CONSERVANTES SET NOME       = V_NOME,
                                     STATUS     = V_STATUS
                          WHERE ID_CONSERVANTE = V_ID_CONSERVANTE;

  ELSIF V_FUNCAO = 3 THEN
            DELETE CONSERVANTES WHERE ID_CONSERVANTE = V_ID_CONSERVANTE;	

  END IF;

  COMMIT WORK;
END PRC_CONSERVANTES;
-- fim bloco




-- conservantes_procedimentos

CREATE TABLE CONSERVANTES_PROCEDIMENTOS
(ID_CONSERVANTE    NUMBER(3)    NOT NULL,
 MNEMONICO         VARCHAR2(4)  NOT NULL)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE CONSERVANTES_PROCEDIMENTOS ADD CONSTRAINT CONSERVANTES_PROCEDIMENTOS_PK 
      PRIMARY KEY (ID_CONSERVANTE, MNEMONICO);
-- fim bloco


CREATE OR REPLACE PROCEDURE PRC_CONSERVANTES_PROCED(V_ID_CONSERVANTE IN CONSERVANTES_PROCEDIMENTOS.ID_CONSERVANTE%TYPE,
                                                     V_MNEMONICO       IN CONSERVANTES_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                     V_FUNCAO          IN NUMBER) IS
BEGIN                                                           
  IF V_FUNCAO = 1 THEN                                          
     INSERT INTO CONSERVANTES_PROCEDIMENTOS VALUES(V_ID_CONSERVANTE, V_MNEMONICO);


  ELSIF V_FUNCAO = 3 THEN
            DELETE CONSERVANTES_PROCEDIMENTOS WHERE ID_CONSERVANTE = V_ID_CONSERVANTE AND MNEMONICO = V_MNEMONICO;	

  END IF;

  COMMIT WORK;
END PRC_CONSERVANTES_PROCED;
-- fim bloco

ALTER TABLE CONSERVANTES_PROCEDIMENTOS ADD CONSTRAINT CONSERV_PROC_CONSERVANTES_FK 
      FOREIGN KEY (ID_CONSERVANTE) REFERENCES CONSERVANTES(ID_CONSERVANTE);
-- fim bloco
ALTER TABLE CONSERVANTES_PROCEDIMENTOS ADD CONSTRAINT CONSERV_PROC_PROCEDIMENTOS_FK 
      FOREIGN KEY (MNEMONICO) REFERENCES PROCEDIMENTOS(MNEMONICO);
