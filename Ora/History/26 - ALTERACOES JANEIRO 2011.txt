CREATE OR REPLACE FUNCTION SEQ_INTERNO_FIM(V_POSTO       IN NUMBER) RETURN NUMBER IS
       RESULT NUMBER;
       V_UNIDADE NUMBER;
BEGIN
     SELECT UNIDADE INTO V_UNIDADE FROM UNIDADES_POSTOS WHERE POSTO = V_POSTO;

     SELECT SEQ_INTERNO_FIM INTO RESULT FROM UNIDADES WHERE UNIDADE = V_UNIDADE;
    RETURN(RESULT);
END SEQ_INTERNO_FIM;

create or replace FUNCTION "GET_LOTE" (V_POSTO IN NUMBER, V_EQUIPAMENTO IN NUMBER) RETURN NUMBER IS 
PRAGMA AUTONOMOUS_TRANSACTION;
RESULT NUMBER;
BEGIN
     SELECT LOTE INTO RESULT FROM POSTOS_EQUIPAMENTOS
      WHERE POSTO = V_POSTO AND EQUIPAMENTO = V_EQUIPAMENTO FOR UPDATE;

     UPDATE POSTOS_EQUIPAMENTOS SET LOTE = LOTE + 1
      WHERE POSTO = V_POSTO AND EQUIPAMENTO = V_EQUIPAMENTO;

    COMMIT;

    RETURN(RESULT);
END GET_LOTE;
