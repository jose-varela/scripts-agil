--
-- CORRECAO DA TRIGGER EXAMES_SITUACOES_TRG
--
create or replace TRIGGER EXAMES_SITUACOES_TRG
  AFTER UPDATE ON EXAMES FOR EACH ROW
   WHEN (NEW.SITUACAO <> OLD.SITUACAO)
 DECLARE
  V_AMOSTRA         EXAMES_SITUACOES.AMOSTRA%TYPE;
  V_CORREL_SITUACAO EXAMES_SITUACOES.CORREL_SITUACAO%TYPE;
 BEGIN
      SELECT GET_AMOSTRA(:NEW.POSTO, :NEW.ATENDIMENTO, :NEW.CORREL) INTO V_AMOSTRA FROM DUAL;
      SELECT GET_ULTIMO_CORREL_SITUACAO(:NEW.POSTO, :NEW.ATENDIMENTO, :NEW.CORREL) INTO V_CORREL_SITUACAO FROM DUAL;
      INSERT INTO EXAMES_SITUACOES VALUES (:NEW.POSTO, :NEW.ATENDIMENTO, :NEW.CORREL, V_AMOSTRA, V_CORREL_SITUACAO, :OLD.SITUACAO, SYSDATE, 'AUTO', 'AUTO');
 END;

--
-- CORRECAO DA FUNCTION IS_URGENTE
--
 create or replace FUNCTION IS_URGENTE(V_POSTO       IN NUMBER,
                                       V_ATENDIMENTO IN NUMBER,
                                       V_CORREL      IN NUMBER) RETURN CHAR IS QT_URGENTE NUMBER; 
 BEGIN
     SELECT COUNT(1) INTO QT_URGENTE FROM EXAMES
      WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND SITUACAO = 'U';
     IF QT_URGENTE > 0 THEN
        RETURN 'S';
     ELSE
		SELECT COUNT(1) INTO QT_URGENTE FROM EXAMES_SITUACOES
		WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND SITUACAO = 'U';
		IF QT_URGENTE = 0 THEN
			RETURN 'N';
		ELSE
			RETURN 'S';
		END IF;
     END IF;

 END IS_URGENTE;
-- fim bloco

CREATE TABLE MAP_EQUIPAMENTOS_MEDICOS(
    EQUIPAMENTO NUMBER(3) NOT NULL,
    ID_MEDICO NUMBER(6) NOT NULL,
    MAP_MEDICO VARCHAR2(20) NOT NULL
)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE MAP_EQUIPAMENTOS_MEDICOS ADD CONSTRAINT MAP_EQUIPAMENTOS_MEDICOS_PK
    PRIMARY KEY (EQUIPAMENTO, ID_MEDICO)
    USING INDEX
    TABLESPACE LIS_INDX PCTFREE 20
    STORAGE (INITIAL 10K NEXT 10K PCTINCREASE 0);
-- fim bloco

ALTER TABLE MAP_EQUIPAMENTOS_MEDICOS ADD CONSTRAINT MAP_EQUIP_MED_MEDICOS_FK
    FOREIGN KEY (ID_MEDICO) REFERENCES MEDICOS(ID_MEDICO);
-- fim bloco

ALTER TABLE MAP_EQUIPAMENTOS_MEDICOS ADD CONSTRAINT MAP_EQUIP_MED_EQUIPAMENTOS_FK
    FOREIGN KEY (EQUIPAMENTO) REFERENCES EQUIPAMENTOS(EQUIPAMENTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_MAP_EQUIPAMENTOS_MEDICOS(V_EQUIPAMENTO IN MAP_EQUIPAMENTOS_MEDICOS.EQUIPAMENTO%TYPE,
                                                         V_ID_MEDICO   IN MAP_EQUIPAMENTOS_MEDICOS.ID_MEDICO%TYPE,
                                                         V_MAP_MEDICO  IN MAP_EQUIPAMENTOS_MEDICOS.MAP_MEDICO%TYPE,
                                                         V_FUNCAO      IN NUMBER) IS      
BEGIN
  IF V_FUNCAO = 1 THEN
    INSERT INTO MAP_EQUIPAMENTOS_MEDICOS VALUES (V_EQUIPAMENTO, V_ID_MEDICO, V_MAP_MEDICO);           
  
  ELSIF V_FUNCAO = 2 THEN
    UPDATE MAP_EQUIPAMENTOS_MEDICOS SET MAP_MEDICO = V_MAP_MEDICO WHERE EQUIPAMENTO = V_EQUIPAMENTO AND ID_MEDICO = V_ID_MEDICO;

  ELSIF V_FUNCAO = 3 THEN
    DELETE MAP_EQUIPAMENTOS_MEDICOS WHERE EQUIPAMENTO = V_EQUIPAMENTO AND ID_MEDICO = V_ID_MEDICO;

  END IF;

  COMMIT WORK;

END PRC_MAP_EQUIPAMENTOS_MEDICOS;

--
-- EXAMES_B2B
--
create or replace
PROCEDURE PRC_EXAMES_B2B(V_ID_EXAME_B2B           IN OUT EXAMES_B2B.ID_EXAME_B2B%TYPE,
                                            V_POSTO                  IN EXAMES_B2B.POSTO%TYPE,
                                            V_ATENDIMENTO            IN EXAMES_B2B.ATENDIMENTO%TYPE,
                                            V_CORREL                 IN EXAMES_B2B.CORREL%TYPE,
                                            V_AMOSTRA                IN EXAMES_B2B.AMOSTRA%TYPE,
                                            V_POSTO_B2B              IN EXAMES_B2B.POSTO_B2B%TYPE,
                                            V_EQUIPAMENTO            IN EXAMES_B2B.EQUIPAMENTO%TYPE,
                                            V_CORREL_B2B             IN EXAMES_B2B.CORREL_B2B%TYPE,
                                            V_SITUACAO_B2B           IN EXAMES_B2B.SITUACAO_B2B%TYPE,
                                            V_DATA_B2B               IN EXAMES_B2B.DATA_B2B%TYPE,
                                            V_USUARIO_B2B            IN EXAMES_B2B.USUARIO_B2B%TYPE,
                                            V_ESTACAO_B2B            IN EXAMES_B2B.ESTACAO_B2B%TYPE,
                                            V_LOTE_B2B               IN EXAMES_B2B.LOTE_B2B%TYPE,
 				                    	    V_CODIGO_EXTERNO         IN EXAMES_B2B.CODIGO_EXTERNO%TYPE,
 					                        V_ETIQUETA_EXTERNO       IN EXAMES_B2B.ETIQUETA_EXTERNO%TYPE,
                                            V_FUNCAO                 IN NUMBER) IS
 BEGIN                                                           
   IF V_FUNCAO = 1 THEN                                          
      SELECT SQ_EXAME_B2B.NEXTVAL INTO V_ID_EXAME_B2B FROM DUAL;
      INSERT INTO EXAMES_B2B VALUES(V_ID_EXAME_B2B, V_POSTO, V_ATENDIMENTO, V_CORREL, V_AMOSTRA, V_POSTO_B2B, 
                                    V_EQUIPAMENTO , V_CORREL_B2B, V_SITUACAO_B2B, V_DATA_B2B, 
                                    V_USUARIO_B2B, V_ESTACAO_B2B, V_LOTE_B2B, V_CODIGO_EXTERNO,V_ETIQUETA_EXTERNO);
   ELSIF V_FUNCAO = 2 THEN
             UPDATE EXAMES_B2B SET POSTO                 = V_POSTO,
                                   ATENDIMENTO           = V_ATENDIMENTO,
                                   CORREL                = V_CORREL,
                                   AMOSTRA               = V_AMOSTRA,
                                   POSTO_B2B             = V_POSTO_B2B,
                                   EQUIPAMENTO           = V_EQUIPAMENTO ,
                                   CORREL_B2B            = V_CORREL_B2B,
                                   SITUACAO_B2B          = V_SITUACAO_B2B,
                                   DATA_B2B              = V_DATA_B2B,
                                   USUARIO_B2B           = V_USUARIO_B2B,
                                   ESTACAO_B2B           = V_ESTACAO_B2B,
 								   CODIGO_EXTERNO        = V_CODIGO_EXTERNO,
                                   ETIQUETA_EXTERNO      = V_ETIQUETA_EXTERNO
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 3 THEN
             DELETE EXAMES_B2B WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 4 THEN
             DELETE EXAMES_B2B WHERE POSTO = V_POSTO;
   ELSIF V_FUNCAO = 5 THEN
             UPDATE EXAMES_B2B SET SITUACAO_B2B          = V_SITUACAO_B2B,
                                   DATA_B2B              = V_DATA_B2B,
                                   USUARIO_B2B           = V_USUARIO_B2B,
                                   ESTACAO_B2B           = V_ESTACAO_B2B
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 6 THEN
 			UPDATE EXAMES_B2B SET SITUACAO_B2B = V_SITUACAO_B2B
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 7 THEN
             UPDATE EXAMES_B2B SET LOTE_B2B     = V_LOTE_B2B
                             WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   END IF;
   COMMIT WORK;
 END PRC_EXAMES_B2B;
