CREATE SEQUENCE SQ_ORIENTACAO
MINVALUE 0
MAXVALUE 999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;

ALTER TABLE ORIENTACOES MODIFY(ORIENTACAO VARCHAR2(50));


CREATE OR REPLACE PROCEDURE PRC_ORIENTACOES(V_ID_ORIENTACAO         IN OUT ORIENTACOES.ID_ORIENTACAO%TYPE,
                                            V_ORIENTACAO            IN     ORIENTACOES.ORIENTACAO%TYPE,
                                            V_STATUS_ORIENTACAO     IN     ORIENTACOES.STATUS_ORIENTACAO%TYPE,
                                            V_PRIORIDADE_ORIENTACAO IN     ORIENTACOES.PRIORIDADE_ORIENTACAO%TYPE,
                                            V_GRUPO_ORIENTACAO      IN     ORIENTACOES.GRUPO_ORIENTACAO%TYPE,
                                            V_FUNCAO                IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_ORIENTACAO.NEXTVAL INTO V_ID_ORIENTACAO FROM DUAL;
     INSERT INTO ORIENTACOES VALUES (V_ID_ORIENTACAO, V_ORIENTACAO, V_STATUS_ORIENTACAO, V_PRIORIDADE_ORIENTACAO, V_GRUPO_ORIENTACAO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE ORIENTACOES SET ORIENTACAO = V_ORIENTACAO, STATUS_ORIENTACAO = V_STATUS_ORIENTACAO, 
                                  PRIORIDADE_ORIENTACAO = V_PRIORIDADE_ORIENTACAO, GRUPO_ORIENTACAO = V_GRUPO_ORIENTACAO
                       WHERE ID_ORIENTACAO = V_ID_ORIENTACAO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE ORIENTACOES WHERE ID_ORIENTACAO = V_ID_ORIENTACAO;
  END IF;

  COMMIT WORK;

END PRC_ORIENTACOES;
/


CREATE OR REPLACE PROCEDURE PRC_ORIENTACOES_PROCEDIMENTOS(V_MNEMONICO         IN ORIENTACOES_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                          V_ID_ORIENTACAO     IN ORIENTACOES_PROCEDIMENTOS.ID_ORIENTACAO%TYPE,
                                                          V_FUNCAO            IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO ORIENTACOES_PROCEDIMENTOS VALUES (V_MNEMONICO, V_ID_ORIENTACAO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE ORIENTACOES_PROCEDIMENTOS WHERE ID_ORIENTACAO = V_ID_ORIENTACAO AND MNEMONICO = V_MNEMONICO;
  END IF;

  COMMIT WORK;

END PRC_ORIENTACOES_PROCEDIMENTOS;
/

CREATE OR REPLACE PROCEDURE PRC_POSTOS(V_POSTO            IN OUT POSTOS.POSTO%TYPE,
                                       V_NOME             IN     POSTOS.NOME%TYPE,
                                       V_INFORMATIZADO    IN     POSTOS.INFORMATIZADO%TYPE,
                                       V_HORA_ENT         IN     POSTOS.HORA_ENT%TYPE,
                                       V_INTERNO          IN     POSTOS.INTERNO%TYPE,
                                       V_DATA_CAD         IN     POSTOS.DATA_CAD%TYPE,
                                       V_REALIZA_EXAMES   IN     POSTOS.REALIZA_EXAMES%TYPE,
                                       V_DIRECIONA_EXAMES IN     POSTOS.DIRECIONA_EXAMES%TYPE,
                                       V_STATUS           IN     POSTOS.STATUS%TYPE,
                                       V_COLETA_MATERIAL  IN     POSTOS.COLETA_MATERIAL%TYPE,
                                       V_TIPO_POSTO       IN     POSTOS.TIPO_POSTO%TYPE,
                                       V_PASS             IN     POSTOS.PASS%TYPE,
                                       V_FUNCAO           IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_POSTO.NEXTVAL INTO V_POSTO FROM DUAL;
     INSERT INTO POSTOS VALUES (V_POSTO, V_NOME, V_INFORMATIZADO, V_HORA_ENT, V_INTERNO, V_DATA_CAD,
                                V_REALIZA_EXAMES, V_DIRECIONA_EXAMES, V_STATUS, V_COLETA_MATERIAL, V_TIPO_POSTO, V_PASS);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE POSTOS SET NOME = V_NOME,     INFORMATIZADO = V_INFORMATIZADO, 
                             HORA_ENT = V_HORA_ENT, INTERNO = V_INTERNO, DATA_CAD = V_DATA_CAD,
                             REALIZA_EXAMES = V_REALIZA_EXAMES, DIRECIONA_EXAMES = V_DIRECIONA_EXAMES,
                             STATUS = V_STATUS, COLETA_MATERIAL = V_COLETA_MATERIAL, TIPO_POSTO = V_TIPO_POSTO, PASS = V_PASS
                        WHERE POSTO = V_POSTO;
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS WHERE POSTO = V_POSTO;

  END IF;

  COMMIT WORK;

END PRC_POSTOS;
/
