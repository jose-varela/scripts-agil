CREATE OR REPLACE PROCEDURE PRC_MAP_B2B_CONVENIOS(V_ID_SERVICO_B2B  IN   MAP_B2B_CONVENIOS.ID_SERVICO_B2B%TYPE,
                                                  V_CONVENIO        IN   MAP_B2B_CONVENIOS.CONVENIO%TYPE,
                                                  V_MAP_CONVENIO    IN   MAP_B2B_CONVENIOS.MAP_CONVENIO%TYPE,
                                                  V_FUNCAO          IN   NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO MAP_B2B_CONVENIOS VALUES (V_ID_SERVICO_B2B, V_CONVENIO, V_MAP_CONVENIO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE MAP_B2B_CONVENIOS SET MAP_CONVENIO = V_MAP_CONVENIO WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND CONVENIO = V_CONVENIO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE MAP_B2B_CONVENIOS WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND CONVENIO = V_CONVENIO;
     
  END IF;             

  COMMIT WORK;

END PRC_MAP_B2B_CONVENIOS;

CREATE OR REPLACE PROCEDURE PRC_ATENDIMENTOS_B2B(V_POSTO              IN ATENDIMENTOS_B2B.POSTO%TYPE,
                                                 V_ATENDIMENTO        IN ATENDIMENTOS_B2B.ATENDIMENTO%TYPE,
                                                 V_ID_SERVICO_B2B     IN ATENDIMENTOS_B2B.ID_SERVICO_B2B%TYPE,
                                                 V_CODIGO_EXTERNO     IN ATENDIMENTOS_B2B.CODIGO_EXTERNO%TYPE,
                                                 V_FUNCAO             IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO ATENDIMENTOS_B2B VALUES (V_POSTO, V_ATENDIMENTO, V_ID_SERVICO_B2B, V_CODIGO_EXTERNO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE ATENDIMENTOS_B2B WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
                            
  END IF;

  COMMIT WORK;
END PRC_ATENDIMENTOS_B2B;
