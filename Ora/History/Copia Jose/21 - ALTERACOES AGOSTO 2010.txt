 
ALTER TABLE POSTOS DROP COLUMN LOGIN;
ALTER TABLE POSTOS DROP COLUMN PASS;

CREATE OR REPLACE PROCEDURE PRC_POSTOS(V_POSTO            IN OUT POSTOS.POSTO%TYPE,
                                       V_NOME             IN     POSTOS.NOME%TYPE,
                                       V_INFORMATIZADO    IN     POSTOS.INFORMATIZADO%TYPE,
                                       V_HORA_ENT         IN     POSTOS.HORA_ENT%TYPE,
                                       V_INTERNO          IN     POSTOS.INTERNO%TYPE,
                                       V_DATA_CAD         IN     POSTOS.DATA_CAD%TYPE,
                                       V_REALIZA_EXAMES   IN     POSTOS.REALIZA_EXAMES%TYPE,
                                       V_DIRECIONA_EXAMES IN     POSTOS.DIRECIONA_EXAMES%TYPE,
                                       V_STATUS           IN     POSTOS.STATUS%TYPE,
                                       V_COLETA_MATERIAL  IN     POSTOS.COLETA_MATERIAL%TYPE,
                                       V_TIPO_POSTO       IN     POSTOS.TIPO_POSTO%TYPE,
                                       V_FUNCAO           IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_POSTO.NEXTVAL INTO V_POSTO FROM DUAL;
     INSERT INTO POSTOS VALUES (V_POSTO, V_NOME, V_INFORMATIZADO, V_HORA_ENT, V_INTERNO, V_DATA_CAD,
                                V_REALIZA_EXAMES, V_DIRECIONA_EXAMES, V_STATUS, V_COLETA_MATERIAL, V_TIPO_POSTO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE POSTOS SET NOME = V_NOME,     INFORMATIZADO = V_INFORMATIZADO, 
                             HORA_ENT = V_HORA_ENT, INTERNO = V_INTERNO, DATA_CAD = V_DATA_CAD,
                             REALIZA_EXAMES = V_REALIZA_EXAMES, DIRECIONA_EXAMES = V_DIRECIONA_EXAMES,
                             STATUS = V_STATUS, COLETA_MATERIAL = V_COLETA_MATERIAL, TIPO_POSTO = V_TIPO_POSTO

                        WHERE POSTO = V_POSTO;
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS WHERE POSTO = V_POSTO;
  END IF;

  COMMIT WORK;

END PRC_POSTOS;
/

CREATE OR REPLACE VIEW EQUIP_TEXTOS_PROCEDIMENTOS AS
SELECT EP.EQUIPAMENTO,
       EP.MNEMONICO,
       EP.MNEMONICO_EQUIPAMENTO,
       EP.STATUS,
       NVL(ET.NUMERO,'') NUMERO
  FROM EQUIPAMENTOS_TEXTOS ET, EQUIPAMENTOS_PROCEDIMENTOS EP
 WHERE ET.EQUIPAMENTO(+) =  EP.EQUIPAMENTO
   AND ET.MNEMONICO(+) =  EP.MNEMONICO;

CREATE OR REPLACE PROCEDURE PRC_TEXTOS(V_MNEMONICO     IN OUT TEXTOS.MNEMONICO%TYPE,
                                       V_NUMERO        IN     TEXTOS.NUMERO%TYPE,
                                       V_DESCRICAO     IN     TEXTOS.DESCRICAO%TYPE,
                                       V_TEXTO         IN     LONG,
                                       V_TIPO          IN     TEXTOS.TIPO%TYPE,
                                       V_STATUS        IN     TEXTOS.STATUS%TYPE,
                                       V_DIRECIONADO   IN     TEXTOS.DIRECIONADO%TYPE,
                                       V_NUMERO_NEW    IN     TEXTOS.NUMERO%TYPE,
                                       V_FUNCAO        IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO TEXTOS     VALUES (V_MNEMONICO, V_NUMERO, V_DESCRICAO, V_TEXTO, V_TIPO, V_STATUS, V_DIRECIONADO); 

  ELSIF V_FUNCAO = 2 THEN
     UPDATE TEXTOS SET DESCRICAO = V_DESCRICAO, TEXTO = V_TEXTO, TIPO = V_TIPO, STATUS = V_STATUS, DIRECIONADO = V_DIRECIONADO 
                       WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE TEXTOS     WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE TEXTOS     SET NUMERO = V_NUMERO_NEW WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE TEXTOS     SET DESCRICAO = V_DESCRICAO WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO;
  END IF;

  COMMIT WORK;

END PRC_TEXTOS;
/

CREATE OR REPLACE PROCEDURE PRC_WORKLIST_RESULTADOS(V_POSTO          IN WORKLIST_RESULTADOS.POSTO%TYPE,
                                                    V_ATENDIMENTO    IN WORKLIST_RESULTADOS.ATENDIMENTO%TYPE,
                                                    V_CORREL         IN WORKLIST_RESULTADOS.CORREL%TYPE,
                                                    V_EQUIPAMENTO    IN WORKLIST_RESULTADOS.EQUIPAMENTO%TYPE,
                                                    V_DATA_RESULTADO IN WORKLIST_RESULTADOS.DATA_RESULTADO%TYPE,
                                                    V_DATA_CAD       IN WORKLIST_RESULTADOS.DATA_CAD%TYPE,
                                                    V_USUARIO        IN WORKLIST_RESULTADOS.USUARIO%TYPE,
                                                    V_RESULTADO      IN WORKLIST_RESULTADOS.RESULTADO%TYPE,
                                                    V_FUNCAO         IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO WORKLIST_RESULTADOS VALUES (V_POSTO, V_ATENDIMENTO, V_CORREL, V_EQUIPAMENTO, 
                                                   V_DATA_RESULTADO, V_DATA_CAD, V_USUARIO, V_RESULTADO); 

  ELSIF V_FUNCAO = 2 THEN
     UPDATE WORKLIST_RESULTADOS SET RESULTADO = V_RESULTADO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
                                      WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND
                                            CORREL = V_CORREL AND EQUIPAMENTO = V_EQUIPAMENTO AND
                                            DATA_RESULTADO = V_DATA_RESULTADO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE WORKLIST_RESULTADOS WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND
                                            CORREL = V_CORREL AND EQUIPAMENTO = V_EQUIPAMENTO;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE WORKLIST_RESULTADOS SET RESULTADO = V_RESULTADO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
                                      WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND
                                            CORREL = V_CORREL AND EQUIPAMENTO = V_EQUIPAMENTO;
  END IF;

  COMMIT WORK;

END PRC_WORKLIST_RESULTADOS;
/
