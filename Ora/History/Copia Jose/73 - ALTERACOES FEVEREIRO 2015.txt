CREATE OR REPLACE FUNCTION GET_NOME_TABELA(V_CONVENIO IN NUMBER) RETURN VARCHAR2 IS vNOME_TABELA VARCHAR2(30);

vSQL           VARCHAR2(256);    --string que recebe o comando SQL dinâmico

BEGIN
     vSQL := 'SELECT DISTINCT ''TABELA'' || C.CATALOGO FROM CONVENIOS C WHERE C.CONVENIO = ' || V_CONVENIO;

     EXECUTE IMMEDIATE vSQL INTO vNOME_TABELA;

     RETURN(vNOME_TABELA);

END GET_NOME_TABELA;

CREATE SEQUENCE SQ_GRUPO_OBJ MINVALUE 0 
MAXVALUE 999999
INCREMENT BY 1
START WITH 0
NOCACHE
NOORDER;

CREATE OR REPLACE PROCEDURE PRC_GRUPOS_OBJ(V_GRUPO_OBJ         IN OUT GRUPOS_OBJ.GRUPO_OBJ%TYPE,
                                           V_NOME_GRUPO        IN     GRUPOS_OBJ.NOME_GRUPO%TYPE,
                                           V_TIPO_OBJ          IN     GRUPOS_OBJ.TIPO_OBJ%TYPE,
                                           V_FUNCAO            IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_GRUPO_OBJ.NEXTVAL INTO V_GRUPO_OBJ FROM DUAL;
     INSERT INTO GRUPOS_OBJ VALUES (V_GRUPO_OBJ, V_NOME_GRUPO, V_TIPO_OBJ);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE GRUPOS_OBJ SET NOME_GRUPO = V_NOME_GRUPO, TIPO_OBJ = V_TIPO_OBJ
      WHERE GRUPO_OBJ = V_GRUPO_OBJ;

  ELSIF V_FUNCAO = 3 THEN
     DELETE GRUPOS_OBJ WHERE GRUPO_OBJ = V_GRUPO_OBJ;
  END IF;

  COMMIT WORK;

END PRC_GRUPOS_OBJ;

                                                  V_POSTO             IN     GRUPOS_OBJ_POSTOS.POSTO%TYPE,
                                                  V_FUNCAO            IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO GRUPOS_OBJ_POSTOS VALUES (V_GRUPO_OBJ, V_POSTO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE GRUPOS_OBJ_POSTOS WHERE GRUPO_OBJ = V_GRUPO_OBJ AND POSTO = V_POSTO;
  END IF;

  COMMIT WORK;

END PRC_GRUPOS_OBJ_POSTOS;



CREATE TABLE EQUIPAMENTOS_VARIAVEIS_FRASES
(EQUIPAMENTO          NUMBER(3) NOT NULL,
 VARIAVEL             NUMBER(6) NOT NULL,
 CODIGO               VARCHAR2(7) NOT NULL,
 MNEMONICO            VARCHAR2(7) NOT NULL,
 FRASE_EQUIPAMENTO    VARCHAR2(2000) NOT NULL,
 TIPO_COMPARACAO      VARCHAR2(3) NOT NULL)
TABLESPACE LIS_DATA;

ALTER TABLE EQUIPAMENTOS_VARIAVEIS_FRASES ADD CONSTRAINT EQUIP_VARIAVEIS_FRASES_PK 
      PRIMARY KEY (EQUIPAMENTO, VARIAVEL, CODIGO, MNEMONICO);

ALTER TABLE EQUIPAMENTOS_VARIAVEIS_FRASES ADD CONSTRAINT EQUIP_VAR_FRASES_EQUIP_FK 
      FOREIGN KEY (EQUIPAMENTO) REFERENCES EQUIPAMENTOS(EQUIPAMENTO)
      ON DELETE CASCADE;

ALTER TABLE EQUIPAMENTOS_VARIAVEIS_FRASES ADD CONSTRAINT EQUIP_VAR_FRASES_FRASES_FK 
      FOREIGN KEY (CODIGO) REFERENCES FRASES(CODIGO)
      ON DELETE CASCADE;

ALTER TABLE EQUIPAMENTOS_VARIAVEIS_FRASES ADD CONSTRAINT EQUIP_VAR_FRASES_VARIAVEIS_FK 
      FOREIGN KEY (VARIAVEL) REFERENCES VARIAVEIS(VARIAVEL)
      ON DELETE CASCADE;
	  

CREATE OR REPLACE PROCEDURE PRC_EQUIP_VARIAVEIS_FRASES(V_EQUIPAMENTO           IN EQUIPAMENTOS_VARIAVEIS_FRASES.EQUIPAMENTO%TYPE,
                                                       V_VARIAVEL              IN EQUIPAMENTOS_VARIAVEIS_FRASES.VARIAVEL%TYPE,
                                                       V_CODIGO                IN EQUIPAMENTOS_VARIAVEIS_FRASES.CODIGO%TYPE,
                                                       V_MNEMONICO             IN EQUIPAMENTOS_VARIAVEIS_FRASES.MNEMONICO%TYPE,
                                                       V_FRASE_EQUIPAMENTO     IN EQUIPAMENTOS_VARIAVEIS_FRASES.FRASE_EQUIPAMENTO%TYPE,
													   V_TIPO_COMPARACAO       IN EQUIPAMENTOS_VARIAVEIS_FRASES.TIPO_COMPARACAO%TYPE,
                                                       V_FUNCAO                IN  NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO EQUIPAMENTOS_VARIAVEIS_FRASES VALUES (V_EQUIPAMENTO, V_VARIAVEL, V_CODIGO, V_MNEMONICO, V_FRASE_EQUIPAMENTO, V_TIPO_COMPARACAO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE EQUIPAMENTOS_VARIAVEIS_FRASES SET FRASE_EQUIPAMENTO = V_FRASE_EQUIPAMENTO, TIPO_COMPARACAO = V_TIPO_COMPARACAO
            WHERE EQUIPAMENTO = V_EQUIPAMENTO AND VARIAVEL = V_VARIAVEL AND CODIGO = V_CODIGO AND MNEMONICO = V_MNEMONICO;
     
  ELSIF V_FUNCAO = 3 THEN
     DELETE EQUIPAMENTOS_VARIAVEIS_FRASES 
            WHERE EQUIPAMENTO = V_EQUIPAMENTO AND VARIAVEL = V_VARIAVEL AND CODIGO = V_CODIGO AND MNEMONICO = V_MNEMONICO;
                                                                                               
  END IF;

  COMMIT WORK;

END PRC_EQUIP_VARIAVEIS_FRASES;
