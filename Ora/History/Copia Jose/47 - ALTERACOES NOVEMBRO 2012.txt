CREATE OR REPLACE PROCEDURE PRC_COMUNICADOS_PROCEDIMENTOS(V_MNEMONICO  IN OUT COMUNICADOS_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                          V_CORREL     IN  COMUNICADOS_PROCEDIMENTOS.CORREL%TYPE,
                                                          V_CONVENIO   IN  COMUNICADOS_PROCEDIMENTOS.CONVENIO%TYPE,
                                                          V_COMUNICADO IN  COMUNICADOS_PROCEDIMENTOS.COMUNICADO%TYPE,
                                                          V_TIPO       IN  COMUNICADOS_PROCEDIMENTOS.TIPO%TYPE,
                                                          V_DATA_CAD   IN  COMUNICADOS_PROCEDIMENTOS.DATA_CAD%TYPE,
                                                          V_USUARIO    IN  COMUNICADOS_PROCEDIMENTOS.USUARIO%TYPE,
                                                          V_FUNCAO     IN  NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO COMUNICADOS_PROCEDIMENTOS VALUES (V_MNEMONICO, V_CORREL, V_CONVENIO, V_COMUNICADO, V_TIPO, V_DATA_CAD, V_USUARIO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE COMUNICADOS_PROCEDIMENTOS SET CONVENIO   = V_CONVENIO,
                                          COMUNICADO = V_COMUNICADO, TIPO = V_TIPO,
                                          DATA_CAD   = V_DATA_CAD,   USUARIO = V_USUARIO
                                    WHERE MNEMONICO = V_MNEMONICO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 3 THEN
     DELETE COMUNICADOS_PROCEDIMENTOS WHERE MNEMONICO = V_MNEMONICO AND CORREL = V_CORREL;

  END IF;

  COMMIT WORK;

END PRC_COMUNICADOS_PROCEDIMENTOS;

CREATE OR REPLACE PROCEDURE PRC_TEXTOS_COMPARTILHADOS(V_MNEMONICO          IN OUT TEXTOS_COMPARTILHADOS.MNEMONICO%TYPE,
                                      		      V_NUMERO             IN     TEXTOS_COMPARTILHADOS.NUMERO%TYPE,
                                       		      V_MNEM_COMPARTILHADO IN     TEXTOS_COMPARTILHADOS.MNEM_COMPARTILHADO%TYPE,
                                       	              V_NUM_COMPARTILHADO  IN     TEXTOS_COMPARTILHADOS.NUM_COMPARTILHADO%TYPE,
                                       		      V_FUNCAO             IN     NUMBER) IS
CURSOR CUR_VARIAVEIS_TEXTOS IS
  SELECT MNEMONICO, NUMERO, VARIAVEL 
    FROM VARIAVEIS_TEXTOS
   WHERE MNEMONICO = V_MNEM_COMPARTILHADO AND NUMERO = V_NUM_COMPARTILHADO 
     AND (VARIAVEL) NOT IN (SELECT VT.VARIAVEL
                              FROM VARIAVEIS_TEXTOS VT
                             WHERE VT.MNEMONICO = V_MNEMONICO 
                               AND VT.NUMERO = V_NUMERO);

BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO TEXTOS_COMPARTILHADOS VALUES (V_MNEMONICO, V_NUMERO, V_MNEM_COMPARTILHADO, V_NUM_COMPARTILHADO); 

     IF (V_MNEMONICO <> V_MNEM_COMPARTILHADO) OR (V_NUMERO <> V_NUM_COMPARTILHADO) THEN

     	  FOR VARIAVEIS_TEXTOS_CUR IN CUR_VARIAVEIS_TEXTOS LOOP
            INSERT INTO VARIAVEIS_TEXTOS VALUES (V_MNEMONICO, V_NUMERO, VARIAVEIS_TEXTOS_CUR.VARIAVEL, 'S');
        END LOOP;
     END IF;


  ELSIF V_FUNCAO = 3 THEN
     DELETE TEXTOS_COMPARTILHADOS WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO AND
                                              MNEM_COMPARTILHADO = V_MNEM_COMPARTILHADO AND 
                                              NUM_COMPARTILHADO = V_NUM_COMPARTILHADO;

     FOR VARIAVEIS_TEXTOS_CUR IN CUR_VARIAVEIS_TEXTOS LOOP
         DELETE VARIAVEIS_TEXTOS WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO AND 
                                             VARIAVEL = VARIAVEIS_TEXTOS_CUR.VARIAVEL;
     END LOOP;

  END IF;

  COMMIT WORK;

END PRC_TEXTOS_COMPARTILHADOS;
