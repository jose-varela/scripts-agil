create or replace FUNCTION GET_DATA_WORKLIST_RESULTADOS(V_POSTO       IN NUMBER,
                                                       V_ATENDIMENTO IN NUMBER,
                                                       V_CORREL      IN NUMBER) RETURN DATE IS RESULT DATE;
BEGIN
     SELECT MAX(WR.DATA_RESULTADO) INTO RESULT FROM WORKLIST_RESULTADOS WR
      WHERE WR.POSTO = V_POSTO
        AND WR.ATENDIMENTO = V_ATENDIMENTO
        AND WR.CORREL = V_CORREL;
    RETURN(RESULT);
END GET_DATA_WORKLIST_RESULTADOS;
