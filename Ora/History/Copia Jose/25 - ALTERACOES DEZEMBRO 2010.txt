CREATE OR REPLACE PROCEDURE PRC_SERVICOS_B2B(V_ID_SERVICO_B2B   IN OUT SERVICOS_B2B.ID_SERVICO_B2B%TYPE,
                                             V_NOME             IN     SERVICOS_B2B.NOME%TYPE,
                                             V_PASTA_DESTINO    IN     SERVICOS_B2B.PASTA_DESTINO%TYPE,
                                             V_PASTA_ORIGEM     IN     SERVICOS_B2B.PASTA_ORIGEM%TYPE,
                                             V_LOGIN            IN     SERVICOS_B2B.LOGIN%TYPE,
                                             V_PASS             IN     SERVICOS_B2B.PASS%TYPE,
                                             V_LOTE             IN     SERVICOS_B2B.LOTE%TYPE,
                                             V_TIPO_SERVICO_B2B IN     SERVICOS_B2B.TIPO_SERVICO_B2B%TYPE,
                                             V_FUNCAO           IN     NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO SERVICOS_B2B VALUES (V_ID_SERVICO_B2B, V_NOME, V_PASTA_DESTINO, V_PASTA_ORIGEM, V_LOGIN, V_PASS, V_LOTE, V_TIPO_SERVICO_B2B);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE SERVICOS_B2B SET NOME = V_NOME, PASTA_DESTINO = V_PASTA_DESTINO, PASTA_ORIGEM = V_PASTA_ORIGEM, LOGIN = V_LOGIN, PASS = V_PASS, LOTE = V_LOTE, TIPO_SERVICO_B2B = V_TIPO_SERVICO_B2B 
      WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B;

  ELSIF V_FUNCAO = 3 THEN
     DELETE SERVICOS_B2B WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B;
     
  ELSIF V_FUNCAO = 4 THEN
     UPDATE SERVICOS_B2B SET LOTE = LOTE + 1 
      WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B;

  END IF;             

  COMMIT WORK;

END PRC_SERVICOS_B2B;
/


-- FACO UPDATE PARA NAO PERDER OS DADOS
UPDATE SERVICOS_B2B SB SET SB.LOGIN = (SELECT DISTINCT(PSB.LOGIN) FROM POSTOS_SERVICOS_B2B PSB WHERE PSB.ID_SERVICO_B2B = SB.ID_SERVICO_B2B AND PSB.LOGIN IS NOT NULL);
UPDATE SERVICOS_B2B SB SET SB.PASS = (SELECT DISTINCT(PSB.PASS) FROM POSTOS_SERVICOS_B2B PSB WHERE PSB.ID_SERVICO_B2B = SB.ID_SERVICO_B2B AND PSB.PASS IS NOT NULL);


-- ELIMINO COLUNAS NAT TAB POSTOS_SERVICOS_B2B
ALTER TABLE POSTOS_SERVICOS_B2B DROP COLUMN LOGIN;
ALTER TABLE POSTOS_SERVICOS_B2B DROP COLUMN PASS;

CREATE OR REPLACE PROCEDURE PRC_POSTOS_SERVICOS_B2B(V_POSTO          IN POSTOS_SERVICOS_B2B.POSTO%TYPE,
                                                    V_ID_SERVICO_B2B IN POSTOS_SERVICOS_B2B.ID_SERVICO_B2B%TYPE,
                                                    V_FUNCAO         IN      NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO POSTOS_SERVICOS_B2B VALUES (V_POSTO, V_ID_SERVICO_B2B);

  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS_SERVICOS_B2B WHERE POSTO = V_POSTO;
  END IF;

  COMMIT WORK;

END PRC_POSTOS_SERVICOS_B2B;
/

CREATE OR REPLACE FUNCTION GET_ULTIMO_WORKLIST_RESULTADO(V_POSTO       IN NUMBER,
                                                         V_ATENDIMENTO IN NUMBER,
                                                         V_CORREL      IN NUMBER,
                                                         V_EQUIPAMENTO IN NUMBER) RETURN DATE IS RESULT DATE;
BEGIN
     SELECT MAX(WR.DATA_RESULTADO) INTO RESULT FROM WORKLIST_RESULTADOS WR
      WHERE WR.POSTO = V_POSTO
        AND WR.ATENDIMENTO = V_ATENDIMENTO
        AND WR.CORREL = V_CORREL
        AND WR.EQUIPAMENTO = V_EQUIPAMENTO;
    RETURN(RESULT);
END GET_ULTIMO_WORKLIST_RESULTADO;
/

CREATE OR REPLACE FUNCTION GET_RES_WORKLIST_RESULTADO(V_POSTO          IN NUMBER,
                                                      V_ATENDIMENTO    IN NUMBER,
                                                      V_CORREL         IN NUMBER,
                                                      V_EQUIPAMENTO    IN NUMBER,
                                                      V_DATA_RESULTADO IN DATE) RETURN VARCHAR2 IS RESULT VARCHAR2(500);
BEGIN
     SELECT WR.RESULTADO INTO RESULT FROM WORKLIST_RESULTADOS WR
      WHERE WR.POSTO = V_POSTO
        AND WR.ATENDIMENTO = V_ATENDIMENTO
        AND WR.CORREL = V_CORREL
        AND WR.EQUIPAMENTO = V_EQUIPAMENTO
        AND WR.DATA_RESULTADO = V_DATA_RESULTADO;
    RETURN(RESULT);
END GET_RES_WORKLIST_RESULTADO;
/
