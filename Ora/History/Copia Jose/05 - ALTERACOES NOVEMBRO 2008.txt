CREATE TABLE SERVICOS_B2B
(ID_SERVICO_B2B  VARCHAR2(4) NOT NULL,
 NOME            VARCHAR2(30) NOT NULL)
TABLESPACE LIS_DATA;

ALTER TABLE SERVICOS_B2B ADD CONSTRAINT SERVICOS_B2B_PK 
      PRIMARY KEY (ID_SERVICO_B2B);


CREATE OR REPLACE PROCEDURE PRC_SERVICOS_B2B(V_ID_SERVICO_B2B  IN OUT SERVICOS_B2B.ID_SERVICO_B2B%TYPE,
                                             V_NOME            IN     SERVICOS_B2B.NOME%TYPE,
                                             V_FUNCAO          IN     NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO SERVICOS_B2B VALUES (V_ID_SERVICO_B2B, V_NOME);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE SERVICOS_B2B SET NOME = V_NOME WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B;

  ELSIF V_FUNCAO = 3 THEN
     DELETE SERVICOS_B2B WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B;
     
  END IF;             

  COMMIT WORK;

END PRC_SERVICOS_B2B;
/


CREATE TABLE MAP_B2B_PROCEDIMENTOS
(ID_SERVICO_B2B  VARCHAR2(4) NOT NULL,
 MNEMONICO       VARCHAR2(4) NOT NULL,
 MAP_B2B         VARCHAR2(10))
TABLESPACE LIS_DATA;

ALTER TABLE MAP_B2B_PROCEDIMENTOS ADD CONSTRAINT MAP_B2B_PROCEDIMENTOS_PK 
      PRIMARY KEY (ID_SERVICO_B2B, MNEMONICO);


CREATE INDEX MAP_B2B_PROCEDIMENTOS_I 
       ON MAP_B2B_PROCEDIMENTOS (ID_SERVICO_B2B, MAP_B2B) 
       TABLESPACE LIS_INDX;


CREATE OR REPLACE PROCEDURE PRC_MAP_B2B_PROCEDIMENTOS(V_ID_SERVICO_B2B  IN   MAP_B2B_PROCEDIMENTOS.ID_SERVICO_B2B%TYPE,
                                                      V_MNEMONICO       IN   MAP_B2B_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                      V_MAP_B2B         IN   MAP_B2B_PROCEDIMENTOS.MAP_B2B%TYPE,
                                                      V_FUNCAO          IN   NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO MAP_B2B_PROCEDIMENTOS VALUES (V_ID_SERVICO_B2B, V_MNEMONICO, V_MAP_B2B);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE MAP_B2B_PROCEDIMENTOS SET MAP_B2B = V_MAP_B2B WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND MNEMONICO = V_MNEMONICO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE MAP_B2B_PROCEDIMENTOS WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND MNEMONICO = V_MNEMONICO;
     
  END IF;             

  COMMIT WORK;

END PRC_MAP_B2B_PROCEDIMENTOS;
/

ALTER TABLE MAP_B2B_PROCEDIMENTOS ADD CONSTRAINT MAP_B2B_PROC_PROCEDIMENTOS_FK 
      FOREIGN KEY (MNEMONICO) REFERENCES PROCEDIMENTOS(MNEMONICO)
      ON DELETE CASCADE;

ALTER TABLE MAP_B2B_PROCEDIMENTOS ADD CONSTRAINT MAP_B2B_PROC_SERVICOS_B2B_FK 
      FOREIGN KEY (ID_SERVICO_B2B) REFERENCES SERVICOS_B2B(ID_SERVICO_B2B)
      ON DELETE CASCADE;
