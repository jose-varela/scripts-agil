CREATE TABLE POSTOS_MEDICOS_RESPONSAVEIS
(POSTO           NUMBER(2) NOT NULL,
 CRM_RESPONSAVEL VARCHAR2(9) NOT NULL,
 SETOR           NUMBER(6)
)
TABLESPACE LIS_DATA;


alter table POSTOS_MEDICOS_RESPONSAVEIS
  add constraint POSTOS_MEDICOS_RESPONSAVEIS_PK primary key (POSTO, CRM_RESPONSAVEL, SETOR)
  using index 
  tablespace LIS_INDX
  pctfree 10
  initrans 2
  maxtrans 255
  storage
  (
    initial 1M
    next 512K
    minextents 1
    maxextents 121
    pctincrease 0
  )

CREATE OR REPLACE PROCEDURE PRC_POSTOS_MEDICOS_RESP(V_POSTO           IN POSTOS_MEDICOS_RESPONSAVEIS.POSTO%TYPE,
                                                    V_CRM_RESPONSAVEL IN POSTOS_MEDICOS_RESPONSAVEIS.CRM_RESPONSAVEL%TYPE,
                                                    V_SETOR           IN POSTOS_MEDICOS_RESPONSAVEIS.SETOR%TYPE,
                                                    V_FUNCAO          IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO POSTOS_MEDICOS_RESPONSAVEIS VALUES (V_POSTO, V_CRM_RESPONSAVEL, V_SETOR);
 
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS_MEDICOS_RESPONSAVEIS WHERE POSTO = V_POSTO AND CRM_RESPONSAVEL = V_CRM_RESPONSAVEL AND SETOR = V_SETOR;

  END IF;

  COMMIT WORK;

END PRC_POSTOS_MEDICOS_RESP;
/

ALTER TABLE POSTOS_MEDICOS_RESPONSAVEIS ADD CONSTRAINT POSTOS_MEDICOS_POSTOS_FK 
      FOREIGN KEY (POSTO) REFERENCES POSTOS(POSTO);
