CREATE SEQUENCE SQ_FATURA MINVALUE 0 
MAXVALUE 999999
INCREMENT BY 1
START WITH 0
NOCACHE
NOORDER;

ALTER TABLE ATENDIMENTOS 
      DROP CONSTRAINT ATENDIMENTOS_FATURAMENTO_CK;

ALTER TABLE ATENDIMENTOS 
      ADD CONSTRAINT ATENDIMENTOS_FATURAMENTO_CK 
          CHECK (FATURAMENTO IN ('C','F','N','A','G'));


alter table FATURAS
  add constraint FATURAS_PK primary key (ID_FATURA)

CREATE OR REPLACE PROCEDURE PRC_FATURAS(V_CONVENIO    IN FATURAS.CONVENIO%TYPE,
                                        V_DATA_DESDE  IN FATURAS.DATA_DESDE%TYPE,
                                        V_DATA_ATE    IN FATURAS.DATA_ATE%TYPE,
                                        V_FECHADA     IN FATURAS.FECHADA%TYPE,
                                        V_DISKETE     IN FATURAS.DISKETE%TYPE,
                                        V_STATUS      IN FATURAS.STATUS%TYPE,
                                        V_USUARIO     IN FATURAS.USUARIO%TYPE,
                                        V_DATA_CAD    IN FATURAS.DATA_CAD%TYPE,
                                        V_VALOR_FAT   IN FATURAS.VALOR_FAT%TYPE,
                                        V_VALOR_PAG   IN FATURAS.VALOR_PAG%TYPE,
                                        V_VALOR_GLO   IN FATURAS.VALOR_GLO%TYPE,
                                        V_ID_FATURA   IN OUT FATURAS.ID_FATURA%TYPE,
                                        V_FUNCAO      IN NUMBER) IS
BEGIN

  IF (V_ID_FATURA = 0) THEN
     SELECT SQ_FATURA.NEXTVAL INTO V_ID_FATURA FROM DUAL;
  END IF;

  IF V_FUNCAO = 1 THEN
     INSERT INTO FATURAS VALUES (V_CONVENIO,  V_DATA_DESDE, V_DATA_ATE, V_FECHADA,
                                 V_DISKETE,   V_STATUS,     V_USUARIO,  V_DATA_CAD,
                                 V_VALOR_FAT, V_VALOR_PAG,  V_VALOR_GLO, V_ID_FATURA);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE FATURAS SET VALOR_FAT = V_VALOR_FAT, VALOR_PAG = V_VALOR_PAG, VALOR_GLO = V_VALOR_GLO,
                              USUARIO   = V_USUARIO,   DATA_CAD  = V_DATA_CAD
                      WHERE CONVENIO = V_CONVENIO AND DATA_DESDE = V_DATA_DESDE;

  ELSIF V_FUNCAO = 3 THEN
     DELETE FATURAS WHERE CONVENIO = V_CONVENIO AND DATA_DESDE = V_DATA_DESDE;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE FATURAS SET DATA_ATE = V_DATA_ATE, USUARIO = V_USUARIO, DATA_CAD = V_DATA_CAD
                      WHERE CONVENIO = V_CONVENIO AND DATA_DESDE = V_DATA_DESDE;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE FATURAS SET FECHADA = V_FECHADA, STATUS = V_STATUS,
                              USUARIO = V_USUARIO, DATA_CAD  = V_DATA_CAD
                      WHERE CONVENIO = V_CONVENIO AND DATA_DESDE = V_DATA_DESDE;
  END IF;

  COMMIT WORK;
END PRC_FATURAS;

CREATE OR REPLACE PROCEDURE PRC_ATENDIMENTOS(V_POSTO                IN     ATENDIMENTOS.POSTO%TYPE,
                                             V_REGISTRO             IN     ATENDIMENTOS.REGISTRO%TYPE,
                                             V_ATENDIMENTO          IN OUT ATENDIMENTOS.ATENDIMENTO%TYPE,
                                             V_CONVENIO             IN     ATENDIMENTOS.CONVENIO%TYPE,
                                             V_MATRICULA            IN     ATENDIMENTOS.MATRICULA%TYPE,
                                             V_DATA_ATD             IN     ATENDIMENTOS.DATA_ATD%TYPE,
                                             V_DATA_FAT             IN     ATENDIMENTOS.DATA_FAT%TYPE,
                                             V_DATA_ENT             IN     ATENDIMENTOS.DATA_ENT%TYPE,
                                             V_HORA_ENT             IN     ATENDIMENTOS.HORA_ENT%TYPE,
                                             V_MED_SOL              IN     ATENDIMENTOS.MED_SOL%TYPE,
                                             V_LIB_FAT              IN     ATENDIMENTOS.LIB_FAT%TYPE,
                                             V_FATURAMENTO          IN     ATENDIMENTOS.FATURAMENTO%TYPE,
                                             V_USUARIO              IN     ATENDIMENTOS.USUARIO%TYPE,
                                             V_DATA_CAD             IN     ATENDIMENTOS.DATA_CAD%TYPE,
                                             V_POSTO_ENTREGA        IN     ATENDIMENTOS.POSTO_ENTREGA%TYPE,
                                             V_SITUACAO             IN     ATENDIMENTOS.SITUACAO%TYPE,
                                             V_DATA_DESDE           IN     ATENDIMENTOS.DATA_FAT%TYPE,
                                             V_DATA_ATE             IN     ATENDIMENTOS.DATA_FAT%TYPE,
                                             V_CONDICAO_FATURAMENTO IN     ATENDIMENTOS.FATURAMENTO%TYPE,
                                             V_CODIGO_PLANO         IN     ATENDIMENTOS.CODIGO_PLANO%TYPE,
                                             V_DATA_VALIDADE        IN     ATENDIMENTOS.DATA_VALIDADE%TYPE,
                                             V_ID_FATURA            IN     ATENDIMENTOS.ID_FATURA%TYPE,
                                             V_FUNCAO               IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 OR V_FUNCAO = 9 THEN
     IF V_FUNCAO = 1 THEN
       SELECT SQ_ATENDIMENTO.NEXTVAL INTO V_ATENDIMENTO FROM DUAL;
     END IF;

     INSERT INTO ATENDIMENTOS VALUES (V_POSTO, V_REGISTRO, V_ATENDIMENTO, V_CONVENIO,
        V_MATRICULA, V_DATA_ATD, V_DATA_FAT, V_DATA_ENT, V_HORA_ENT, V_MED_SOL, V_LIB_FAT,
        V_FATURAMENTO, V_USUARIO, V_DATA_CAD, V_POSTO_ENTREGA, V_SITUACAO, V_CODIGO_PLANO, V_DATA_VALIDADE, V_ID_FATURA);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE ATENDIMENTOS SET CONVENIO     = V_CONVENIO,      MATRICULA = V_MATRICULA,
           DATA_ATD = V_DATA_ATD, DATA_FAT      = V_DATA_FAT,      DATA_ENT  = V_DATA_ENT,
           HORA_ENT = V_HORA_ENT, MED_SOL       = V_MED_SOL,       LIB_FAT   = V_LIB_FAT,
           USUARIO  = V_USUARIO,  FATURAMENTO   = V_FATURAMENTO,   DATA_CAD  = V_DATA_CAD,      
           POSTO    = V_POSTO,    POSTO_ENTREGA = V_POSTO_ENTREGA, SITUACAO  = V_SITUACAO,
           CODIGO_PLANO = V_CODIGO_PLANO, DATA_VALIDADE = V_DATA_VALIDADE, ID_FATURA = V_ID_FATURA
        WHERE REGISTRO = V_REGISTRO AND  ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE ATENDIMENTOS WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE ATENDIMENTOS SET DATA_ENT  = V_DATA_ENT,
                                   HORA_ENT = V_HORA_ENT,
                                   POSTO_ENTREGA = V_POSTO_ENTREGA,
                                   DATA_CAD = V_DATA_CAD
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE ATENDIMENTOS SET LIB_FAT = V_LIB_FAT
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 6 THEN
     IF V_CONDICAO_FATURAMENTO = '' THEN
        UPDATE ATENDIMENTOS SET FATURAMENTO = V_FATURAMENTO, ID_FATURA = V_ID_FATURA
               WHERE CONVENIO = V_CONVENIO AND DATA_FAT >= V_DATA_DESDE
                 AND DATA_FAT <= V_DATA_ATE AND LIB_FAT = 'S';
     ELSE
        UPDATE ATENDIMENTOS SET FATURAMENTO = V_FATURAMENTO, ID_FATURA = V_ID_FATURA
               WHERE CONVENIO = V_CONVENIO   AND DATA_FAT >= V_DATA_DESDE
                  AND DATA_FAT <= V_DATA_ATE AND LIB_FAT = 'S' AND FATURAMENTO = V_CONDICAO_FATURAMENTO;
     END IF;

  ELSIF V_FUNCAO = 7 THEN
     UPDATE ATENDIMENTOS SET LIB_FAT = V_LIB_FAT,
                                 DATA_FAT = V_DATA_FAT
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 8 THEN
     UPDATE ATENDIMENTOS SET FATURAMENTO = V_FATURAMENTO
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 10 THEN
     UPDATE ATENDIMENTOS SET DATA_ENT  = V_DATA_ENT,
                                   HORA_ENT = V_HORA_ENT
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 11 THEN
     UPDATE ATENDIMENTOS SET MED_SOL  = V_MED_SOL
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 12 THEN
     UPDATE ATENDIMENTOS SET CONVENIO  = V_CONVENIO,
                                   DATA_FAT  = V_DATA_FAT,
                                   LIB_FAT   = V_LIB_FAT,
                                   MED_SOL   = V_MED_SOL,
                                   MATRICULA = V_MATRICULA
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 13 THEN
     UPDATE ATENDIMENTOS SET MATRICULA = V_MATRICULA
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;

  ELSIF V_FUNCAO = 14 THEN
     UPDATE ATENDIMENTOS SET SITUACAO = V_SITUACAO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
          WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO;
  END IF;

  COMMIT WORK;
END PRC_ATENDIMENTOS;

CREATE OR REPLACE TRIGGER ATENDIMENTOS_FATURAMENTO_TRG
 AFTER UPDATE ON ATENDIMENTOS
FOR EACH ROW
WHEN (NEW.FATURAMENTO <> OLD.FATURAMENTO)
DECLARE
BEGIN
     IF :NEW.FATURAMENTO = 'F' THEN
            UPDATE EXAMES E SET E.VALOR_PAGO = E.VALOR_HON
             WHERE E.POSTO = :NEW.POSTO
               AND E.ATENDIMENTO = :NEW.ATENDIMENTO
               AND E.SITUACAO <> 'C' 
               AND E.LIB_FAT = 'S';

     ELSIF :NEW.FATURAMENTO = 'C' THEN
            UPDATE EXAMES E SET E.VALOR_PAGO = 0
             WHERE E.POSTO = :NEW.POSTO
               AND E.ATENDIMENTO = :NEW.ATENDIMENTO
               AND E.SITUACAO <> 'C' 
               AND E.LIB_FAT = 'S';
     END IF;
END;

CREATE OR REPLACE PROCEDURE PRC_CAIXA1(V_CONTA_HOSP     IN CAIXA.CONTA_HOSP%TYPE,
                                      V_REGISTRO       IN CAIXA.REGISTRO%TYPE,
                                      V_CONTA          IN CAIXA.CONTA%TYPE,
                                      V_DATA_PAG       IN CAIXA.DATA_PAG%TYPE,
                                      V_VALOR          IN CAIXA.VALOR%TYPE,
                                      V_DESCONTO       IN CAIXA.DESCONTO%TYPE,
                                      V_TIPO_DTO       IN CAIXA.TIPO_DTO%TYPE,
                                      V_USUARIO        IN CAIXA.USUARIO%TYPE,
                                      V_DATA_CAD       IN CAIXA.DATA_CAD%TYPE,
                                      V_CORREL         IN OUT CAIXA.CORREL%TYPE,
                                      V_TIPO_PAG       IN CAIXA.TIPO_PAG%TYPE,
                                      V_CATALOGO       IN CAIXA.CATALOGO%TYPE,
                                      V_FUNCAO         IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT NVL(MAX(C.CORREL)+1,1) INTO V_CORREL FROM CAIXA C 
      WHERE C.CONTA_HOSP = V_CONTA_HOSP;

     INSERT INTO CAIXA VALUES (V_CONTA_HOSP,  V_REGISTRO,  V_CONTA,     V_DATA_PAG,
                                     V_VALOR,       V_DESCONTO,  V_TIPO_DTO,  V_USUARIO,
                                     V_DATA_CAD,    V_CORREL,    V_TIPO_PAG,  V_CATALOGO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CAIXA SET VALOR = V_VALOR
                      WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CAIXA WHERE CONTA_HOSP = V_CONTA_HOSP;

  ELSIF V_FUNCAO = 4 THEN
     DELETE CAIXA WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE CAIXA SET VALOR = V_VALOR, DATA_PAG = V_DATA_PAG, DESCONTO = V_DESCONTO, TIPO_DTO = V_TIPO_DTO, TIPO_PAG = V_TIPO_PAG
                      WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;
  END IF;

  COMMIT WORK;
END PRC_CAIXA1;
