CREATE TABLE TEMPOS_COLETA
(ID_TEMPO_COLETA    NUMBER(3)    NOT NULL,
 NOME               VARCHAR2(50) NOT NULL,
 STATUS             VARCHAR2(1)  NOT NULL)
TABLESPACE LIS_DATA;


CREATE TABLE TEMPOS_COLETA_PROCEDIMENTOS
(ID_TEMPO_COLETA    NUMBER(3)    NOT NULL,
 MNEMONICO          VARCHAR2(4)  NOT NULL)
TABLESPACE LIS_DATA;


CREATE SEQUENCE SQ_TEMPO_COLETA
MINVALUE 0
MAXVALUE 999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;

ALTER TABLE TEMPOS_COLETA ADD CONSTRAINT TEMPOS_COLETA_PK 
      PRIMARY KEY (ID_TEMPO_COLETA);


ALTER TABLE TEMPOS_COLETA_PROCEDIMENTOS ADD CONSTRAINT TEMPOS_COLETA_PROCEDIMENTOS_PK 
      PRIMARY KEY (ID_TEMPO_COLETA, MNEMONICO);


ALTER TABLE TEMPOS_COLETA_PROCEDIMENTOS ADD CONSTRAINT TEMP_COLETA_PROC_PROC_FK 
      FOREIGN KEY (MNEMONICO) REFERENCES PROCEDIMENTOS(MNEMONICO);

ALTER TABLE TEMPOS_COLETA_PROCEDIMENTOS ADD CONSTRAINT TEMP_COLETA_PROC_TEMP_COLE_FK 
      FOREIGN KEY (ID_TEMPO_COLETA) REFERENCES TEMPOS_COLETA(ID_TEMPO_COLETA);



CREATE OR REPLACE PROCEDURE PRC_TEMPOS_COLETA(V_ID_TEMPO_COLETA    IN OUT TEMPOS_COLETA.ID_TEMPO_COLETA%TYPE,
                                              V_NOME               IN TEMPOS_COLETA.NOME%TYPE,
                                              V_STATUS             IN TEMPOS_COLETA.STATUS%TYPE,
                                              V_FUNCAO             IN NUMBER) IS
BEGIN                                                           
  IF V_FUNCAO = 1 THEN                                          
     SELECT SQ_TEMPO_COLETA.NEXTVAL INTO V_ID_TEMPO_COLETA FROM DUAL;
     INSERT INTO TEMPOS_COLETA VALUES(V_ID_TEMPO_COLETA, V_NOME, V_STATUS);

  ELSIF V_FUNCAO = 2 THEN
            UPDATE TEMPOS_COLETA SET NOME       = V_NOME,
                                     STATUS     = V_STATUS
                          WHERE ID_TEMPO_COLETA = V_ID_TEMPO_COLETA;

  ELSIF V_FUNCAO = 3 THEN
            DELETE TEMPOS_COLETA WHERE ID_TEMPO_COLETA = V_ID_TEMPO_COLETA;	

  END IF;

  COMMIT WORK;
END PRC_TEMPOS_COLETA;
/

CREATE OR REPLACE PROCEDURE PRC_TEMPOS_COLETA_PROCED(V_ID_TEMPO_COLETA IN TEMPOS_COLETA_PROCEDIMENTOS.ID_TEMPO_COLETA%TYPE,
                                                     V_MNEMONICO       IN TEMPOS_COLETA_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                     V_FUNCAO          IN NUMBER) IS
BEGIN                                                           
  IF V_FUNCAO = 1 THEN                                          
     INSERT INTO TEMPOS_COLETA_PROCEDIMENTOS VALUES(V_ID_TEMPO_COLETA, V_MNEMONICO);


  ELSIF V_FUNCAO = 3 THEN
            DELETE TEMPOS_COLETA_PROCEDIMENTOS WHERE ID_TEMPO_COLETA = V_ID_TEMPO_COLETA AND MNEMONICO = V_MNEMONICO;	

  END IF;

  COMMIT WORK;
END PRC_TEMPOS_COLETA_PROCED;
/

CREATE OR REPLACE PROCEDURE PRC_FRASES(V_CODIGO          IN OUT FRASES.CODIGO%TYPE,
                                       V_FRASE           IN     FRASES.FRASE%TYPE,
                                       V_TIPO_FRASE      IN     FRASES.TIPO_FRASE%TYPE,
                                       V_MNEMONICO_FRASE IN     FRASES.MNEMONICO_FRASE%TYPE,
                                       V_FRASE_FORMAT    IN     FRASES.FRASE_FORMAT%TYPE,
                                       V_FUNCAO          IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     IF V_CODIGO IS NULL THEN
        SELECT TO_CHAR(SQ_FRASE.NEXTVAL) INTO V_CODIGO FROM DUAL;
     END IF;

     INSERT INTO FRASES VALUES (V_CODIGO, V_FRASE, V_TIPO_FRASE, V_MNEMONICO_FRASE, V_FRASE_FORMAT);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE FRASES SET FRASE = V_FRASE, TIPO_FRASE = V_TIPO_FRASE, MNEMONICO_FRASE = V_MNEMONICO_FRASE, FRASE_FORMAT = V_FRASE_FORMAT
                       WHERE CODIGO = V_CODIGO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE FRASES WHERE CODIGO = V_CODIGO;
  END IF;

  COMMIT WORK;

END PRC_FRASES;
