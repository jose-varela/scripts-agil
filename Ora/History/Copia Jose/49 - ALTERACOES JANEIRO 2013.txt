ALTER TABLE EQUIPAMENTOS_TEXTOS ADD MAP_EQUIPAMENTO VARCHAR2(200);
ALTER TABLE EQUIPAMENTOS_TEXTOS ADD STATUS VARCHAR2(1);

UPDATE EQUIPAMENTOS_TEXTOS ET SET MAP_EQUIPAMENTO = (SELECT EP.MNEMONICO_EQUIPAMENTO FROM EQUIPAMENTOS_PROCEDIMENTOS EP
                                                      WHERE EP.EQUIPAMENTO = ET.EQUIPAMENTO
                                                        AND EP.MNEMONICO = ET.MNEMONICO
														AND EP.POSTO_REA = ET.POSTO_REA);

UPDATE EQUIPAMENTOS_TEXTOS ET SET STATUS = (SELECT EP.STATUS FROM EQUIPAMENTOS_PROCEDIMENTOS EP
                                                      WHERE EP.EQUIPAMENTO = ET.EQUIPAMENTO
                                                        AND EP.MNEMONICO = ET.MNEMONICO
														AND EP.POSTO_REA = ET.POSTO_REA);
														

CREATE OR REPLACE PROCEDURE PRC_EQUIPAMENTOS_TEXTOS(V_EQUIPAMENTO           IN EQUIPAMENTOS_TEXTOS.EQUIPAMENTO%TYPE,
													V_MNEMONICO             IN EQUIPAMENTOS_TEXTOS.MNEMONICO%TYPE,
													V_NUMERO                IN EQUIPAMENTOS_TEXTOS.NUMERO%TYPE,
													V_POSTO_REA             IN EQUIPAMENTOS_TEXTOS.POSTO_REA%TYPE,
													V_MAP_EQUIPAMENTO       IN EQUIPAMENTOS_TEXTOS.MAP_EQUIPAMENTO%TYPE,
													V_STATUS                IN EQUIPAMENTOS_TEXTOS.STATUS%TYPE,
													V_FUNCAO                IN NUMBER) IS
BEGIN
	 IF V_FUNCAO = 1 THEN
	    INSERT INTO EQUIPAMENTOS_TEXTOS VALUES (V_EQUIPAMENTO, V_MNEMONICO, V_NUMERO, V_POSTO_REA, V_MAP_EQUIPAMENTO, V_STATUS);

	 ELSIF V_FUNCAO = 2 THEN
		UPDATE EQUIPAMENTOS_TEXTOS SET MAP_EQUIPAMENTO = V_MAP_EQUIPAMENTO, STATUS = V_STATUS
		 WHERE EQUIPAMENTO = V_EQUIPAMENTO AND MNEMONICO = V_MNEMONICO 
		   AND POSTO_REA = V_POSTO_REA AND NUMERO = V_NUMERO;
		 
	 ELSIF V_FUNCAO = 3 THEN
		DELETE EQUIPAMENTOS_TEXTOS WHERE EQUIPAMENTO = V_EQUIPAMENTO AND MNEMONICO = V_MNEMONICO 
									  AND POSTO_REA = V_POSTO_REA AND NUMERO = V_NUMERO;
																								   
	 END IF;

	 COMMIT WORK;

END PRC_EQUIPAMENTOS_TEXTOS;
/
