
CREATE TABLE CONVERSOES
(ID_CONVERSAO       NUMBER(4) NOT NULL,
 NOME_CONVERSAO     VARCHAR2(50) NOT NULL,
 TIPO_CONVERSAO     VARCHAR2(2) NOT NULL,
 FORMULA_CONVERSAO  VARCHAR2(500))
TABLESPACE LIS_DATA;

ALTER TABLE CONVERSOES ADD CONSTRAINT CONVERSOES_PK 
      PRIMARY KEY (ID_CONVERSAO);

CREATE SEQUENCE SQ_CONVERSAO MINVALUE 0 
MAXVALUE 9999
INCREMENT BY 1
START WITH 0
NOCACHE
NOORDER;

CREATE OR REPLACE PROCEDURE PRC_CONVERSOES(V_ID_CONVERSAO        IN OUT CONVERSOES.ID_CONVERSAO%TYPE,
                                           V_NOME_CONVERSAO      IN     CONVERSOES.NOME_CONVERSAO%TYPE,
                                           V_TIPO_CONVERSAO      IN     CONVERSOES.TIPO_CONVERSAO%TYPE,
										   V_FORMULA_CONVERSAO   IN     CONVERSOES.FORMULA_CONVERSAO%TYPE,
                                           V_FUNCAO              IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_CONVERSAO.NEXTVAL INTO V_ID_CONVERSAO FROM DUAL;
     INSERT INTO CONVERSOES VALUES (V_ID_CONVERSAO, V_NOME_CONVERSAO, V_TIPO_CONVERSAO, V_FORMULA_CONVERSAO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CONVERSOES SET NOME_CONVERSAO = V_NOME_CONVERSAO, TIPO_CONVERSAO = V_TIPO_CONVERSAO, FORMULA_CONVERSAO = V_FORMULA_CONVERSAO
      WHERE ID_CONVERSAO = V_ID_CONVERSAO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CONVERSOES WHERE ID_CONVERSAO = V_ID_CONVERSAO;
  END IF;

  COMMIT WORK;

END PRC_CONVERSOES;



CREATE TABLE EQUIP_VARIAVEIS_CONVERSOES
(EQUIPAMENTO        NUMBER(3) NOT NULL,
 MNEMONICO          VARCHAR2(7) NOT NULL,
 POSTO_REA          NUMBER(2) NOT NULL,
 NUMERO             NUMBER(2) NOT NULL,
 VARIAVEL           NUMBER(6) NOT NULL, 
 ID_CONVERSAO       NUMBER(4) NOT NULL)

      PRIMARY KEY (EQUIPAMENTO, MNEMONICO, POSTO_REA, NUMERO, VARIAVEL, ID_CONVERSAO);

ALTER TABLE EQUIP_VARIAVEIS_CONVERSOES ADD CONSTRAINT EQUIP_VAR_CONVERSOES_EQUIP_FK 
      FOREIGN KEY (EQUIPAMENTO) REFERENCES EQUIPAMENTOS(EQUIPAMENTO);

CREATE OR REPLACE PROCEDURE PRC_EQUIP_VARIAVEIS_CONVERSOES(V_EQUIPAMENTO         IN EQUIP_VARIAVEIS_CONVERSOES.EQUIPAMENTO %TYPE,
                                                           V_MNEMONICO           IN EQUIP_VARIAVEIS_CONVERSOES.MNEMONICO%TYPE,
                                                           V_POSTO_REA           IN EQUIP_VARIAVEIS_CONVERSOES.POSTO_REA%TYPE,
                                                           V_NUMERO              IN EQUIP_VARIAVEIS_CONVERSOES.NUMERO%TYPE,
                                                           V_VARIAVEL            IN EQUIP_VARIAVEIS_CONVERSOES.VARIAVEL%TYPE,
                                                           V_ID_CONVERSAO        IN EQUIP_VARIAVEIS_CONVERSOES.ID_CONVERSAO%TYPE,
                                                           V_FUNCAO              IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO EQUIP_VARIAVEIS_CONVERSOES VALUES (V_EQUIPAMENTO, V_MNEMONICO, V_POSTO_REA, V_NUMERO, V_VARIAVEL, V_ID_CONVERSAO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE EQUIP_VARIAVEIS_CONVERSOES WHERE EQUIPAMENTO = V_EQUIPAMENTO AND MNEMONICO = V_MNEMONICO AND POSTO_REA = V_POSTO_REA AND NUMERO = V_NUMERO AND VARIAVEL = V_VARIAVEL AND ID_CONVERSAO = V_ID_CONVERSAO;
  END IF;

  COMMIT WORK;

END PRC_EQUIP_VARIAVEIS_CONVERSOES;
