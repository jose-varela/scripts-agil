CREATE OR REPLACE PROCEDURE PRC_POSTOS_SERVICOS_B2B(V_POSTO          IN POSTOS_SERVICOS_B2B.POSTO%TYPE,
                                                    V_ID_SERVICO_B2B IN POSTOS_SERVICOS_B2B.ID_SERVICO_B2B%TYPE,
						    V_SEQ_INICIAL    IN POSTOS_SERVICOS_B2B.SEQ_INICIAL%TYPE,
  						    V_SEQ_ATUAL      IN POSTOS_SERVICOS_B2B.SEQ_ATUAL%TYPE,
						    V_SEQ_FINAL      IN POSTOS_SERVICOS_B2B.SEQ_FINAL%TYPE,
                                                    V_FUNCAO         IN      NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO POSTOS_SERVICOS_B2B VALUES (V_POSTO, V_ID_SERVICO_B2B, V_SEQ_INICIAL, V_SEQ_ATUAL, V_SEQ_FINAL);
  
  ELSIF V_FUNCAO = 2 THEN
	 UPDATE POSTOS_SERVICOS_B2B SET SEQ_INICIAL = V_SEQ_INICIAL, SEQ_FINAL = V_SEQ_FINAL, SEQ_ATUAL = V_SEQ_ATUAL
          WHERE POSTO = V_POSTO;
	 
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS_SERVICOS_B2B WHERE POSTO = V_POSTO;
  
  ELSIF V_FUNCAO = 4 THEN
	 UPDATE POSTOS_SERVICOS_B2B SET SEQ_ATUAL = SEQ_INICIAL + 1
          WHERE POSTO = V_POSTO;
  END IF;

  COMMIT WORK;

END PRC_POSTOS_SERVICOS_B2B;
