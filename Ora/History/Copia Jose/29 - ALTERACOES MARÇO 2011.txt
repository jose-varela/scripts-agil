DROP FUNCTION GET_CORREL_DE;


CREATE OR REPLACE FUNCTION IS_EXAME_CORRELACIONADO_DE(V_POSTO       IN NUMBER,
                                                      V_ATENDIMENTO IN NUMBER,
                                                      V_CORREL_DE   IN NUMBER) RETURN VARCHAR2 IS QT_EXAMES_CORRELACIONADOS NUMBER; 
BEGIN
    SELECT COUNT(1) INTO QT_EXAMES_CORRELACIONADOS FROM EXAMES_CORRELACIONADOS
     WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL_DE = V_CORREL_DE;

    IF QT_EXAMES_CORRELACIONADOS = 0 THEN
       RETURN 'Z';
    ELSE
       RETURN 'A';
    END IF;
END IS_EXAME_CORRELACIONADO_DE;                                      
/

DROP FUNCTION GET_CORREL_PARA;

CREATE OR REPLACE FUNCTION IS_EXAME_CORRELACIONADO_PARA(V_POSTO       IN NUMBER,
                                                        V_ATENDIMENTO IN NUMBER,
                                                        V_CORREL_PARA IN NUMBER) RETURN VARCHAR2 IS QT_EXAMES_CORRELACIONADOS NUMBER; 
BEGIN
    SELECT COUNT(1) INTO QT_EXAMES_CORRELACIONADOS FROM EXAMES_CORRELACIONADOS
     WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL_PARA = V_CORREL_PARA;

    IF QT_EXAMES_CORRELACIONADOS = 0 THEN
       RETURN 'Z';
    ELSE
       RETURN 'A';
    END IF;
END IS_EXAME_CORRELACIONADO_PARA;                                      
/


CREATE OR REPLACE FUNCTION GET_ORDEM_CORRELACAO(V_POSTO       IN NUMBER,
                                                V_ATENDIMENTO IN NUMBER,
                                                V_CORREL      IN NUMBER) RETURN NUMBER IS RESULT NUMBER; 
BEGIN
      SELECT NVL(ORDEM_CORRELACAO,0) INTO RESULT FROM EXAMES_CORRELACIONADOS
       WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND (CORREL_DE = V_CORREL OR CORREL_PARA = V_CORREL);
  
      RETURN(RESULT);

END GET_ORDEM_CORRELACAO;                                      
/


CREATE OR REPLACE FUNCTION GET_MAX_ORDEM_CORRELACAO(V_POSTO       IN NUMBER,
                                                    V_ATENDIMENTO IN NUMBER) RETURN NUMBER IS RESULT NUMBER;
BEGIN
     SELECT NVL(MAX(EC.ORDEM_CORRELACAO),0) INTO RESULT FROM EXAMES_CORRELACIONADOS EC
      WHERE EC.POSTO = V_POSTO
        AND EC.ATENDIMENTO = V_ATENDIMENTO;
    RETURN(RESULT);
END GET_MAX_ORDEM_CORRELACAO;
/


CREATE OR REPLACE PROCEDURE PRC_EXAMES_CORRELACIONADOS(V_POSTO            IN EXAMES_CORRELACIONADOS.POSTO%TYPE,
                                                       V_ATENDIMENTO      IN EXAMES_CORRELACIONADOS.ATENDIMENTO%TYPE,
                                                       V_CORREL_DE        IN EXAMES_CORRELACIONADOS.CORREL_DE%TYPE,
                                                       V_CORREL_PARA      IN EXAMES_CORRELACIONADOS.CORREL_PARA%TYPE,
                                                       V_ORDEM_CORRELACAO IN OUT EXAMES_CORRELACIONADOS.ORDEM_CORRELACAO%TYPE,
                                                       V_FUNCAO           IN NUMBER) IS
BEGIN                                                       
  IF V_FUNCAO = 1 THEN                 
     SELECT GET_MAX_ORDEM_CORRELACAO(V_POSTO,V_ATENDIMENTO)+1 INTO V_ORDEM_CORRELACAO FROM DUAL;
                     
     INSERT INTO EXAMES_CORRELACIONADOS
            VALUES (V_POSTO, V_ATENDIMENTO, V_CORREL_DE, V_CORREL_PARA, V_ORDEM_CORRELACAO);

  ELSIF V_FUNCAO = 3 THEN
            DELETE EXAMES_CORRELACIONADOS WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND
                                                ORDEM_CORRELACAO = V_ORDEM_CORRELACAO;
  END IF;

  COMMIT WORK;
END PRC_EXAMES_CORRELACIONADOS;
/
