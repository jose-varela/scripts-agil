CREATE OR REPLACE PROCEDURE PRC_GRUPOS_ETIQUETAS(V_GRUPO_ETIQUETA         IN OUT GRUPOS_ETIQUETAS.GRUPO_ETIQUETA%TYPE,
                                                 V_NOME                   IN     GRUPOS_ETIQUETAS.NOME%TYPE,
                                                 V_MATERIAL               IN     GRUPOS_ETIQUETAS.MATERIAL%TYPE,
                                                 V_QT_ETIQUETAS           IN     GRUPOS_ETIQUETAS.QT_ETIQUETAS%TYPE,
                                                 V_SEPARA_ETIQUETAS       IN     GRUPOS_ETIQUETAS.SEPARA_ETIQUETAS%TYPE,
                                                 V_TIPO_CODIGO            IN     GRUPOS_ETIQUETAS.TIPO_CODIGO%TYPE,
                                                 V_MARGEM_ESQUERDA_CODIGO IN     GRUPOS_ETIQUETAS.MARGEM_ESQUERDA_CODIGO%TYPE,
                                                 V_RATIO_CODIGO           IN     GRUPOS_ETIQUETAS.RATIO_CODIGO%TYPE,
                                                 V_REALCA_NOME            IN     GRUPOS_ETIQUETAS.REALCA_NOME%TYPE,
                                                 V_FUNCAO                 IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_GRUPO_ETIQUETA.NEXTVAL INTO V_GRUPO_ETIQUETA FROM DUAL;
     INSERT INTO GRUPOS_ETIQUETAS VALUES (V_GRUPO_ETIQUETA, V_NOME, V_MATERIAL, V_QT_ETIQUETAS, V_SEPARA_ETIQUETAS, V_TIPO_CODIGO, V_MARGEM_ESQUERDA_CODIGO, V_RATIO_CODIGO, V_REALCA_NOME);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE GRUPOS_ETIQUETAS SET NOME = V_NOME, MATERIAL = V_MATERIAL, QT_ETIQUETAS = V_QT_ETIQUETAS, SEPARA_ETIQUETAS = V_SEPARA_ETIQUETAS,
            TIPO_CODIGO = V_TIPO_CODIGO, MARGEM_ESQUERDA_CODIGO = V_MARGEM_ESQUERDA_CODIGO, RATIO_CODIGO = V_RATIO_CODIGO, REALCA_NOME = V_REALCA_NOME
      WHERE GRUPO_ETIQUETA = V_GRUPO_ETIQUETA;

  ELSIF V_FUNCAO = 3 THEN
     DELETE GRUPOS_ETIQUETAS WHERE GRUPO_ETIQUETA = V_GRUPO_ETIQUETA;
  END IF;

  COMMIT WORK;

END PRC_GRUPOS_ETIQUETAS;

CREATE OR REPLACE PROCEDURE PRC_POSTOS(V_POSTO            IN OUT POSTOS.POSTO%TYPE,
                                       V_NOME             IN     POSTOS.NOME%TYPE,
                                       V_INFORMATIZADO    IN     POSTOS.INFORMATIZADO%TYPE,
                                       V_HORA_ENT         IN     POSTOS.HORA_ENT%TYPE,
                                       V_INTERNO          IN     POSTOS.INTERNO%TYPE,
                                       V_DATA_CAD         IN     POSTOS.DATA_CAD%TYPE,
                                       V_REALIZA_EXAMES   IN     POSTOS.REALIZA_EXAMES%TYPE,
                                       V_DIRECIONA_EXAMES IN     POSTOS.DIRECIONA_EXAMES%TYPE,
                                       V_STATUS           IN     POSTOS.STATUS%TYPE,
                                       V_COLETA_MATERIAL  IN     POSTOS.COLETA_MATERIAL%TYPE,
                                       V_TIPO_POSTO       IN     POSTOS.TIPO_POSTO%TYPE,
                                       V_PASS             IN     POSTOS.PASS%TYPE,
                                       V_ENDERECO         IN     POSTOS.ENDERECO%TYPE,
                                       V_MNEMONICO_POSTO  IN     POSTOS.MNEMONICO_POSTO%TYPE,
                                       V_NOME_FICHA       IN     POSTOS.NOME_FICHA%TYPE,
                                       V_CABECALHO_LAUDO  IN     POSTOS.CABECALHO_LAUDO%TYPE,
                                       V_FUNCAO           IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_POSTO.NEXTVAL INTO V_POSTO FROM DUAL;
     INSERT INTO POSTOS VALUES (V_POSTO, V_NOME, V_INFORMATIZADO, V_HORA_ENT, V_INTERNO, V_DATA_CAD,
                                V_REALIZA_EXAMES, V_DIRECIONA_EXAMES, V_STATUS, V_COLETA_MATERIAL, 
                                V_TIPO_POSTO, V_PASS, V_ENDERECO, V_MNEMONICO_POSTO, V_NOME_FICHA, EMPTY_BLOB);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE POSTOS SET NOME = V_NOME,     INFORMATIZADO = V_INFORMATIZADO, 
                             HORA_ENT = V_HORA_ENT, INTERNO = V_INTERNO, DATA_CAD = V_DATA_CAD,
                             REALIZA_EXAMES = V_REALIZA_EXAMES, DIRECIONA_EXAMES = V_DIRECIONA_EXAMES,
                             STATUS = V_STATUS, COLETA_MATERIAL = V_COLETA_MATERIAL, TIPO_POSTO = V_TIPO_POSTO, PASS = V_PASS,
                             ENDERECO = V_ENDERECO, MNEMONICO_POSTO = V_MNEMONICO_POSTO, NOME_FICHA = V_NOME_FICHA
                        WHERE POSTO = V_POSTO;
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS WHERE POSTO = V_POSTO;

  END IF;

  COMMIT WORK;

END PRC_POSTOS;
