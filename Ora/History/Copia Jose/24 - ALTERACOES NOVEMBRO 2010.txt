  RENAME COLUMN CENTRO_RASTREO TO CENTRO_RASTREIO;


DROP SEQUENCE LIS.SQ_CENTRO_RASTREO;

CREATE SEQUENCE LIS.SQ_CENTRO_RASTREIO;


CREATE OR REPLACE PROCEDURE PRC_CENTROS_RASTREO(V_CENTRO_RASTREIO  IN OUT CENTROS_RASTREO.CENTRO_RASTREIO%TYPE,
                                                V_NOME             IN     CENTROS_RASTREO.NOME%TYPE,
                                                V_TIPO_CENTRO      IN     CENTROS_RASTREO.NOME%TYPE,
                                                V_FUNCAO           IN     NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_CENTRO_RASTREIO.NEXTVAL INTO V_CENTRO_RASTREIO FROM DUAL;
     INSERT INTO CENTROS_RASTREO VALUES (V_CENTRO_RASTREIO, V_NOME, V_TIPO_CENTRO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CENTROS_RASTREO SET NOME = V_NOME, TIPO_CENTRO = V_TIPO_CENTRO
      WHERE CENTRO_RASTREIO = V_CENTRO_RASTREIO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CENTROS_RASTREO WHERE CENTRO_RASTREIO = V_CENTRO_RASTREIO;
     
  END IF;             

  COMMIT WORK;

END PRC_CENTROS_RASTREO;

CREATE TABLE LOTES_RASTROS
(ID_LOTE_RASTRO  NUMBER(6) NOT NULL,
 DATA_RASTRO     DATE NOT NULL,
 TIPO_RASTRO     VARCHAR2(1) NOT NULL)
TABLESPACE LIS_DATA;

ALTER TABLE LIS.LOTES_RASTROS
  ADD PRIMARY KEY(ID_LOTE_RASTRO);


CREATE SEQUENCE SQ_LOTE_RASTRO MINVALUE 0 
MAXVALUE 99999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;

CREATE OR REPLACE PROCEDURE PRC_LOTES_RASTROS(V_ID_LOTE_RASTRO IN OUT LOTES_RASTROS.ID_LOTE_RASTRO%TYPE,
                                              V_DATA_RASTRO    IN     LOTES_RASTROS.DATA_RASTRO%TYPE,
                                              V_TIPO_RASTRO    IN     LOTES_RASTROS.TIPO_RASTRO%TYPE,
                                              V_FUNCAO         IN     NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_LOTE_RASTRO.NEXTVAL INTO V_ID_LOTE_RASTRO FROM DUAL;
     INSERT INTO LOTES_RASTROS VALUES (V_ID_LOTE_RASTRO, V_DATA_RASTRO, V_TIPO_RASTRO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE LOTES_RASTROS WHERE ID_LOTE_RASTRO = V_ID_LOTE_RASTRO;
     
  END IF;             

  COMMIT WORK;

END PRC_LOTES_RASTROS;

CREATE OR REPLACE PROCEDURE PRC_RASTROS_AMOSTRAS(V_POSTO           IN RASTROS_AMOSTRAS.POSTO%TYPE,
                                                 V_ATENDIMENTO     IN RASTROS_AMOSTRAS.ATENDIMENTO%TYPE,
                                                 V_CORREL          IN RASTROS_AMOSTRAS.CORREL%TYPE,
                                                 V_AMOSTRA         IN RASTROS_AMOSTRAS.AMOSTRA%TYPE,
                                                 V_CENTRO_RASTREIO IN RASTROS_AMOSTRAS.CENTRO_RASTREIO%TYPE,
                                                 V_USUARIO_RASTRO  IN RASTROS_AMOSTRAS.USUARIO_RASTRO%TYPE,
                                                 V_DATA_RASTRO     IN RASTROS_AMOSTRAS.DATA_RASTRO%TYPE,
                                                 V_ESTACAO_RASTRO  IN RASTROS_AMOSTRAS.ESTACAO_RASTRO%TYPE,
                                                 V_POSTO_RASTRO    IN RASTROS_AMOSTRAS.POSTO_RASTRO%TYPE,
                                                 V_ID_LOTE_RASTRO  IN RASTROS_AMOSTRAS.ID_LOTE_RASTRO%TYPE,
                                                 V_FUNCAO          IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO RASTROS_AMOSTRAS VALUES (V_POSTO, V_ATENDIMENTO, V_CORREL, V_AMOSTRA,
                                          V_CENTRO_RASTREIO, V_USUARIO_RASTRO, V_DATA_RASTRO, V_ESTACAO_RASTRO, 
                                          V_POSTO_RASTRO, V_ID_LOTE_RASTRO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE RASTROS_AMOSTRAS WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO
                                   AND CORREL = V_CORREL AND AMOSTRA = V_AMOSTRA;
  END IF;

  COMMIT WORK;

END PRC_RASTROS_AMOSTRAS;
