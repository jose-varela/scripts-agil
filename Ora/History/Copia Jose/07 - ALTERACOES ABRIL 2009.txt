CREATE TABLE POSTOS_SERVICOS_B2B
(POSTO           NUMBER(2) NOT NULL,
 ID_SERVICO_B2B  VARCHAR2(4) NOT NULL)
TABLESPACE LIS_DATA;

ALTER TABLE POSTOS_SERVICOS_B2B ADD CONSTRAINT POSTOS_SERVICOS_B2B_PK 
      PRIMARY KEY (POSTO, ID_SERVICO_B2B);

ALTER TABLE POSTOS_SERVICOS_B2B ADD CONSTRAINT POSTOS_B2B_SERVICOS_B2B_FK 
      FOREIGN KEY (ID_SERVICO_B2B) REFERENCES SERVICOS_B2B(ID_SERVICO_B2B)
      ON DELETE CASCADE;

ALTER TABLE POSTOS_SERVICOS_B2B ADD CONSTRAINT POSTOS_B2B_POSTOS_FK 
      FOREIGN KEY (POSTO) REFERENCES POSTOS(POSTO)
      ON DELETE CASCADE;


CREATE OR REPLACE PROCEDURE PRC_POSTOS_SERVICOS_B2B(V_POSTO          IN POSTOS_SERVICOS_B2B.POSTO%TYPE,
                                                    V_ID_SERVICO_B2B IN POSTOS_SERVICOS_B2B.ID_SERVICO_B2B%TYPE,
                                                    V_FUNCAO         IN      NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO POSTOS_SERVICOS_B2B VALUES (V_POSTO, V_ID_SERVICO_B2B);

  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS_SERVICOS_B2B WHERE POSTO = V_POSTO;
  END IF;

  COMMIT WORK;

END PRC_POSTOS_SERVICOS_B2B;
