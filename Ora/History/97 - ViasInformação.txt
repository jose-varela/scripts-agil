-- CRIAMOS SEQUENCIA DE VIAS COMECANDO DE 3
------------------------------------------------------------------------------------------------------------------
CREATE SEQUENCE SQ_VIAS_INFORMACAO
MINVALUE 0 
MAXVALUE 9999
INCREMENT BY 1
START WITH 3
NOCACHE
NOORDER;

CREATE OR REPLACE PROCEDURE PRC_VIAS_INFORMACAO(V_ID_VIA_INFORMACAO     IN OUT CIS.VIAS_INFORMACAO.ID_VIA_INFORMACAO%TYPE,
                                                V_NOME                  IN     CIS.VIAS_INFORMACAO.NOME%TYPE,
                                                V_TAG                   IN     CIS.VIAS_INFORMACAO.TAG%TYPE,
                                                V_FUNCAO                IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT CIS.SQ_VIAS_INFORMACAO.NEXTVAL INTO V_ID_VIA_INFORMACAO FROM DUAL;
     INSERT INTO CIS.VIAS_INFORMACAO VALUES (V_ID_VIA_INFORMACAO, V_NOME, V_TAG);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CIS.VIAS_INFORMACAO SET NOME = V_NOME, TAG = V_TAG
                       WHERE ID_VIA_INFORMACAO = V_ID_VIA_INFORMACAO;
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE CIS.VIAS_INFORMACAO WHERE ID_VIA_INFORMACAO = V_ID_VIA_INFORMACAO;

  END IF;

  COMMIT WORK;

END PRC_VIAS_INFORMACAO;

------------------------------------------------------------------------------------------------------------------
-- INSERIMOS VIAS PARA OS CONVENIOS 0 -> PARTICULAR / 1->CONVENIOS
------------------------------------------------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE PRC_INS(V_ID_VIA    IN   NUMBER) IS
                    
         CURSOR CR_PARTICULAR IS  
                SELECT ID_CONVENIO FROM CONVENIOS WHERE PAG_ATO = 'S';

         CURSOR CR_CONVENIOS IS  
                SELECT ID_CONVENIO FROM CONVENIOS WHERE PAG_ATO <> 'S';
                
         V_CONVENIO NUMBER;
BEGIN
     
     OPEN CR_PARTICULAR;
     LOOP
         
         FETCH CR_PARTICULAR INTO V_CONVENIO;
         EXIT WHEN CR_PARTICULAR%NOTFOUND;
         
         INSERT INTO CONVENIOS_VIAS_INFORMACAO VALUES (0,V_CONVENIO); 
         
     END LOOP;
     
     CLOSE CR_PARTICULAR;

     OPEN CR_CONVENIOS;
     LOOP
         
         FETCH CR_CONVENIOS INTO V_CONVENIO;
         EXIT WHEN CR_CONVENIOS%NOTFOUND;
         
         INSERT INTO CONVENIOS_VIAS_INFORMACAO VALUES (1,V_CONVENIO); 
         
     END LOOP;
     
     CLOSE CR_CONVENIOS;
  
END PRC_INS;


EXEC PRC_INS(1);

COMMIT;

ALTER TABLE FATURAS ADD (ID_VIA_INFORMACAO NUMBER(2));
UPDATE FATURAS SET ID_VIA_INFORMACAO = 1;

      FOREIGN KEY (ID_VIA_INFORMACAO) REFERENCES VIAS_INFORMACAO(ID_VIA_INFORMACAO);


CREATE OR REPLACE PROCEDURE PRC_FATURAS(V_ID_CONVENIO       IN CIS.FATURAS.ID_CONVENIO%TYPE,
                                        V_CORREL_FATURA     IN OUT CIS.FATURAS.CORREL_FATURA%TYPE,
                                        V_DATA_DESDE        IN CIS.FATURAS.DATA_DESDE%TYPE,
                                        V_DATA_ATE          IN CIS.FATURAS.DATA_ATE%TYPE,
                                        V_FECHADA           IN CIS.FATURAS.FECHADA%TYPE,
                                        V_STATUS            IN CIS.FATURAS.STATUS%TYPE,
                                        V_DATA_HORA_ALT     IN CIS.FATURAS.DATA_HORA_ALT%TYPE,
                                        V_USUARIO_ALT       IN CIS.FATURAS.USUARIO_ALT%TYPE,
                                        V_VALOR_FAT         IN CIS.FATURAS.VALOR_FAT%TYPE,
                                        V_VALOR_PAG         IN CIS.FATURAS.VALOR_PAG%TYPE,
                                        V_VALOR_GLO         IN CIS.FATURAS.VALOR_GLO%TYPE,
                                        V_ID_FATURA         IN OUT CIS.FATURAS.ID_FATURA%TYPE,
                                        V_TIPO_GUIA         IN CIS.FATURAS.TIPO_GUIA%TYPE,
                                        V_CODIGO_PESSOA     IN CIS.FATURAS.CODIGO_PESSOA%TYPE,
                                        V_ID_VIA_INFORMACAO IN CIS.FATURAS.ID_VIA_INFORMACAO%TYPE,
                                        V_FUNCAO            IN NUMBER) IS
BEGIN 
  IF V_FUNCAO = 1 THEN
     SELECT NVL(MAX(CORREL_FATURA), 0) + 1 INTO V_CORREL_FATURA FROM FATURAS 
         WHERE ID_CONVENIO = V_ID_CONVENIO;
     SELECT CIS.SQ_FATURA.NEXTVAL INTO V_ID_FATURA FROM DUAL;

     INSERT INTO CIS.FATURAS VALUES (V_ID_CONVENIO,  V_CORREL_FATURA, V_DATA_DESDE, V_DATA_ATE, V_FECHADA,
                                     V_STATUS,  V_DATA_HORA_ALT, V_USUARIO_ALT, V_VALOR_FAT, V_VALOR_PAG,  
                                     V_VALOR_GLO, V_ID_FATURA, V_TIPO_GUIA, V_CODIGO_PESSOA, V_ID_VIA_INFORMACAO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CIS.FATURAS SET VALOR_FAT = V_VALOR_FAT, VALOR_PAG = V_VALOR_PAG, VALOR_GLO = V_VALOR_GLO,
                            DATA_HORA_ALT  = V_DATA_HORA_ALT, USUARIO_ALT   = V_USUARIO_ALT, 
                            TIPO_GUIA = V_TIPO_GUIA, CODIGO_PESSOA = V_CODIGO_PESSOA, ID_VIA_INFORMACAO = V_ID_VIA_INFORMACAO
                      WHERE ID_FATURA = V_ID_FATURA;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CIS.FATURAS WHERE ID_FATURA = V_ID_FATURA;

  ELSIF V_FUNCAO = 4 THEN
     UPDATE CIS.FATURAS SET DATA_ATE = V_DATA_ATE, DATA_HORA_ALT = V_DATA_HORA_ALT, USUARIO_ALT = V_USUARIO_ALT
                      WHERE ID_FATURA = V_ID_FATURA;
  ELSIF V_FUNCAO = 5 THEN
     UPDATE CIS.FATURAS SET FECHADA = V_FECHADA, STATUS = V_STATUS,
                            DATA_HORA_ALT  = V_DATA_HORA_ALT, USUARIO_ALT = V_USUARIO_ALT
                      WHERE ID_FATURA = V_ID_FATURA;
  END IF;

  COMMIT WORK;
END PRC_FATURAS;

CREATE TABLE COLUNAS_VIAS_INFORMACAO
(ID_COLUNAS_VIAS     NUMBER(3)    NOT NULL,
 ID_VIA_INFORMACAO   NUMBER(2)    NOT NULL,
 ID_GRUPO            NUMBER(3)    NOT NULL,
 COLUNA		     VARCHAR2(15) NOT NULL,
 OBRIGATORIO         VARCHAR2(1)  NOT NULL,
 FORM                VARCHAR2(20) NOT NULL,
 GRID		     VARCHAR2(15) NOT NULL)
TABLESPACE CIS_DATA
PCTUSED 60
STORAGE ( INITIAL 10 M
          NEXT 2 M
        );

ALTER TABLE COLUNAS_VIAS_INFORMACAO ADD CONSTRAINT ID_COLUNAS_VIAS_PK  
      PRIMARY KEY (ID_COLUNAS_VIAS);

ALTER TABLE COLUNAS_VIAS_INFORMACAO ADD CONSTRAINT COLUNAS_VIAS_INFORMACAO_FK 
      FOREIGN KEY (ID_VIA_INFORMACAO) REFERENCES VIAS_INFORMACAO(ID_VIA_INFORMACAO) 
      ON DELETE CASCADE;

CREATE SEQUENCE SQ_COLUNAS_VIAS_INFORMACAO
MINVALUE 0
MAXVALUE 999999
INCREMENT BY 1
START WITH 0
NOCACHE
NOORDER;

CREATE OR REPLACE PROCEDURE PRC_COLUNAS_VIAS_INFORMACAO(V_ID_COLUNAS_VIAS    IN  OUT  COLUNAS_VIAS_INFORMACAO.ID_COLUNAS_VIAS%TYPE,
                                                        V_ID_VIA_INFORMACAO  IN       COLUNAS_VIAS_INFORMACAO.ID_VIA_INFORMACAO%TYPE,
                                                        V_ID_GRUPO           IN       COLUNAS_VIAS_INFORMACAO.ID_GRUPO%TYPE,
                                                        V_COLUNA	     IN       COLUNAS_VIAS_INFORMACAO.COLUNA%TYPE,
                                                        V_OBRIGATORIO        IN       COLUNAS_VIAS_INFORMACAO.OBRIGATORIO%TYPE,
                                                        V_FORM               IN       COLUNAS_VIAS_INFORMACAO.FORM%TYPE,
                                                        V_GRID               IN       COLUNAS_VIAS_INFORMACAO.GRID%TYPE,
                                                        V_FUNCAO             IN       NUMBER) IS
 BEGIN
   IF V_FUNCAO = 1 THEN
      SELECT SQ_COLUNAS_VIAS_INFORMACAO.NEXTVAL INTO V_ID_COLUNAS_VIAS FROM DUAL;
      INSERT INTO COLUNAS_VIAS_INFORMACAO VALUES (V_ID_COLUNAS_VIAS, V_ID_VIA_INFORMACAO, V_ID_GRUPO, V_COLUNA, V_OBRIGATORIO, V_FORM, V_GRID);
   ELSIF V_FUNCAO = 2 THEN
      UPDATE COLUNAS_VIAS_INFORMACAO SET ID_VIA_INFORMACAO = V_ID_VIA_INFORMACAO, ID_GRUPO = V_ID_GRUPO,
                                   COLUNA = V_COLUNA, OBRIGATORIO = V_OBRIGATORIO, 
                                   FORM = V_FORM, GRID = v_GRID
                                 WHERE ID_COLUNAS_VIAS = V_ID_COLUNAS_VIAS;
   ELSIF V_FUNCAO = 3 THEN
      DELETE COLUNAS_VIAS_INFORMACAO WHERE ID_COLUNAS_VIAS = V_ID_COLUNAS_VIAS;
   END IF;                                 
  COMMIT WORK;
END PRC_COLUNAS_VIAS_INFORMACAO;
