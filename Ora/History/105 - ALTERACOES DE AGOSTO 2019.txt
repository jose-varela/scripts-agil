create or replace PROCEDURE PRC_VARIAVEIS_TEXTOS(V_MNEMONICO     IN VARIAVEIS_TEXTOS.MNEMONICO%TYPE,
                                                 V_NUMERO        IN VARIAVEIS_TEXTOS.NUMERO%TYPE,
                                                 V_VARIAVEL      IN VARIAVEIS_TEXTOS.VARIAVEL%TYPE,
                                                 V_COMPARTILHADO IN VARIAVEIS_TEXTOS.COMPARTILHADO%TYPE,
                                                 V_FUNCAO        IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO VARIAVEIS_TEXTOS VALUES (V_MNEMONICO, V_NUMERO, V_VARIAVEL, V_COMPARTILHADO);

  ELSIF V_FUNCAO = 3 THEN
     DELETE VARIAVEIS_TEXTOS WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO AND VARIAVEL = V_VARIAVEL;

  ELSIF V_FUNCAO = 4 THEN
     DELETE VARIAVEIS_TEXTOS WHERE MNEMONICO = V_MNEMONICO AND NUMERO = V_NUMERO;

  END IF;

  COMMIT WORK;

END PRC_VARIAVEIS_TEXTOS;
