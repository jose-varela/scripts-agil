create or replace
PROCEDURE PRC_CLIENTES_B2B(V_REGISTRO  IN CLIENTES_B2B.REGISTRO%TYPE,
                                              V_EQUIPAMENTO     IN CLIENTES_B2B.EQUIPAMENTO%TYPE,
                                              V_CODIGO_EXTERNO     IN CLIENTES_B2B.CODIGO_EXTERNO%TYPE,
                                              V_NOME_EXTERNO       IN CLIENTES_B2B.NOME_EXTERNO%TYPE,
                                              V_DATA_NAS_EXTERNO   IN CLIENTES_B2B.DATA_NAS_EXTERNO%TYPE,
                                              V_SEXO_EXTERNO       IN CLIENTES_B2B.SEXO_EXTERNO%TYPE,
                                              V_IDENTIDADE_EXTERNO IN CLIENTES_B2B.IDENTIDADE_EXTERNO%TYPE,
                                              V_CPF_EXTERNO        IN CLIENTES_B2B.CPF_EXTERNO%TYPE,
                                              V_FUNCAO             IN NUMBER) IS
 BEGIN
   DECLARE
          V_ACHEI  NUMBER(8);
          CURSOR CUR_PROCURA IS
                    SELECT REGISTRO FROM CLIENTES_B2B
                     WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
   BEGIN                  
        IF V_FUNCAO = 1 THEN
           INSERT INTO CLIENTES_B2B VALUES (V_REGISTRO, V_EQUIPAMENTO, V_CODIGO_EXTERNO, V_NOME_EXTERNO, V_DATA_NAS_EXTERNO, V_SEXO_EXTERNO, V_IDENTIDADE_EXTERNO, V_CPF_EXTERNO);
        ELSIF V_FUNCAO = 2 THEN
            UPDATE CLIENTES_B2B SET NOME_EXTERNO = V_NOME_EXTERNO, DATA_NAS_EXTERNO = V_DATA_NAS_EXTERNO, SEXO_EXTERNO = V_SEXO_EXTERNO, IDENTIDADE_EXTERNO = V_IDENTIDADE_EXTERNO, CPF_EXTERNO = V_CPF_EXTERNO
                              WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
        ELSIF V_FUNCAO = 3 THEN
            DELETE CLIENTES_B2B WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
        ELSIF V_FUNCAO = 4 THEN
            OPEN CUR_PROCURA;
            FETCH CUR_PROCURA INTO V_ACHEI;
            IF CUR_PROCURA%NOTFOUND THEN
               INSERT INTO CLIENTES_B2B VALUES (V_REGISTRO, V_EQUIPAMENTO, V_CODIGO_EXTERNO, V_NOME_EXTERNO, V_DATA_NAS_EXTERNO, V_SEXO_EXTERNO, V_IDENTIDADE_EXTERNO, V_CPF_EXTERNO);
 				   ELSE   
               UPDATE CLIENTES_B2B SET REGISTRO = V_REGISTRO
                                 WHERE EQUIPAMENTO = V_EQUIPAMENTO AND CODIGO_EXTERNO = V_CODIGO_EXTERNO;
           END IF;                      
      END IF;
   END;
   COMMIT WORK;
 END PRC_CLIENTES_B2B;

create or replace PROCEDURE PRC_EXAMES_B2B(V_ID_EXAME_B2B           IN OUT EXAMES_B2B.ID_EXAME_B2B%TYPE,
                                            V_POSTO                  IN EXAMES_B2B.POSTO%TYPE,
                                            V_ATENDIMENTO            IN EXAMES_B2B.ATENDIMENTO%TYPE,
                                            V_CORREL                 IN EXAMES_B2B.CORREL%TYPE,
                                            V_AMOSTRA                IN EXAMES_B2B.AMOSTRA%TYPE,
                                            V_POSTO_B2B              IN EXAMES_B2B.POSTO_B2B%TYPE,
                                            V_EQUIPAMENTO            IN EXAMES_B2B.EQUIPAMENTO%TYPE,
                                            V_CORREL_B2B             IN EXAMES_B2B.CORREL_B2B%TYPE,
                                            V_SITUACAO_B2B           IN EXAMES_B2B.SITUACAO_B2B%TYPE,
                                            V_DATA_B2B               IN EXAMES_B2B.DATA_B2B%TYPE,
                                            V_USUARIO_B2B            IN EXAMES_B2B.USUARIO_B2B%TYPE,
                                            V_ESTACAO_B2B            IN EXAMES_B2B.ESTACAO_B2B%TYPE,
                                            V_LOTE_B2B               IN EXAMES_B2B.LOTE_B2B%TYPE,
 				                            V_CODIGO_EXTERNO         IN EXAMES_B2B.CODIGO_EXTERNO%TYPE,
 					                        V_ETIQUETA_EXTERNO       IN EXAMES_B2B.ETIQUETA_EXTERNO%TYPE,
                                            V_PEDIDO_EXTERNO         IN EXAMES_B2B.PEDIDO_EXTERNO%TYPE,
                                            V_FUNCAO                 IN NUMBER) IS
 BEGIN                                                           
   IF V_FUNCAO = 1 THEN                                          
      SELECT SQ_EXAME_B2B.NEXTVAL INTO V_ID_EXAME_B2B FROM DUAL;
      INSERT INTO EXAMES_B2B VALUES(V_ID_EXAME_B2B, V_POSTO, V_ATENDIMENTO, V_CORREL, V_AMOSTRA, V_POSTO_B2B, 
                                    V_EQUIPAMENTO , V_CORREL_B2B, V_SITUACAO_B2B, V_DATA_B2B, 
                                    V_USUARIO_B2B, V_ESTACAO_B2B, V_LOTE_B2B, V_CODIGO_EXTERNO,V_ETIQUETA_EXTERNO,
                                    V_PEDIDO_EXTERNO);
   ELSIF V_FUNCAO = 2 THEN
             UPDATE EXAMES_B2B SET POSTO                 = V_POSTO,
                                   ATENDIMENTO           = V_ATENDIMENTO,
                                   CORREL                = V_CORREL,
                                   AMOSTRA               = V_AMOSTRA,
                                   POSTO_B2B             = V_POSTO_B2B,
                                   EQUIPAMENTO           = V_EQUIPAMENTO ,
                                   CORREL_B2B            = V_CORREL_B2B,
                                   SITUACAO_B2B          = V_SITUACAO_B2B,
                                   DATA_B2B              = V_DATA_B2B,
                                   USUARIO_B2B           = V_USUARIO_B2B,
                                   ESTACAO_B2B           = V_ESTACAO_B2B,
 				   CODIGO_EXTERNO        = V_CODIGO_EXTERNO,
 				   ETIQUETA_EXTERNO      = V_ETIQUETA_EXTERNO,
                                   PEDIDO_EXTERNO        = V_PEDIDO_EXTERNO
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 3 THEN
             DELETE EXAMES_B2B WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 4 THEN
             DELETE EXAMES_B2B WHERE POSTO = V_POSTO;
   ELSIF V_FUNCAO = 5 THEN
             UPDATE EXAMES_B2B SET SITUACAO_B2B          = V_SITUACAO_B2B,
                                   DATA_B2B              = V_DATA_B2B,
                                   USUARIO_B2B           = V_USUARIO_B2B,
                                   ESTACAO_B2B           = V_ESTACAO_B2B
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 6 THEN
 			UPDATE EXAMES_B2B SET SITUACAO_B2B = V_SITUACAO_B2B
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   ELSIF V_FUNCAO = 7 THEN
 			UPDATE EXAMES_B2B SET PEDIDO_EXTERNO = V_PEDIDO_EXTERNO,
                                  LOTE_B2B = V_LOTE_B2B
                           WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
   END IF;
   COMMIT WORK;
 END PRC_EXAMES_B2B;
