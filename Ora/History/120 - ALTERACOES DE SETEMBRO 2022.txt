--
-- INSUMOS
--
CREATE TABLE INSUMOS (
	ID_INSUMO NUMBER(4) NOT NULL,
	DESCRICAO VARCHAR2(60) NOT NULL,
	VALOR_INSUMO NUMBER(12,2)
 );
-- fim bloco
 
ALTER TABLE INSUMOS ADD CONSTRAINT INSUMO_PK PRIMARY KEY (ID_INSUMO);
-- fim bloco

CREATE SEQUENCE SQ_INSUMOS
MINVALUE 0
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco


CREATE OR REPLACE PROCEDURE PRC_INSUMOS(
                                        V_ID_INSUMO    IN OUT INSUMOS.ID_INSUMO%TYPE,
                                        V_DESCRICAO    IN     INSUMOS.DESCRICAO%TYPE,
                                        V_VALOR_INSUMO IN     INSUMOS.VALOR_INSUMO%TYPE,
                                        V_FUNCAO       IN NUMBER
                                        ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_INSUMOS.NEXTVAL INTO V_ID_INSUMO FROM DUAL;
        INSERT INTO INSUMOS VALUES(V_ID_INSUMO,  
		                                    V_DESCRICAO,                
		                                    V_VALOR_INSUMO);             
		

    ELSIF V_FUNCAO = 2 THEN
        UPDATE INSUMOS SET DESCRICAO = V_DESCRICAO, 
									VALOR_INSUMO = V_VALOR_INSUMO
                              WHERE ID_INSUMO = V_ID_INSUMO;

    ELSIF V_FUNCAO = 3 THEN
        DELETE INSUMOS WHERE ID_INSUMO = V_ID_INSUMO;

    END IF;

    COMMIT WORK;
END PRC_INSUMOS;
-- fim bloco
 
--
-- CUSTOS_PROCEDIMENTOS
--
CREATE TABLE CUSTOS_PROCEDIMENTOS(
	ID_CUSTO NUMBER(4) NOT NULL,
	MNEMONICO VARCHAR2(7 BYTE) NOT NULL,
	ID_INSUMO NUMBER(4) NOT NULL,
	QT_INSUMO NUMBER(9,2)
);
-- fim bloco 

ALTER TABLE CUSTOS_PROCEDIMENTOS ADD CONSTRAINT CUSTOS_PROCEDIMENTOS_PK PRIMARY KEY (ID_CUSTO);
-- fim bloco

ALTER TABLE CUSTOS_PROCEDIMENTOS ADD CONSTRAINT CUSTOS_PROCED_INSUMOS_FK FOREIGN KEY (ID_INSUMO) REFERENCES INSUMOS(ID_INSUMO);
-- fim bloco
 
CREATE SEQUENCE SQ_CUSTOS_PROCEDIMENTOS
MINVALUE 0
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_CUSTOS_PROCEDIMENTOS(
                                        V_ID_CUSTO    IN OUT CUSTOS_PROCEDIMENTOS.ID_INSUMO%TYPE,
                                        V_MNEMONICO   IN     CUSTOS_PROCEDIMENTOS.MNEMONICO%TYPE,
                                        V_ID_INSUMO   IN     CUSTOS_PROCEDIMENTOS.ID_INSUMO%TYPE,
										V_QT_INSUMO   IN     CUSTOS_PROCEDIMENTOS.QT_INSUMO%TYPE,
                                        V_FUNCAO      IN NUMBER
                                        ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_CUSTOS_PROCEDIMENTOS.NEXTVAL INTO V_ID_CUSTO FROM DUAL;
        INSERT INTO CUSTOS_PROCEDIMENTOS VALUES(V_ID_CUSTO,  
		                                    V_MNEMONICO,                
		                                    V_ID_INSUMO,
											V_QT_INSUMO);             
		

    ELSIF V_FUNCAO = 2 THEN
        UPDATE CUSTOS_PROCEDIMENTOS SET MNEMONICO = V_MNEMONICO, 
									    ID_INSUMO = V_ID_INSUMO,
										QT_INSUMO = V_QT_INSUMO
                              WHERE ID_CUSTO = V_ID_CUSTO;

    ELSIF V_FUNCAO = 3 THEN
        DELETE CUSTOS_PROCEDIMENTOS WHERE ID_CUSTO = V_ID_CUSTO;

    END IF;

    COMMIT WORK;
END PRC_CUSTOS_PROCEDIMENTOS;
-- fim bloco
 
CREATE OR REPLACE FUNCTION GET_CUSTO_PROCEDIMENTO(V_MNEMONICO IN VARCHAR2) RETURN NUMBER IS 
L_VALOR_CUSTO NUMBER;
BEGIN
	SELECT COALESCE(SUM(I.VALOR_INSUMO * CP.QT_INSUMO),0) INTO L_VALOR_CUSTO
	  FROM INSUMOS I
	  JOIN CUSTOS_PROCEDIMENTOS CP ON (I.ID_INSUMO=CP.ID_INSUMO)
	 WHERE CP.MNEMONICO = V_MNEMONICO;
	
	RETURN (L_VALOR_CUSTO);
END GET_CUSTO_PROCEDIMENTO;
