CREATE OR REPLACE PROCEDURE PRC_COMPLEMENTO_CLIENTES(V_REGISTRO               IN     COMPLEMENTO_CLIENTES.REGISTRO%TYPE,
                                                     V_TIPAGEM_SANGUINEA      IN     COMPLEMENTO_CLIENTES.TIPAGEM_SANGUINEA%TYPE,
                                                     V_NOME_RESPONSAVEL       IN     COMPLEMENTO_CLIENTES.NOME_RESPONSAVEL%TYPE,
                                                     V_PARENTESCO_RESPONSAVEL IN     COMPLEMENTO_CLIENTES.PARENTESCO_RESPONSAVEL%TYPE,
                                                     V_FOTO                   IN     COMPLEMENTO_CLIENTES.FOTO%TYPE,
                                                     V_IMPRESSAO_DIGITAL      IN     COMPLEMENTO_CLIENTES.IMPRESSAO_DIGITAL%TYPE,
                                                     V_FUNCAO                 IN     NUMBER) IS
BEGIN
     DECLARE
            V_ACHEI  NUMBER(8);
            
            CURSOR CUR_PROCURA IS
                   SELECT REGISTRO FROM COMPLEMENTO_CLIENTES 
                    WHERE REGISTRO = V_REGISTRO;

     BEGIN
          IF V_FUNCAO = 1 THEN
             INSERT INTO COMPLEMENTO_CLIENTES VALUES (V_REGISTRO, V_TIPAGEM_SANGUINEA, 
                                                      V_NOME_RESPONSAVEL, V_PARENTESCO_RESPONSAVEL, 
                                                      EMPTY_BLOB(), V_IMPRESSAO_DIGITAL);

          ELSIF V_FUNCAO = 2 THEN
             UPDATE COMPLEMENTO_CLIENTES SET TIPAGEM_SANGUINEA  = V_TIPAGEM_SANGUINEA,  
                                             NOME_RESPONSAVEL = V_NOME_RESPONSAVEL,
                                             PARENTESCO_RESPONSAVEL = V_PARENTESCO_RESPONSAVEL
              WHERE REGISTRO = V_REGISTRO;

          ELSIF V_FUNCAO = 3 THEN
              DELETE COMPLEMENTO_CLIENTES WHERE REGISTRO = V_REGISTRO;


          ELSIF V_FUNCAO = 4 THEN
                OPEN CUR_PROCURA;
                FETCH CUR_PROCURA INTO V_ACHEI;
  
                IF CUR_PROCURA%NOTFOUND THEN
                   INSERT INTO COMPLEMENTO_CLIENTES VALUES (V_REGISTRO, V_TIPAGEM_SANGUINEA, 
                                                            V_NOME_RESPONSAVEL, V_PARENTESCO_RESPONSAVEL, 
                                                            EMPTY_BLOB(), V_IMPRESSAO_DIGITAL);
                ELSE
                   UPDATE COMPLEMENTO_CLIENTES SET TIPAGEM_SANGUINEA  = V_TIPAGEM_SANGUINEA,  
                                                   NOME_RESPONSAVEL = V_NOME_RESPONSAVEL,
                                                   PARENTESCO_RESPONSAVEL = V_PARENTESCO_RESPONSAVEL
                    WHERE REGISTRO = V_REGISTRO;
                END IF;
                CLOSE CUR_PROCURA;

          ELSIF V_FUNCAO = 5 THEN
                UPDATE COMPLEMENTO_CLIENTES SET IMPRESSAO_DIGITAL = V_IMPRESSAO_DIGITAL
                    WHERE REGISTRO = V_REGISTRO;
          END IF;

     END;
     COMMIT WORK;
END PRC_COMPLEMENTO_CLIENTES;
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_PREPAROS(V_MNEMONICO           IN OUT PREPAROS.MNEMONICO%TYPE,
                                         V_PREPARO             IN PREPAROS.PREPARO%TYPE,
                                         V_INFORMACAO_TECNICA  IN PREPAROS.INFORMACAO_TECNICA%TYPE,
                                         V_INFORMACAO_CLIENTE  IN PREPAROS.INFORMACAO_CLIENTE%TYPE,
                                         V_FUNCAO        IN NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO PREPAROS VALUES (V_MNEMONICO, V_PREPARO, EMPTY_BLOB(), EMPTY_BLOB()); 

  ELSIF V_FUNCAO = 2 THEN
     UPDATE PREPAROS SET PREPARO = V_PREPARO
                        WHERE MNEMONICO = V_MNEMONICO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE PREPAROS WHERE MNEMONICO = V_MNEMONICO;

  END IF;

  COMMIT WORK;

END PRC_PREPAROS;

CREATE OR REPLACE PROCEDURE PRC_TRANSACOES_INTEGRACOES(V_ID_TRANSACAO_INTEGRACAO IN OUT TRANSACOES_INTEGRACOES.ID_TRANSACAO_INTEGRACAO%TYPE,
                                            V_ID_INTEGRACAO         IN TRANSACOES_INTEGRACOES.ID_INTEGRACAO%TYPE,
                                            V_DATA_TRANSACAO       IN TRANSACOES_INTEGRACOES.DATA_TRANSACAO%TYPE,
                                            V_STATUS_TRANSACAO       IN TRANSACOES_INTEGRACOES.STATUS_TRANSACAO%TYPE,
                                            V_TRANSACAO       IN TRANSACOES_INTEGRACOES.TRANSACAO%TYPE,
                                            V_ESTACAO_TRANSACAO       IN TRANSACOES_INTEGRACOES.ESTACAO_TRANSACAO%TYPE,
                                            V_TIPO_INTEGRACAO       IN TRANSACOES_INTEGRACOES.TIPO_INTEGRACAO%TYPE,
                                            V_PAI       IN TRANSACOES_INTEGRACOES.PAI%TYPE,
                                            V_RESPOSTA       IN TRANSACOES_INTEGRACOES.RESPOSTA%TYPE,
                                            V_FUNCAO       IN NUMBER) IS      
BEGIN
  IF V_FUNCAO = 1 THEN
	  SELECT SQ_TRANSACOES_INTEGRACOES.NEXTVAL INTO V_ID_TRANSACAO_INTEGRACAO FROM DUAL;
	 
    INSERT INTO TRANSACOES_INTEGRACOES VALUES (V_ID_TRANSACAO_INTEGRACAO, V_ID_INTEGRACAO, V_DATA_TRANSACAO, 
                                               V_STATUS_TRANSACAO, V_TRANSACAO, V_ESTACAO_TRANSACAO, V_TIPO_INTEGRACAO,
                                               V_PAI, V_RESPOSTA);           
  
  ELSIF V_FUNCAO = 2 THEN
    UPDATE TRANSACOES_INTEGRACOES SET ID_INTEGRACAO = V_ID_INTEGRACAO, DATA_TRANSACAO = V_DATA_TRANSACAO, 
                                      STATUS_TRANSACAO = V_STATUS_TRANSACAO, TRANSACAO = V_TRANSACAO, ESTACAO_TRANSACAO = V_ESTACAO_TRANSACAO,
                                      TIPO_INTEGRACAO = V_TIPO_INTEGRACAO, PAI = V_PAI, RESPOSTA = V_RESPOSTA
                                  WHERE ID_TRANSACAO_INTEGRACAO = V_ID_TRANSACAO_INTEGRACAO;	
  
  ELSIF V_FUNCAO = 3 THEN
    DELETE TRANSACOES_INTEGRACOES WHERE ID_TRANSACAO_INTEGRACAO = V_ID_TRANSACAO_INTEGRACAO;
  END IF;
  
END PRC_TRANSACOES_INTEGRACOES;

CREATE OR REPLACE PROCEDURE PRC_POSTOS_ENTIDADES(V_ID_POSTO       IN POSTOS_ENTIDADES.ID_POSTO%TYPE,
                                                 V_ID_ENTIDADE    IN POSTOS_ENTIDADES.ID_ENTIDADE%TYPE,
                                                 V_FUNCAO      IN      NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO POSTOS_ENTIDADES VALUES (V_ID_POSTO, V_ID_ENTIDADE);
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS_ENTIDADES WHERE ID_POSTO = V_ID_POSTO AND ID_ENTIDADE = V_ID_ENTIDADE;
  
  END IF;

  COMMIT WORK;

END PRC_POSTOS_ENTIDADES;
