
--
-- TERMOS
--
CREATE TABLE TERMOS (
 ID_TERMO     NUMBER(4)  NOT NULL,
 DESCRICAO    VARCHAR(80), 
 TERMO        BLOB,
 DATA_CAD     DATE          NOT NULL,
 USUARIO      VARCHAR2(10)  NOT NULL,
 TIPO_TERMO   VARCHAR2(4),
 STATUS_TERMO VARCHAR2(1)   NOT NULL
 );
-- fim bloco
 
 
ALTER TABLE TERMOS ADD CONSTRAINT TERMO_PK PRIMARY KEY (ID_TERMO);
-- fim bloco

CREATE SEQUENCE SQ_TERMOS
MINVALUE 0
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco


CREATE OR REPLACE PROCEDURE PRC_TERMOS(
                                        V_ID_TERMO      IN OUT TERMOS.ID_TERMO%TYPE,
                                        V_DESCRICAO     IN     TERMOS.DESCRICAO%TYPE,
                                        V_TERMO         IN     TERMOS.TERMO%TYPE,
                                        V_DATA_CAD      IN     TERMOS.DATA_CAD%TYPE,
										V_USUARIO       IN     TERMOS.USUARIO%TYPE,
										V_TIPO_TERMO    IN     TERMOS.TIPO_TERMO%TYPE,
										V_STATUS_TERMO  IN     TERMOS.STATUS_TERMO%TYPE,
                                        V_FUNCAO        IN NUMBER
                                        ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_TERMOS.NEXTVAL INTO V_ID_TERMO FROM DUAL;
        INSERT INTO TERMOS VALUES(V_ID_TERMO,  
		                          V_DESCRICAO,                
		                          EMPTY_BLOB(),
								  V_DATA_CAD,
								  V_USUARIO,
                                  V_TIPO_TERMO,
                                  V_STATUS_TERMO
								  );             		

    ELSIF V_FUNCAO = 2 THEN
        UPDATE TERMOS SET DESCRICAO    = V_DESCRICAO, 
						  TERMO        = V_TERMO,
						  DATA_CAD     = V_DATA_CAD,
					      USUARIO      = V_USUARIO,
						  TIPO_TERMO   = V_TIPO_TERMO,
						  STATUS_TERMO = V_STATUS_TERMO
                    WHERE ID_TERMO = V_ID_TERMO;

    ELSIF V_FUNCAO = 3 THEN
        DELETE TERMOS WHERE ID_TERMO = V_ID_TERMO;

    ELSIF V_FUNCAO = 4 THEN
        UPDATE TERMOS SET TERMO        = V_TERMO,
						  DATA_CAD     = V_DATA_CAD,
					      USUARIO      = V_USUARIO
                    WHERE ID_TERMO = V_ID_TERMO;
    END IF;

    COMMIT WORK;
END PRC_TERMOS;

-- fim bloco

ALTER TABLE TERMOS_PROCEDIMENTOS ADD CONSTRAINT TERMOS_PROCED_TERMOS_FK FOREIGN KEY (ID_TERMO) REFERENCES TERMOS(ID_TERMO);
-- fim bloco

CREATE SEQUENCE SQ_TERMOS_PROCEDIMENTOS
MINVALUE 0
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

create or replace
PROCEDURE PRC_TERMOS_PROCEDIMENTOS(V_MNEMONICO  IN  TERMOS_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                     V_ID_TERMO              IN     TERMOS_PROCEDIMENTOS.ID_TERMO%TYPE,
                                                     V_DATA_CAD              IN     TERMOS_PROCEDIMENTOS.DATA_CAD%TYPE,
                                                     V_USUARIO               IN     TERMOS_PROCEDIMENTOS.USUARIO%TYPE,
                                                     V_ID_TERMO_PROCEDIMENTO IN OUT TERMOS_PROCEDIMENTOS.ID_TERMO_PROCEDIMENTO%TYPE,
                                                     V_FUNCAO                IN  NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_TERMOS_PROCEDIMENTOS.NEXTVAL INTO V_ID_TERMO_PROCEDIMENTO FROM DUAL;
     
     INSERT INTO TERMOS_PROCEDIMENTOS VALUES (V_MNEMONICO, V_ID_TERMO, V_DATA_CAD, V_USUARIO, V_ID_TERMO_PROCEDIMENTO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE TERMOS_PROCEDIMENTOS SET ID_TERMO = V_ID_TERMO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
                                            WHERE ID_TERMO_PROCEDIMENTO = V_ID_TERMO_PROCEDIMENTO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE TERMOS_PROCEDIMENTOS WHERE ID_TERMO_PROCEDIMENTO = V_ID_TERMO_PROCEDIMENTO;

  END IF;

  COMMIT WORK;

END PRC_TERMOS_PROCEDIMENTOS;
