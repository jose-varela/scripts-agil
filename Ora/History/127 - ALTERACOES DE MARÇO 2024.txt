
--
-- TERMOS
--
CREATE TABLE TERMOS (
 ID_TERMO     NUMBER(4)  NOT NULL,
 DESCRICAO    VARCHAR(80), 
 TERMO        BLOB,
 DATA_CAD     DATE          NOT NULL,
 USUARIO      VARCHAR2(10)  NOT NULL,
 TIPO_TERMO   VARCHAR2(4),
 STATUS_TERMO VARCHAR2(1)   NOT NULL
 );
-- fim bloco
 
 
ALTER TABLE TERMOS ADD CONSTRAINT TERMO_PK PRIMARY KEY (ID_TERMO);
-- fim bloco

CREATE SEQUENCE SQ_TERMOS
MINVALUE 0
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco


CREATE OR REPLACE PROCEDURE PRC_TERMOS(
                                        V_ID_TERMO      IN OUT TERMOS.ID_TERMO%TYPE,
                                        V_DESCRICAO     IN     TERMOS.DESCRICAO%TYPE,
                                        V_TERMO         IN     TERMOS.TERMO%TYPE,
                                        V_DATA_CAD      IN     TERMOS.DATA_CAD%TYPE,
										V_USUARIO       IN     TERMOS.USUARIO%TYPE,
										V_TIPO_TERMO    IN     TERMOS.TIPO_TERMO%TYPE,
										V_STATUS_TERMO  IN     TERMOS.STATUS_TERMO%TYPE,
                                        V_FUNCAO        IN NUMBER
                                        ) IS
BEGIN
    IF V_FUNCAO = 1 THEN
        SELECT SQ_TERMOS.NEXTVAL INTO V_ID_TERMO FROM DUAL;
        INSERT INTO TERMOS VALUES(V_ID_TERMO,  
		                          V_DESCRICAO,                
		                          EMPTY_BLOB(),
								  V_DATA_CAD,
								  V_USUARIO,
                                  V_TIPO_TERMO,
                                  V_STATUS_TERMO
								  );             		

    ELSIF V_FUNCAO = 2 THEN
        UPDATE TERMOS SET DESCRICAO    = V_DESCRICAO, 
						  TERMO        = V_TERMO,
						  DATA_CAD     = V_DATA_CAD,
					      USUARIO      = V_USUARIO,
						  TIPO_TERMO   = V_TIPO_TERMO,
						  STATUS_TERMO = V_STATUS_TERMO
                    WHERE ID_TERMO = V_ID_TERMO;

    ELSIF V_FUNCAO = 3 THEN
        DELETE TERMOS WHERE ID_TERMO = V_ID_TERMO;

    ELSIF V_FUNCAO = 4 THEN
        UPDATE TERMOS SET TERMO        = V_TERMO,
						  DATA_CAD     = V_DATA_CAD,
					      USUARIO      = V_USUARIO
                    WHERE ID_TERMO = V_ID_TERMO;
    END IF;

    COMMIT WORK;
END PRC_TERMOS;

create or replace
PROCEDURE PRC_TERMOS_PROCEDIMENTOS(V_MNEMONICO  IN  TERMOS_PROCEDIMENTOS.MNEMONICO%TYPE,
                                                     V_ID_TERMO              IN     TERMOS_PROCEDIMENTOS.ID_TERMO%TYPE,
                                                     V_DATA_CAD              IN     TERMOS_PROCEDIMENTOS.DATA_CAD%TYPE,
                                                     V_USUARIO               IN     TERMOS_PROCEDIMENTOS.USUARIO%TYPE,
                                                     V_ID_TERMO_PROCEDIMENTO IN OUT TERMOS_PROCEDIMENTOS.ID_TERMO_PROCEDIMENTO%TYPE,
                                                     V_FUNCAO                IN  NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_TERMOS_PROCEDIMENTOS.NEXTVAL INTO V_ID_TERMO_PROCEDIMENTO FROM DUAL;
     
     INSERT INTO TERMOS_PROCEDIMENTOS VALUES (V_MNEMONICO, V_ID_TERMO, V_DATA_CAD, V_USUARIO, V_ID_TERMO_PROCEDIMENTO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE TERMOS_PROCEDIMENTOS SET ID_TERMO = V_ID_TERMO, DATA_CAD = V_DATA_CAD, USUARIO = V_USUARIO
                                            WHERE ID_TERMO_PROCEDIMENTO = V_ID_TERMO_PROCEDIMENTO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE TERMOS_PROCEDIMENTOS WHERE ID_TERMO_PROCEDIMENTO = V_ID_TERMO_PROCEDIMENTO;

  END IF;

  COMMIT WORK;

END PRC_TERMOS_PROCEDIMENTOS;

create or replace FUNCTION GET_COUNT_EXAMES_APOIO(V_POSTO          IN NUMBER,
                                                  V_ATENDIMENTO    IN NUMBER,
                                                  V_SITUACAO_APOIO IN VARCHAR2) RETURN NUMBER IS RESULT NUMBER;
BEGIN
  IF V_SITUACAO_APOIO = 'TEM QUE FAZER' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E, POSTOS P
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO <> 'C'
        AND E.SITUACAO <> 'F'
		AND P.POSTO = E.POSTO_REA
		AND P.TIPO_POSTO = 'A';
--        AND E.PENDENCIA <> 'S'
--        AND E.SITUACAO IN ('R','I','L')
--        AND E.SITUACAO_INTERFACE IN ('N','L','U','D','T','G');


  ELSIF V_SITUACAO_APOIO = 'TODOS' THEN
     SELECT COUNT(1) INTO RESULT FROM EXAMES E, POSTOS P
      WHERE E.POSTO = V_POSTO
        AND E.ATENDIMENTO = V_ATENDIMENTO
        AND E.SITUACAO <> 'C'
        AND E.SITUACAO <> 'F'
		AND P.POSTO = E.POSTO_REA
		AND P.TIPO_POSTO = 'A';

   END IF;
   RETURN(RESULT);
END GET_COUNT_EXAMES_APOIO;

CREATE VIEW ATENDIMENTOS_VALORES
AS SELECT A.POSTO, 
         A.ATENDIMENTO,
         NVL(CA.CONTA_HOSP,0) CONTA_HOSP,
         A.DATA_ATD,
         A.REGISTRO,
         A.CONVENIO,
         A.MATRICULA,
         A.CODIGO_PLANO,
         A.DATA_FAT,
         A.ID_MEDICO_SOLICITANTE,
         A.SITUACAO,
         A.LIB_FAT,
         A.FATURAMENTO,
         A.ID_FATURA,
         A.GUIA_PRINCIPAL,
         A.CARATER_ATENDIMENTO,
         A.IDENTIFICACAO_BENEFICIARIO,
         D.VALOR VALOR_DEVEDOR,
         SUM(C.VALOR) VALOR_PAGO,
         SUM(C.DESCONTO) VALOR_DESCONTO,
         GET_VALOR_EXAMES(A.POSTO, A.ATENDIMENTO) VALOR_EXAMES,
         GET_VALOR_EXAMES_CADASTRADOS(A.POSTO, A.ATENDIMENTO) VALOR_EXAMES_CADASTRADOS,
         GET_VALOR_EXAMES_FATURADOS(A.POSTO, A.ATENDIMENTO) VALOR_EXAMES_FATURADOS,
         GET_VALOR_PAGO_ATENDIMENTO(A.POSTO, A.ATENDIMENTO) VALOR_ATENDIMENTO,
         GET_COUNT_EXAMES_SITUACAO(A.POSTO, A.ATENDIMENTO, 'TODOS') QT_EXAMES,
         GET_COUNT_EXAMES_APOIO(A.POSTO, A.ATENDIMENTO, 'TODOS') QT_EXAMES_APOIO
    FROM ATENDIMENTOS A, 
         CONTAS_ATENDIMENTOS CA, 
         CAIXA C,
         DEVEDORES D
   WHERE CA.POSTO(+) = A.POSTO 
     AND CA.ATENDIMENTO(+) = A.ATENDIMENTO
     AND D.CONTA_HOSP(+) = CA.CONTA_HOSP
     AND C.CONTA_HOSP(+) = CA.CONTA_HOSP
     AND C.STATUS(+) = 'A'
   GROUP BY A.POSTO, 
            A.ATENDIMENTO,
            CA.CONTA_HOSP,
            A.DATA_ATD,
            A.REGISTRO,
            A.CONVENIO,
            A.MATRICULA,
            A.CODIGO_PLANO,
            A.DATA_FAT,
            A.ID_MEDICO_SOLICITANTE,
            A.SITUACAO,
            A.LIB_FAT,
            A.FATURAMENTO,
            A.ID_FATURA,
            A.GUIA_PRINCIPAL,
            A.CARATER_ATENDIMENTO,
            A.IDENTIFICACAO_BENEFICIARIO,
            D.VALOR;
