CREATE OR REPLACE FUNCTION GET_USUARIO_PRIMEIRA_COLETA(V_POSTO       IN NUMBER,
                                                       V_ATENDIMENTO IN NUMBER,
                                                       V_CORREL      IN NUMBER) RETURN VARCHAR2 IS RESULT VARCHAR2(20);
BEGIN
     SELECT AM.USUARIO_COLE INTO RESULT FROM AMOSTRAS AM
      WHERE AM.POSTO = V_POSTO
        AND AM.ATENDIMENTO = V_ATENDIMENTO
        AND AM.CORREL = V_CORREL
        AND AM.AMOSTRA = 1;
    RETURN(RESULT);
END GET_USUARIO_PRIMEIRA_COLETA;

create or replace FUNCTION IS_SITUACAO_EXAME(V_POSTO       IN NUMBER,
                                             V_ATENDIMENTO IN NUMBER,
                                             V_CORREL      IN NUMBER,
                                             V_SITUACAO    IN VARCHAR2) RETURN CHAR IS QT_SITUACAO NUMBER; 
BEGIN
    SELECT COUNT(1) INTO QT_SITUACAO FROM EXAMES_SITUACOES
     WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND CORREL = V_CORREL AND SITUACAO = V_SITUACAO;

    IF QT_SITUACAO = 0 THEN
       RETURN 'N';
    ELSE
       RETURN 'S';
    END IF;
END IS_SITUACAO_EXAME;

CREATE SEQUENCE SQ_RESPONSAVEIS_FINANCEIROS
MINVALUE 0 
MAXVALUE 99999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

CREATE TABLE RESPONSAVEIS_FINANCEIROS (
    ID_RESPONSAVEL_FINANCEIRO NUMBER(4) NOT NULL,
    RESPONSAVEL_FINANCEIRO VARCHAR2(50) NOT NULL,
    CPF_CNPJ VARCHAR2(15) NOT NULL,
    EMAIL VARCHAR2(25),
    TELEFONE VARCHAR2(23),
    TIPO VARCHAR2(2) NOT NULL,
    USUARIO VARCHAR2(10) NOT NULL,
    DATA_CAD DATE NOT NULL
)
TABLESPACE LIS_DATA;

    PRIMARY KEY (ID_RESPONSAVEL_FINANCEIRO)
    USING INDEX
    TABLESPACE LIS_INDX PCTFREE 20
    STORAGE (INITIAL 10K NEXT 10K PCTINCREASE 0);

CREATE OR REPLACE PROCEDURE PRC_RESPONSAVEIS_FINANCEIROS(V_ID_RESPONSAVEL_FINANCEIRO  IN OUT RESPONSAVEIS_FINANCEIROS.ID_RESPONSAVEL_FINANCEIRO%TYPE,
                                                         V_RESPONSAVEL_FINANCEIRO     IN RESPONSAVEIS_FINANCEIROS.RESPONSAVEL_FINANCEIRO%TYPE,
                                                         V_CPF_CNPJ                   IN RESPONSAVEIS_FINANCEIROS.CPF_CNPJ%TYPE,
                                                         V_EMAIL                      IN RESPONSAVEIS_FINANCEIROS.EMAIL%TYPE,
                                                         V_TELEFONE                   IN RESPONSAVEIS_FINANCEIROS.TELEFONE%TYPE,
                                                         V_TIPO                       IN RESPONSAVEIS_FINANCEIROS.TIPO%TYPE,
                                                         V_USUARIO                    IN RESPONSAVEIS_FINANCEIROS.USUARIO%TYPE,
                                                         V_DATA_CAD                   IN RESPONSAVEIS_FINANCEIROS.DATA_CAD%TYPE,
                                                         V_FUNCAO                     IN NUMBER) IS   
BEGIN
  IF V_FUNCAO = 1 THEN
	  SELECT SQ_RESPONSAVEIS_FINANCEIROS.NEXTVAL INTO V_ID_RESPONSAVEL_FINANCEIRO FROM DUAL;
	 
    INSERT INTO RESPONSAVEIS_FINANCEIROS VALUES (V_ID_RESPONSAVEL_FINANCEIRO, V_RESPONSAVEL_FINANCEIRO, 
                                                 V_CPF_CNPJ, V_EMAIL, V_TELEFONE, V_TIPO, V_USUARIO, V_DATA_CAD);           
  
  ELSIF V_FUNCAO = 2 THEN
     UPDATE RESPONSAVEIS_FINANCEIROS SET RESPONSAVEL_FINANCEIRO = V_RESPONSAVEL_FINANCEIRO, 
                                         CPF_CNPJ = V_CPF_CNPJ, EMAIL = V_EMAIL,
                                         TELEFONE = V_TELEFONE, TIPO = V_TIPO,
                                         USUARIO = V_USUARIO, DATA_CAD = V_DATA_CAD
                                     WHERE ID_RESPONSAVEL_FINANCEIRO = V_ID_RESPONSAVEL_FINANCEIRO;	
  
  ELSIF V_FUNCAO = 3 THEN
     DELETE RESPONSAVEIS_FINANCEIROS WHERE ID_RESPONSAVEL_FINANCEIRO = V_ID_RESPONSAVEL_FINANCEIRO;

  END IF;

  COMMIT WORK;

END PRC_RESPONSAVEIS_FINANCEIROS;

CREATE OR REPLACE PROCEDURE PRC_CAIXA(V_CONTA_HOSP     IN CAIXA.CONTA_HOSP%TYPE,
                                      V_REGISTRO       IN CAIXA.REGISTRO%TYPE,
                                      V_CONTA          IN CAIXA.CONTA%TYPE,
                                      V_DATA_PAG       IN CAIXA.DATA_PAG%TYPE,
                                      V_VALOR          IN CAIXA.VALOR%TYPE,
                                      V_DESCONTO       IN CAIXA.DESCONTO%TYPE,
                                      V_TIPO_DTO       IN CAIXA.TIPO_DTO%TYPE,
                                      V_USUARIO        IN CAIXA.USUARIO%TYPE,
                                      V_DATA_CAD       IN CAIXA.DATA_CAD%TYPE,
                                      V_CORREL         IN OUT CAIXA.CORREL%TYPE,
                                      V_ID_FORMA_PAGAMENTO       IN CAIXA.ID_FORMA_PAGAMENTO%TYPE,
                                      V_CATALOGO       IN CAIXA.CATALOGO%TYPE,
                                      V_USUARIO_DTO    IN CAIXA.USUARIO_DTO%TYPE,
                                      V_USUARIO_ALT    IN CAIXA.USUARIO_ALT%TYPE,
                                      V_DATA_ALT       IN CAIXA.DATA_ALT%TYPE,
                                      V_ID_BANDEIRA    IN CAIXA.ID_BANDEIRA%TYPE,
                                      V_ID_OPERADORA   IN CAIXA.ID_OPERADORA%TYPE,
                                      V_COD_PAGAMENTO  IN CAIXA.COD_PAGAMENTO%TYPE,
                                      V_PARCELAS       IN CAIXA.PARCELAS%TYPE,
                                      V_ID_CAIXA       IN OUT CAIXA.ID_CAIXA%TYPE,
                                      V_ID_RESPONSAVEL_FINANCEIRO IN CAIXA.ID_RESPONSAVEL_FINANCEIRO%TYPE,
                                      V_FUNCAO         IN NUMBER) IS
BEGIN
  IF (V_FUNCAO = 1) OR (V_FUNCAO = 6) THEN
  
     IF (V_FUNCAO = 1) THEN
        SELECT NVL(MAX(C.CORREL)+1,1) INTO V_CORREL FROM CAIXA C 
        WHERE C.CONTA_HOSP = V_CONTA_HOSP;

        SELECT SQ_CAIXA.NEXTVAL INTO V_ID_CAIXA FROM DUAL;
     END IF;

     INSERT INTO CAIXA VALUES (V_CONTA_HOSP,  V_REGISTRO,    V_CONTA,     V_DATA_PAG,
                               V_VALOR,       V_DESCONTO,    V_TIPO_DTO,  V_USUARIO,
                               V_DATA_CAD,    V_CORREL,      V_ID_FORMA_PAGAMENTO,  V_CATALOGO, 
   			                       V_USUARIO_DTO, V_USUARIO_ALT, V_DATA_ALT, V_ID_BANDEIRA,
                               V_ID_OPERADORA, V_COD_PAGAMENTO, V_PARCELAS, V_ID_CAIXA,
                               V_ID_RESPONSAVEL_FINANCEIRO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CAIXA SET VALOR = V_VALOR
                      WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CAIXA WHERE CONTA_HOSP = V_CONTA_HOSP;

  ELSIF V_FUNCAO = 4 THEN
     DELETE CAIXA WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE CAIXA SET VALOR = V_VALOR, 
  	                  DATA_PAG = V_DATA_PAG, 
                      DESCONTO = V_DESCONTO, 
                      TIPO_DTO = V_TIPO_DTO, 
                      ID_FORMA_PAGAMENTO = V_ID_FORMA_PAGAMENTO,
                      USUARIO_ALT = V_USUARIO_ALT,
					            DATA_ALT = V_DATA_ALT,
                      ID_BANDEIRA = V_ID_BANDEIRA,
                      ID_OPERADORA = V_ID_OPERADORA,
                      COD_PAGAMENTO = V_COD_PAGAMENTO,
                      PARCELAS = V_PARCELAS,
                      ID_RESPONSAVEL_FINANCEIRO = V_ID_RESPONSAVEL_FINANCEIRO
                 WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;
  END IF;

  COMMIT WORK;
END PRC_CAIXA;
