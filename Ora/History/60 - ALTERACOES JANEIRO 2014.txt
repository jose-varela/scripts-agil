-- criamos mapeamento para postos

CREATE TABLE MAP_B2B_POSTOS
(ID_SERVICO_B2B    VARCHAR2(6) NOT NULL,
 POSTO             NUMBER(3) NOT NULL,
 MAP_POSTO         VARCHAR2(10) NOT NULL)
TABLESPACE LIS_DATA;
-- fim bloco

ALTER TABLE MAP_B2B_POSTOS ADD CONSTRAINT MAP_B2B_POSTOS_PK 
      PRIMARY KEY (ID_SERVICO_B2B, POSTO);
-- fim bloco

CREATE INDEX MAP_B2B_POSTOS_I 
       ON MAP_B2B_POSTOS (ID_SERVICO_B2B, MAP_POSTO) 
       TABLESPACE LIS_INDX;
-- fim bloco
	  
ALTER TABLE MAP_B2B_POSTOS ADD CONSTRAINT MAP_B2B_POSTOS_SERV_B2B_FK 
      FOREIGN KEY (ID_SERVICO_B2B) REFERENCES SERVICOS_B2B(ID_SERVICO_B2B)
      ON DELETE CASCADE;
-- fim bloco
	  
ALTER TABLE MAP_B2B_POSTOS ADD CONSTRAINT MAP_B2B_POSTOS_POSTO_FK 
      FOREIGN KEY (POSTO) REFERENCES POSTOS(POSTO)
      ON DELETE CASCADE;
-- fim bloco
  	  

CREATE OR REPLACE PROCEDURE PRC_MAP_B2B_POSTOS(V_ID_SERVICO_B2B  IN   MAP_B2B_POSTOS.ID_SERVICO_B2B%TYPE,
                                                  V_POSTO        IN   MAP_B2B_POSTOS.POSTO%TYPE,
                                                  V_MAP_POSTO    IN   MAP_B2B_POSTOS.MAP_POSTO%TYPE,
                                                  V_FUNCAO       IN   NUMBER) IS                         
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO MAP_B2B_POSTOS VALUES (V_ID_SERVICO_B2B, V_POSTO, V_MAP_POSTO);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE MAP_B2B_POSTOS SET MAP_POSTO = V_MAP_POSTO WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND POSTO = V_POSTO;

  ELSIF V_FUNCAO = 3 THEN
     DELETE MAP_B2B_POSTOS WHERE ID_SERVICO_B2B = V_ID_SERVICO_B2B AND POSTO = V_POSTO;
     
  END IF;             

  COMMIT WORK;

END PRC_MAP_B2B_POSTOS;
