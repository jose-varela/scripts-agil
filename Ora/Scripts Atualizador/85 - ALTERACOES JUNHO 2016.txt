                                                    V_LOGIN             IN POSTOS_EQUIPAMENTOS.LOGIN%TYPE,
                                                    V_PASS              IN POSTOS_EQUIPAMENTOS.PASS%TYPE,
                                                    V_LOTE              IN POSTOS_EQUIPAMENTOS.LOTE%TYPE,
                                                    V_FUNCAO            IN      NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     INSERT INTO POSTOS_EQUIPAMENTOS VALUES (V_POSTO, V_EQUIPAMENTO, V_LOGIN, V_PASS, V_LOTE);
  
  ELSIF V_FUNCAO = 2 THEN
	 UPDATE POSTOS_EQUIPAMENTOS SET LOGIN = V_LOGIN, PASS = V_PASS, LOTE = V_LOTE
          WHERE POSTO = V_POSTO AND EQUIPAMENTO = V_EQUIPAMENTO;
	 
  ELSIF V_FUNCAO = 3 THEN
     DELETE POSTOS_EQUIPAMENTOS WHERE POSTO = V_POSTO AND EQUIPAMENTO = V_EQUIPAMENTO;
  
  ELSIF V_FUNCAO = 4 THEN
     UPDATE POSTOS_EQUIPAMENTOS SET LOTE = LOTE + 1 
      WHERE POSTO = V_POSTO AND EQUIPAMENTO = V_EQUIPAMENTO;

  END IF;

  COMMIT WORK;

END PRC_POSTOS_EQUIPAMENTOS;

create or replace
PROCEDURE PRC_ATENDIMENTOS_B2B(V_POSTO                   IN ATENDIMENTOS_B2B.POSTO%TYPE,
                                                 V_ATENDIMENTO             IN ATENDIMENTOS_B2B.ATENDIMENTO%TYPE,
                                                 V_EQUIPAMENTO          IN ATENDIMENTOS_B2B.EQUIPAMENTO%TYPE,
                                                 V_CODIGO_EXTERNO          IN ATENDIMENTOS_B2B.CODIGO_EXTERNO%TYPE,
                                                 V_DATA_B2B                IN ATENDIMENTOS_B2B.DATA_B2B%TYPE,
                                                 V_SITUACAO_B2B            IN ATENDIMENTOS_B2B.SITUACAO_B2B%TYPE,
                                                 V_FUNCAO                  IN NUMBER) IS
BEGIN
     DECLARE
            V_ACHEI  NUMBER(8);
            
            CURSOR CUR_PROCURA IS
                   SELECT ATENDIMENTO FROM ATENDIMENTOS_B2B
                    WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND EQUIPAMENTO = V_EQUIPAMENTO;
      BEGIN
           IF V_FUNCAO = 1 THEN
              INSERT INTO ATENDIMENTOS_B2B VALUES (V_POSTO, V_ATENDIMENTO, V_EQUIPAMENTO, V_CODIGO_EXTERNO, V_DATA_B2B, V_SITUACAO_B2B);

           ELSIF V_FUNCAO = 2 THEN
              UPDATE ATENDIMENTOS_B2B SET CODIGO_EXTERNO = V_CODIGO_EXTERNO, SITUACAO_B2B = V_SITUACAO_B2B, DATA_B2B = V_DATA_B2B
	                                WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND EQUIPAMENTO = V_EQUIPAMENTO;

           ELSIF V_FUNCAO = 3 THEN
              DELETE ATENDIMENTOS_B2B WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND EQUIPAMENTO = V_EQUIPAMENTO;
	 
           ELSIF V_FUNCAO = 4 THEN
                OPEN CUR_PROCURA;
                FETCH CUR_PROCURA INTO V_ACHEI;
  
                IF CUR_PROCURA%NOTFOUND THEN
                   INSERT INTO ATENDIMENTOS_B2B VALUES (V_POSTO, V_ATENDIMENTO, V_EQUIPAMENTO, V_CODIGO_EXTERNO, V_DATA_B2B, V_SITUACAO_B2B);
				ELSE   
                    UPDATE ATENDIMENTOS_B2B SET CODIGO_EXTERNO = V_CODIGO_EXTERNO, SITUACAO_B2B = V_SITUACAO_B2B, DATA_B2B = V_DATA_B2B
	                                      WHERE POSTO = V_POSTO AND ATENDIMENTO = V_ATENDIMENTO AND EQUIPAMENTO = V_EQUIPAMENTO;
                END IF;
           END IF;
      END;
	  
      COMMIT WORK;
END PRC_ATENDIMENTOS_B2B;

PROCEDURE PRC_EXAMES_B2B(V_ID_EXAME_B2B           IN OUT EXAMES_B2B.ID_EXAME_B2B%TYPE,
                                           V_POSTO                  IN EXAMES_B2B.POSTO%TYPE,
                                           V_ATENDIMENTO            IN EXAMES_B2B.ATENDIMENTO%TYPE,
                                           V_CORREL                 IN EXAMES_B2B.CORREL%TYPE,
                                           V_AMOSTRA                IN EXAMES_B2B.AMOSTRA%TYPE,
                                           V_POSTO_B2B              IN EXAMES_B2B.POSTO_B2B%TYPE,
                                           V_EQUIPAMENTO            IN EXAMES_B2B.EQUIPAMENTO%TYPE,
                                           V_CORREL_B2B             IN EXAMES_B2B.CORREL_B2B%TYPE,
                                           V_SITUACAO_B2B           IN EXAMES_B2B.SITUACAO_B2B%TYPE,
                                           V_DATA_B2B               IN EXAMES_B2B.DATA_B2B%TYPE,
                                           V_USUARIO_B2B            IN EXAMES_B2B.USUARIO_B2B%TYPE,
                                           V_ESTACAO_B2B            IN EXAMES_B2B.ESTACAO_B2B%TYPE,
                                           V_LOTE_B2B               IN EXAMES_B2B.LOTE_B2B%TYPE,
				                    	   V_CODIGO_EXTERNO         IN EXAMES_B2B.CODIGO_EXTERNO%TYPE,
					                       V_ETIQUETA_EXTERNO       IN EXAMES_B2B.ETIQUETA_EXTERNO%TYPE,
                                           V_FUNCAO                 IN NUMBER) IS
BEGIN                                                           
  IF V_FUNCAO = 1 THEN                                          
     SELECT SQ_EXAME_B2B.NEXTVAL INTO V_ID_EXAME_B2B FROM DUAL;
     INSERT INTO EXAMES_B2B VALUES(V_ID_EXAME_B2B, V_POSTO, V_ATENDIMENTO, V_CORREL, V_AMOSTRA, V_POSTO_B2B, 
                                   V_EQUIPAMENTO , V_CORREL_B2B, V_SITUACAO_B2B, V_DATA_B2B, 
                                   V_USUARIO_B2B, V_ESTACAO_B2B, V_LOTE_B2B, V_CODIGO_EXTERNO,V_ETIQUETA_EXTERNO);

  ELSIF V_FUNCAO = 2 THEN
            UPDATE EXAMES_B2B SET POSTO                 = V_POSTO,
                                  ATENDIMENTO           = V_ATENDIMENTO,
                                  CORREL                = V_CORREL,
                                  AMOSTRA               = V_AMOSTRA,
                                  POSTO_B2B             = V_POSTO_B2B,
                                  EQUIPAMENTO           = V_EQUIPAMENTO ,
                                  CORREL_B2B            = V_CORREL_B2B,
                                  SITUACAO_B2B          = V_SITUACAO_B2B,
                                  DATA_B2B              = V_DATA_B2B,
                                  USUARIO_B2B           = V_USUARIO_B2B,
                                  ESTACAO_B2B           = V_ESTACAO_B2B,
								  CODIGO_EXTERNO = V_CODIGO_EXTERNO,
								  ETIQUETA_EXTERNO      = V_ETIQUETA_EXTERNO
                          WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;

  ELSIF V_FUNCAO = 3 THEN
            DELETE EXAMES_B2B WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;

  ELSIF V_FUNCAO = 4 THEN
            DELETE EXAMES_B2B WHERE POSTO = V_POSTO;
			
  ELSIF V_FUNCAO = 5 THEN
            UPDATE EXAMES_B2B SET SITUACAO_B2B          = V_SITUACAO_B2B,
                                  DATA_B2B              = V_DATA_B2B,
                                  USUARIO_B2B           = V_USUARIO_B2B,
                                  ESTACAO_B2B           = V_ESTACAO_B2B
                          WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;

  ELSIF V_FUNCAO = 6 THEN
			UPDATE EXAMES_B2B SET SITUACAO_B2B = V_SITUACAO_B2B
                          WHERE ID_EXAME_B2B = V_ID_EXAME_B2B;
  END IF;

  COMMIT WORK;
END PRC_EXAMES_B2B;

CREATE OR REPLACE FUNCTION GET_ULTIMO_EVOLUCAO_RESULTADO(V_POSTO       IN NUMBER,
                                                         V_ATENDIMENTO IN NUMBER,
                                                         V_CORREL      IN NUMBER) RETURN DATE IS RESULT DATE;
BEGIN
     SELECT MAX(ER.DATA_EVOLUCAO) INTO RESULT FROM EVOLUCAO_RESULTADOS ER
      WHERE ER.POSTO = V_POSTO
        AND ER.ATENDIMENTO = V_ATENDIMENTO
        AND ER.CORREL = V_CORREL;
    RETURN(RESULT);
END GET_ULTIMO_EVOLUCAO_RESULTADO;

CREATE OR REPLACE PROCEDURE PRC_GRUPOS_ETIQUETAS(V_GRUPO_ETIQUETA         IN OUT GRUPOS_ETIQUETAS.GRUPO_ETIQUETA%TYPE,
                                                 V_NOME                   IN     GRUPOS_ETIQUETAS.NOME%TYPE,
                                                 V_MATERIAL               IN     GRUPOS_ETIQUETAS.MATERIAL%TYPE,
                                                 V_QT_ETIQUETAS           IN     GRUPOS_ETIQUETAS.QT_ETIQUETAS%TYPE,
                                                 V_SEPARA_ETIQUETAS       IN     GRUPOS_ETIQUETAS.SEPARA_ETIQUETAS%TYPE,
                                                 V_TIPO_CODIGO            IN     GRUPOS_ETIQUETAS.TIPO_CODIGO%TYPE,
                                                 V_MARGEM_ESQUERDA_CODIGO IN     GRUPOS_ETIQUETAS.MARGEM_ESQUERDA_CODIGO%TYPE,
                                                 V_RATIO_CODIGO           IN     GRUPOS_ETIQUETAS.RATIO_CODIGO%TYPE,
                                                 V_REALCA_NOME            IN     GRUPOS_ETIQUETAS.REALCA_NOME%TYPE,
                                                 V_IMPRESSORA             IN     GRUPOS_ETIQUETAS.IMPRESSORA%TYPE,
                                                 V_DIGITO_CODIGO          IN     GRUPOS_ETIQUETAS.DIGITO_CODIGO%TYPE,
												 V_QT_MAX_MNEMONICOS      IN     GRUPOS_ETIQUETAS.QT_MAX_MNEMONICOS%TYPE,
                                                 V_FUNCAO                 IN     NUMBER) IS
BEGIN
  IF V_FUNCAO = 1 THEN
     SELECT SQ_GRUPO_ETIQUETA.NEXTVAL INTO V_GRUPO_ETIQUETA FROM DUAL;
     INSERT INTO GRUPOS_ETIQUETAS VALUES (V_GRUPO_ETIQUETA, V_NOME, V_MATERIAL, V_QT_ETIQUETAS, V_SEPARA_ETIQUETAS, V_TIPO_CODIGO, V_MARGEM_ESQUERDA_CODIGO, 
                                          V_RATIO_CODIGO, V_REALCA_NOME, V_IMPRESSORA, V_DIGITO_CODIGO, V_QT_MAX_MNEMONICOS);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE GRUPOS_ETIQUETAS SET NOME = V_NOME, MATERIAL = V_MATERIAL, QT_ETIQUETAS = V_QT_ETIQUETAS, SEPARA_ETIQUETAS = V_SEPARA_ETIQUETAS,

            IMPRESSORA = V_IMPRESSORA, DIGITO_CODIGO = V_DIGITO_CODIGO, QT_MAX_MNEMONICOS = V_QT_MAX_MNEMONICOS
      WHERE GRUPO_ETIQUETA = V_GRUPO_ETIQUETA;

  ELSIF V_FUNCAO = 3 THEN
     DELETE GRUPOS_ETIQUETAS WHERE GRUPO_ETIQUETA = V_GRUPO_ETIQUETA;
  END IF;

  COMMIT WORK;

END PRC_GRUPOS_ETIQUETAS;
