CREATE TABLE TRANSACOES_INTEGRACOES(
    ID_TRANSACAO_INTEGRACAO NUMBER(9) NOT NULL,
    ID_INTEGRACAO NUMBER(2) NOT NULL,
    DATA_TRANSACAO DATE,
    STATUS_TRANSACAO VARCHAR2(1) NOT NULL,
    TRANSACAO CLOB NOT NULL,
    ESTACAO_TRANSACAO VARCHAR2(20) ,
    TIPO_INTEGRACAO VARCHAR2(25)
);
-- fim bloco

CREATE SEQUENCE SQ_TRANSACOES_INTEGRACOES
MINVALUE 0 
MAXVALUE 999999999
INCREMENT BY 1
START WITH 1
NOCACHE
NOORDER;
-- fim bloco

ALTER TABLE TRANSACOES_INTEGRACOES ADD CONSTRAINT TRANSACOES_INTEGRACOES_PK
    PRIMARY KEY (ID_TRANSACAO_INTEGRACAO)
    USING INDEX
    TABLESPACE LIS_INDX PCTFREE 20
    STORAGE (INITIAL 10K NEXT 10K PCTINCREASE 0);
-- fim bloco
  
ALTER TABLE TRANSACOES_INTEGRACOES ADD CONSTRAINT TRANS_INTEGRA_INTEGRACAO_FK
    FOREIGN KEY (ID_INTEGRACAO) REFERENCES EQUIPAMENTOS(EQUIPAMENTO);
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_TRANSACOES_INTEGRACOES(V_ID_TRANSACAO_INTEGRACAO IN OUT TRANSACOES_INTEGRACOES.ID_TRANSACAO_INTEGRACAO%TYPE,
                                            V_ID_INTEGRACAO         IN TRANSACOES_INTEGRACOES.ID_INTEGRACAO%TYPE,
                                            V_DATA_TRANSACAO       IN TRANSACOES_INTEGRACOES.DATA_TRANSACAO%TYPE,
                                            V_STATUS_TRANSACAO       IN TRANSACOES_INTEGRACOES.STATUS_TRANSACAO%TYPE,
                                            V_TRANSACAO       IN TRANSACOES_INTEGRACOES.TRANSACAO%TYPE,
                                            V_ESTACAO_TRANSACAO       IN TRANSACOES_INTEGRACOES.ESTACAO_TRANSACAO%TYPE,
                                            V_TIPO_INTEGRACAO       IN TRANSACOES_INTEGRACOES.TIPO_INTEGRACAO%TYPE,
                                            V_FUNCAO       IN NUMBER) IS      
BEGIN
  IF V_FUNCAO = 1 THEN
	  SELECT SQ_TRANSACOES_INTEGRACOES.NEXTVAL INTO V_ID_TRANSACAO_INTEGRACAO FROM DUAL;
	 
    INSERT INTO TRANSACOES_INTEGRACOES VALUES (V_ID_TRANSACAO_INTEGRACAO, V_ID_INTEGRACAO, V_DATA_TRANSACAO, 
                                               V_STATUS_TRANSACAO, V_TRANSACAO, V_ESTACAO_TRANSACAO, V_TIPO_INTEGRACAO);           
  
  ELSIF V_FUNCAO = 2 THEN
    UPDATE TRANSACOES_INTEGRACOES SET ID_INTEGRACAO = V_ID_INTEGRACAO, DATA_TRANSACAO = V_DATA_TRANSACAO, 
                                      STATUS_TRANSACAO = V_STATUS_TRANSACAO, TRANSACAO = V_TRANSACAO, ESTACAO_TRANSACAO = V_ESTACAO_TRANSACAO,
                                      TIPO_INTEGRACAO = V_TIPO_INTEGRACAO
                                  WHERE ID_TRANSACAO_INTEGRACAO = V_ID_TRANSACAO_INTEGRACAO;	
  
  ELSIF V_FUNCAO = 3 THEN
    DELETE TRANSACOES_INTEGRACOES WHERE ID_TRANSACAO_INTEGRACAO = V_ID_TRANSACAO_INTEGRACAO;

  END IF;

  COMMIT WORK;

END PRC_TRANSACOES_INTEGRACOES;

ALTER TABLE CAIXA ADD ID_CAIXA NUMBER(9);
-- fim bloco

ALTER TABLE CAIXA ADD CONSTRAINT CAIXA_UK UNIQUE (ID_CAIXA);
-- fim bloco

CREATE SEQUENCE SQ_CAIXA
MINVALUE 0 
MAXVALUE 999999999
INCREMENT BY 1
START WITH 0
NOCACHE
NOORDER;
-- fim bloco

UPDATE CAIXA SET ID_CAIXA = SQ_CAIXA.NEXTVAL;
-- fim bloco

CREATE OR REPLACE PROCEDURE PRC_CAIXA(V_CONTA_HOSP     IN CAIXA.CONTA_HOSP%TYPE,
                                      V_REGISTRO       IN CAIXA.REGISTRO%TYPE,
                                      V_CONTA          IN CAIXA.CONTA%TYPE,
                                      V_DATA_PAG       IN CAIXA.DATA_PAG%TYPE,
                                      V_VALOR          IN CAIXA.VALOR%TYPE,
                                      V_DESCONTO       IN CAIXA.DESCONTO%TYPE,
                                      V_TIPO_DTO       IN CAIXA.TIPO_DTO%TYPE,
                                      V_USUARIO        IN CAIXA.USUARIO%TYPE,
                                      V_DATA_CAD       IN CAIXA.DATA_CAD%TYPE,
                                      V_CORREL         IN OUT CAIXA.CORREL%TYPE,
                                      V_ID_FORMA_PAGAMENTO       IN CAIXA.ID_FORMA_PAGAMENTO%TYPE,
                                      V_CATALOGO       IN CAIXA.CATALOGO%TYPE,
                                      V_USUARIO_DTO    IN CAIXA.USUARIO_DTO%TYPE,
                                      V_USUARIO_ALT    IN CAIXA.USUARIO_ALT%TYPE,
                                      V_DATA_ALT       IN CAIXA.DATA_ALT%TYPE,
                                      V_ID_BANDEIRA    IN CAIXA.ID_BANDEIRA%TYPE,
                                      V_ID_OPERADORA   IN CAIXA.ID_OPERADORA%TYPE,
                                      V_COD_PAGAMENTO  IN CAIXA.COD_PAGAMENTO%TYPE,
                                      V_PARCELAS       IN CAIXA.PARCELAS%TYPE,
                                      V_ID_CAIXA       IN OUT CAIXA.ID_CAIXA%TYPE,
                                      V_FUNCAO         IN NUMBER) IS
BEGIN
  IF (V_FUNCAO = 1) OR (V_FUNCAO = 6) THEN
  
     IF (V_FUNCAO = 1) THEN
        SELECT NVL(MAX(C.CORREL)+1,1) INTO V_CORREL FROM CAIXA C 
        WHERE C.CONTA_HOSP = V_CONTA_HOSP;

        SELECT SQ_CAIXA.NEXTVAL INTO V_ID_CAIXA FROM DUAL;
     END IF;

     INSERT INTO CAIXA VALUES (V_CONTA_HOSP,  V_REGISTRO,    V_CONTA,     V_DATA_PAG,
                               V_VALOR,       V_DESCONTO,    V_TIPO_DTO,  V_USUARIO,
                               V_DATA_CAD,    V_CORREL,      V_ID_FORMA_PAGAMENTO,  V_CATALOGO, 
   			       V_USUARIO_DTO, V_USUARIO_ALT, V_DATA_ALT, V_ID_BANDEIRA, V_ID_OPERADORA, V_COD_PAGAMENTO, V_PARCELAS, V_ID_CAIXA);

  ELSIF V_FUNCAO = 2 THEN
     UPDATE CAIXA SET VALOR = V_VALOR
                      WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 3 THEN
     DELETE CAIXA WHERE CONTA_HOSP = V_CONTA_HOSP;

  ELSIF V_FUNCAO = 4 THEN
     DELETE CAIXA WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;

  ELSIF V_FUNCAO = 5 THEN
     UPDATE CAIXA SET VALOR = V_VALOR, 
  	                  DATA_PAG = V_DATA_PAG, 
                      DESCONTO = V_DESCONTO, 
                      TIPO_DTO = V_TIPO_DTO, 
                      ID_FORMA_PAGAMENTO = V_ID_FORMA_PAGAMENTO,
                      USUARIO_ALT = V_USUARIO_ALT,
					            DATA_ALT = V_DATA_ALT,
                      ID_BANDEIRA = V_ID_BANDEIRA,
                      ID_OPERADORA = V_ID_OPERADORA,
                      COD_PAGAMENTO = V_COD_PAGAMENTO,
                      PARCELAS = V_PARCELAS
                WHERE CONTA_HOSP = V_CONTA_HOSP AND REGISTRO = V_REGISTRO AND CORREL = V_CORREL;
  END IF;

  COMMIT WORK;
END PRC_CAIXA;
